# ប្រធានបទ 2.1 — កាតពិន្ទុតម្លៃសាធារណៈ និងក្របខ័ណ្ឌលទ្ធផល

**យុទ្ធសាស្ត្រដែលបង្កើតឡើងលើត្រីកោណយុទ្ធសាស្ត្រ នៅតែជាគ្រាន់តែរឿងនៃបំណងប្រាថ្នា រហូតដល់អង្គការមួយប្តេជ្ញាវាស់វែងជាសាធារណៈថា តើខ្លួនបានបង្កើតភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាពពិតប្រាកដឬទេ ហើយកាតពិន្ទុតម្លៃសាធារណៈ គំរូតក្កវិទ្យា និងគណនេយ្យភាពផ្អែកលើលទ្ធផល គឺជាវិធីមានវិន័យដើម្បីធ្វើឱ្យការប្តេជ្ញានោះមាននិរន្តរភាព។**

## ហេតុអ្វីបានជាបញ្ហានេះសំខាន់សម្រាប់តម្លៃសាធារណៈ

អ្នកដឹកនាំវិស័យសាធារណៈ និងវិស័យសង្គមភាគច្រើនអាចរៀបចំឯកសារយុទ្ធសាស្ត្រដែលនិយាយអំពីលទ្ធផលបានយ៉ាងស្ទាត់ជំនាញ។ ប៉ុន្តែមានអ្នកតិចតួចណាស់ដែលអាចបង្ហាញភស្តុតាងនៅមួយឆ្នាំក្រោយ ថាតើលទ្ធផលទាំងនោះបានផ្លាស់ប្តូរពិតប្រាកដឬទេ។ គម្លាតរវាងរឿងទាំងពីរនេះ ជាធម្មតាមិនមែនកើតពីកង្វះទិន្នន័យទេ ព្រោះអង្គការភាគច្រើនប្រមូលទិន្នន័យក្នុងបរិមាណដ៏ច្រើនសន្ធឹកសន្ធាប់ ប៉ុន្តែកើតពីកង្វះរចនាសម្ព័ន្ធមានវិន័យ ដែលបំប្លែងចំនួនសកម្មភាពដែលមិនទាន់ដំណើរការ ទៅជាការអះអាងដែលអាចការពារបានអំពីតម្លៃដែលបានបង្កើត។ ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព បានកំណត់ភាពស្របច្បាប់ និងការគាំទ្រ សំណើតម្លៃ និងសមត្ថភាពប្រតិបត្តិការ ជាជើងទាំងបីដែលយុទ្ធសាស្ត្ររបស់អង្គការត្រូវបំពេញព្រមគ្នា (សូមមើល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព)។ ប្រធានបទនេះនិយាយអំពីឧបករណ៍ដែលធ្វើឱ្យជើងទាំងបីនោះអាចវាស់វែងបាន ជាជាងគ្រាន់តែអះអាង។

ភាពចាំបាច់នេះកើតមានភ្លាមៗ និងកើតឡើងជាបន្តបន្ទាប់។ អ្នកផ្តល់ថវិកា ឬសភានីតិបញ្ញត្តិសួរនៅជុំថវិកាបន្ទាប់ថា "តើយើងទទួលបានអ្វីពីប្រាក់នេះ" ហើយអង្គការដែលមានតែចំនួនផលិតផល (គ្លីនិកដែលបានបើក សំណុំរឿងដែលបានដោះស្រាយ វគ្គបណ្តុះបណ្តាលដែលបានរៀបចំ) គ្មានចម្លើយចំពោះសំណួរអំពីលទ្ធផលដែលនៅពីក្រោមនោះទេ។ ក្រុមកម្មវិធីមួយជឿថាខ្លួនកំពុងទទួលបានជោគជ័យ ព្រោះសកម្មភាពមានច្រើន និងមមាញឹក ដោយមិនបានញែកសកម្មភាពចេញពីការផ្លាស់ប្តូរក្នុងពិភពលោក ដែលសកម្មភាពនោះមានគោលបំណងបង្កើតឡើង។ ក្រុមប្រឹក្សាភិបាលអនុម័តគំនិតផ្តួចផ្តើមថ្មីមួយ ដោយផ្អែកលើសាច់រឿងដែលមានអំណាចបញ្ចុះបញ្ចូល រួចមួយឆ្នាំក្រោយមកគ្មានមូលដ្ឋានដែលបានព្រមព្រៀងគ្នាសម្រាប់សម្រេចថាត្រូវបន្ត ពង្រីក ឬបញ្ឈប់វា។ ករណីទាំងនេះសុទ្ធតែជាការបរាជ័យនៃការវាស់វែងដែលមានផលវិបាកជាយុទ្ធសាស្ត្រ មិនមែនជាភាពរអាក់រអួលបច្ចេកទេសទេ។

ឧបករណ៍ក្នុងប្រធានបទនេះ គឺកាតពិន្ទុតម្លៃសាធារណៈរបស់ Mark Moore គំរូតក្កវិទ្យា និងគណនេយ្យភាពផ្អែកលើលទ្ធផល រួមទាំងក្របខ័ណ្ឌដែលពាក់ព័ន្ធ ដែលមិនដោះស្រាយបញ្ហាវាស់វែងទាំងអស់ដែលអង្គការវិស័យសាធារណៈ ឬវិស័យសង្គមជួបប្រទះឡើយ។ ប្រធានបទបន្តបន្ទាប់ដោះស្រាយស្រទាប់ដែលពិបាក និងមានលក្ខណៈបច្ចេកទេសជាងនេះ គឺការបំប្លែងលទ្ធផលទៅជាតម្លៃរូបិយវត្ថុតាមរយៈការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍ និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម (សូមមើល ប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម) ការវាយតម្លៃលទ្ធផលដែលពិបាកបំប្លែងទៅជារូបិយវត្ថុ តាមរយៈផលចំណេញសង្គមពីការវិនិយោគ (សូមមើល ប្រធានបទ 2.3 — ផលចំណេញសង្គមពីការវិនិយោគ និងការវាស់វែងផលប៉ះពាល់) និងការបង្ហាញថាកម្មវិធីមួយបានបង្កឱ្យមានការផ្លាស់ប្តូរដែលសង្កេតឃើញពិតប្រាកដ តាមរយៈ[ការវាយតម្លៃកម្មវិធី](https://en.wikipedia.org/wiki/Program_evaluation)ជាផ្លូវការ (សូមមើល ប្រធានបទ 2.4 — សេដ្ឋមាត្រនៃវិស័យសាធារណៈ និងការវាយតម្លៃកម្មវិធី)។ អ្វីដែលប្រធានបទនេះផ្តល់ជូន គឺស្រទាប់នៅពីក្រោមទាំងបី៖ វាក្យសព្ទរួម និងក្របខ័ណ្ឌប្រើប្រាស់បានយូរអង្វែងមួយចំនួនតូច សម្រាប់បញ្ជាក់ មុនពេលចាប់ផ្តើមការងារបច្ចេកទេសទាំងនោះ ថាតើអង្គការពិតជាព្យាយាមផ្លាស់ប្តូរអ្វី សម្រាប់អ្នកណា ហើយនឹងដឹងដោយរបៀបណា។

## គំនិតស្នូល

**កាតពិន្ទុតម្លៃសាធារណៈ** គឺជាសំណើរបស់ [Mark H. Moore](https://en.wikipedia.org/wiki/Mark_H._Moore) ក្នុងការតាមដានជើងនីមួយៗនៃត្រីកោណយុទ្ធសាស្ត្រដោយប្រើសំណុំសូចនាករផ្ទាល់ខ្លួន ជាជាងបែរទៅប្រើវិធានការហិរញ្ញវត្ថុ និងប្រតិបត្តិការតាមលំនាំដើម ដូចដែល[កាតពិន្ទុតុល្យភាព](https://en.wikipedia.org/wiki/Balanced_scorecard)នៃវិស័យឯកជនប្រើ។ Moore បានកែសម្រួលកាតពិន្ទុតុល្យភាព ដែលជាឧបករណ៍របស់ Robert Kaplan និង David Norton សម្រាប់តាមដានអាជីវកម្មតាមទស្សនៈហិរញ្ញវត្ថុ អតិថិជន ដំណើរការផ្ទៃក្នុង និងការរៀនសូត្រ ព្រោះលទ្ធផលហិរញ្ញវត្ថុតែមួយមិនអាចតំណាងឱ្យអង្គការដែលហេតុផលស្នូលនៃអត្ថិភាពរបស់ខ្លួន គឺការបង្កើតតម្លៃដែលតម្លៃទីផ្សារមិនអាចចាប់យកបាននោះទេ។ ផ្ទុយទៅវិញ កាតពិន្ទុតម្លៃសាធារណៈសុំឱ្យអង្គការកំណត់វិធានការក្រោមក្បាលរឿងបីដែលឆ្លុះបញ្ចាំងត្រីកោណ គឺគណនីភាពស្របច្បាប់ និងការគាំទ្រ (តើបរិយាកាសផ្តល់សិទ្ធិ ដូចជាអ្នកផ្តល់ថវិកា អ្នកត្រួតពិនិត្យ និងសាធារណជន ពេញចិត្តឬទេ) គណនីតម្លៃសាធារណៈ (តើសំណើតម្លៃកំពុងត្រូវបានសម្រេចឬទេ គឺលទ្ធផលសម្រាប់ពលរដ្ឋ និងអតិថិជន) និងគណនីសមត្ថភាពប្រតិបត្តិការ (តើអង្គការអាចរក្សា និងកែលម្អការផ្តល់សេវាបានឬទេ គឺបុគ្គលិក ប្រព័ន្ធ ហិរញ្ញវត្ថុ និងភាពជាដៃគូ)។ វិន័យស្ថិតនៅត្រង់ការមិនអនុញ្ញាតឱ្យគណនីណាមួយតំណាងឱ្យទាំងមូល៖ អង្គការដែលរាយការណ៍តែប្រសិទ្ធភាពហិរញ្ញវត្ថុ ឬតែការពេញចិត្តរបស់អតិថិជន មិនទាន់បានរាយការណ៍ធៀបនឹងត្រីកោណសោះឡើយ។

**គំរូតក្កវិទ្យា** ផ្តល់រចនាសម្ព័ន្ធខាងក្នុងដល់គណនីតម្លៃនៃកាតពិន្ទុ។ [គំរូតក្កវិទ្យា](https://en.wikipedia.org/wiki/Logic_model) បង្ហាញខ្សែសង្វាក់មូលហេតុដែលកម្មវិធីអះអាងថាខ្លួនដំណើរការ គឺធាតុចូល (ប្រាក់ បុគ្គលិក និងធនធានផ្សេងទៀតដែលបានប្តេជ្ញា) សកម្មភាព (អ្វីដែលអង្គការធ្វើជាមួយធនធានទាំងនោះ) ផលិតផល (ផលដែលអាចរាប់បានភ្លាមៗនៃសកម្មភាពទាំងនោះ ដូចជាវគ្គដែលបានផ្តល់ ប័ណ្ណអនុញ្ញាតដែលបានចេញ គ្រួសារដែលបានទៅជួប) លទ្ធផល (ការផ្លាស់ប្តូរនៃចំណេះដឹង អាកប្បកិរិយា ស្ថានភាព ឬឋានៈ ដែលផលិតផលមានគោលបំណងបង្កើត ជាទូទៅក្នុងរយៈពេលប៉ុន្មានខែ ឬពីរបីឆ្នាំ) និងផលប៉ះពាល់ (ការផ្លាស់ប្តូរក្នុងរយៈពេលវែងជាង ដែលភាគច្រើនមានលក្ខណៈកម្រិតប្រជាជន ហើយលទ្ធផលមានគោលបំណងចូលរួមចំណែក ដែលជាទូទៅចែករំលែកជាមួយអ្នកដើរតួផ្សេងទៀត ហើយពិបាកបញ្ជាក់ថាជាលទ្ធផលរបស់កម្មវិធីណាមួយ)។ ភាពខុសគ្នារវាងផលិតផល និងលទ្ធផល គឺជាចំណុចមានផលវិបាកបំផុតតែមួយគត់ក្នុងប្រធានបទនេះ៖ ផលិតផលគឺជាអ្វីដែលអង្គការបានធ្វើ ឯលទ្ធផលគឺជាអ្វីដែលបានផ្លាស់ប្តូរសម្រាប់អ្នកផ្សេងដោយសារការនោះ។ ផលិតផលរបស់កម្មវិធីបណ្តុះបណ្តាលគឺចំនួនអ្នកដែលបានទទួលការបណ្តុះបណ្តាល ឯលទ្ធផលរបស់វាគឺអ្វីដែលអ្នកទាំងនោះអាចធ្វើបានបន្ទាប់មក ឬរបៀបដែលកាលៈទេសៈរបស់ពួកគេផ្លាស់ប្តូរ ដែលពួកគេមិនអាចធ្វើបានពីមុន។

**ទ្រឹស្តីនៃការផ្លាស់ប្តូរ** គឺជាវិន័យដែលជិតស្និទ្ធនឹងគំរូតក្កវិទ្យា ក្នុងការធ្វើឱ្យសម្មតិកម្មនៅពីក្រោយគំរូតក្កវិទ្យាច្បាស់លាស់ ជាជាងបង្កប់ន័យ គឺហេតុអ្វីបានជាអង្គការជឿថាសកម្មភាព A នឹងបង្កើតលទ្ធផល B ពិតប្រាកដ និងក្រោមលក្ខខណ្ឌណាដែលការអះអាងអំពីមូលហេតុនោះអាចបរាជ័យ។ [ទ្រឹស្តីនៃការផ្លាស់ប្តូរ](https://en.wikipedia.org/wiki/Theory_of_change) មិនមែនគ្រាន់តែជាគំរូតក្កវិទ្យាលម្អិតជាងនោះទេ ប៉ុន្តែវាសង្កត់ធ្ងន់លើហេតុផលបែប "ប្រសិនបើ... នោះ... ព្រោះ" និងសម្មតិកម្មតាមបរិបទ ដែលដ្យាក្រាមប្រអប់និងព្រួញរបស់គំរូតក្កវិទ្យាច្រើនតែទុកឱ្យមិនបានបញ្ជាក់។ ប្រធានបទ 2.3 ប្រើទ្រឹស្តីនៃការផ្លាស់ប្តូរជាចំណុចចាប់ផ្តើមសម្រាប់ការវាស់វែងផលប៉ះពាល់ក្នុងវិស័យសង្គមកាន់តែស៊ីជម្រៅ (សូមមើល ប្រធានបទ 2.3 — ផលចំណេញសង្គមពីការវិនិយោគ និងការវាស់វែងផលប៉ះពាល់) ឯនៅទីនេះ វាសំខាន់ជាចម្បងក្នុងនាមជាវិន័យដែលរក្សាឱ្យគំរូតក្កវិទ្យាស្មោះត្រង់អំពីអ្វីដែលវាអះអាងថាបង្កឱ្យមានពិតប្រាកដ ខុសពីអ្វីដែលវាគ្រាន់តែសង្ឃឹមថានឹងកើតមាន។

**គណនេយ្យភាពផ្អែកលើលទ្ធផល** (RBA) គឺជាសមាជិកដែលល្បីបំផុតនៃក្រុមក្របខ័ណ្ឌផ្អែកលើលទ្ធផល ដែលប្រើប្រាស់ទូទាំងរដ្ឋាភិបាល និងវិស័យសង្គម ដើម្បីភ្ជាប់លទ្ធផលកម្រិតប្រជាជនទៅនឹងការអនុវត្តរបស់កម្មវិធីជាក់លាក់។ RBA ត្រូវបានអភិវឌ្ឍដោយ Mark Friedman ហើយបានបង្ហាញក្នុងសៀវភៅរបស់គាត់ *Trying Hard Is Not Good Enough* ដោយញែកគណនេយ្យភាពពីរប្រភេទផ្សេងគ្នា ដែលអង្គការច្រើនតែច្រលំចូលគ្នា។ គណនេយ្យភាពប្រជាជនសួរថា តើសហគមន៍ ឬយុត្តាធិការទាំងមូលមានស្ថានភាពយ៉ាងណាលើលក្ខខណ្ឌនៃសុខុមាលភាព ដូចជាភាពក្រីក្រក្មេង សុវត្ថិភាពសហគមន៍ និងសមិទ្ធិផលអប់រំ ដោយប្រើសូចនាករចម្បងមួយចំនួនតូច ដែលគ្មានទីភ្នាក់ងារណាមួយជាម្ចាស់ ឬគ្រប់គ្រងបានទាំងស្រុងឡើយ។ គណនេយ្យភាពនៃការអនុវត្តសួរថា តើកម្មវិធី ឬទីភ្នាក់ងារជាក់លាក់មួយកំពុងធ្វើបានល្អប៉ុណ្ណា ដោយប្រើ[សូចនាករនៃការអនុវត្ត](https://en.wikipedia.org/wiki/Performance_indicator)បីប្រភេទ ដែលក្របខ័ណ្ឌរបស់ Friedman ទទូចឱ្យញែកចេញពីគ្នា គឺបានធ្វើអ្វីច្រើនប៉ុណ្ណា (កិច្ចខិតខំប្រឹងប្រែង ឬចំនួនផលិតផល) បានធ្វើល្អប៉ុណ្ណា (គុណភាព) និងថាតើមាននរណាម្នាក់មានស្ថានភាពប្រសើរឡើងឬទេ (ផល ឬលទ្ធផល)។ វិធីសាស្ត្រអនុវត្តជាក់ស្តែងរបស់ RBA ដែលជួនកាលហៅថាការវិភាគ "បត់ខ្សែកោង" ធ្វើការថយក្រោយពីសូចនាករដែលបានជ្រើសរើស ដោយសួរថា ទិន្នន័យនិយាយរឿងអ្វី អ្វីជាមូលហេតុនៃរឿងនោះ អ្នកណាជាដៃគូដែលមានតួនាទីក្នុងការផ្លាស់ប្តូរវា អ្វីដែលនឹងមានប្រសិទ្ធភាព និងអ្វីដែលអង្គការស្នើធ្វើ ដែលជាទម្រង់មានរចនាសម្ព័ន្ធនៃហេតុផល ដែលគំរូតក្កវិទ្យា និងទ្រឹស្តីនៃការផ្លាស់ប្តូរធ្វើជាផ្លូវការដោយឡែកពីគ្នា។ RBA គឺជាឧទាហរណ៍មួយនៃទម្រង់ទូលំទូលាយជាងនៃការគ្រប់គ្រងផ្អែកលើលទ្ធផល ដែលជិតស្និទ្ធនឹងការអនុវត្តជាប្រចាំនៃ[ការត្រួតពិនិត្យ និងការវាយតម្លៃ](https://en.wikipedia.org/wiki/Monitoring_and_evaluation)ដែលប្រើទូទាំងការអភិវឌ្ឍអន្តរជាតិ និងរដ្ឋបាលសាធារណៈ ហើយត្រូវបានលើកស្ទួយក្នុងទម្រង់ផ្សេងៗដោយរដ្ឋាភិបាលជាតិ អ្នកផ្តល់ថវិកាសប្បុរសធម៌ និងស្ថាប័នពហុភាគី ជាវិន័យរួមសម្រាប់ភ្ជាប់ការចំណាយទៅនឹងលទ្ធផល ជាជាងទៅនឹងផលិតផលតែប៉ុណ្ណោះ។

**ផលិតផល លទ្ធផល និងកាតពិន្ទុរួមគ្នា។** ហេតុផលដែលក្របខ័ណ្ឌទាំងនេះស្ថិតក្នុងប្រធានបទតែមួយ ជាជាងបីប្រធានបទ គឺដោយសារពួកវាឆ្លើយសំណួរគោលដូចគ្នានៅចំណុចខុសៗគ្នាក្នុងវដ្តនៃការធ្វើផែនការ និងការរាយការណ៍របស់អង្គការ។ គំរូតក្កវិទ្យាត្រូវបានគូរនៅពេលរចនា ដើម្បីបញ្ជាក់ការអះអាងអំពីមូលហេតុ។ គណនេយ្យភាពផ្អែកលើលទ្ធផល និងក្របខ័ណ្ឌប្រហាក់ប្រហែលត្រូវបានអនុវត្តនៅពេលរាយការណ៍ ដើម្បីរក្សាឱ្យវិធានការនៃការអនុវត្តស្មោះត្រង់អំពីភាពខុសគ្នារវាងកិច្ចខិតខំប្រឹងប្រែង និងផល។ កាតពិន្ទុតម្លៃសាធារណៈស្ថិតនៅពីលើទាំងពីរ ជារចនាសម្ព័ន្ធអចិន្ត្រៃយ៍ដែលធានាថា ភស្តុតាងអំពីលទ្ធផល ទោះបង្កើតឡើងតាមរបៀបណាក៏ដោយ ទៅដល់ការពិភាក្សាអំពីភាពស្របច្បាប់ និងសមត្ថភាព ដែលជាកន្លែងធ្វើសេចក្តីសម្រេចអំពីថវិកា និងអាណត្តិ ជាជាងស្ថិតនៅក្នុងការរាយការណ៍របស់ក្រុមកម្មវិធីផ្ទាល់។ ក្របខ័ណ្ឌទាំងបីរួមគ្នា គឺជាអ្វីដែលបំប្លែង[តម្លៃសាធារណៈ](https://en.wikipedia.org/wiki/Public_value)ពីការអះអាងដែលអង្គការធ្វើអំពីខ្លួនឯង ទៅជាការអះអាងដែលអង្គការអាចការពារបានដោយភស្តុតាង។

## ការអនុវត្តល្អបំផុត

1. **សូមគូរគំរូតក្កវិទ្យាមុនសូចនាករ មិនមែនក្រោយទេ។** ការជ្រើសរើសអ្វីដែលត្រូវវាស់ មុនពេលព្រមព្រៀងគ្នាអំពីអ្វីដែលអ្នកអះអាងថាបង្កឱ្យមាន នាំឱ្យបានបញ្ជីនៃអ្វីក៏ដោយដែលងាយរាប់។ សូមគូរខ្សែសង្វាក់ពីធាតុចូលដល់ផលប៉ះពាល់ជាមុនសិន រួចជ្រើសរើសវិធានការដែលសាកល្បងតំណនីមួយៗក្នុងខ្សែសង្វាក់នោះ។

**ប្រសិនបើលទ្ធផលប្រជាជនចម្បងរបស់យើងមិនផ្លាស់ប្តូរក្នុងរយៈពេលពីរវដ្តរាយការណ៍បន្ថែមទៀត តើយើងនឹងធ្វើអ្វីខុសគ្នា ហើយអ្វីដែលយើងនឹងមិនផ្លាស់ប្តូរ?** សំណួរនេះសាកល្បងថា តើអង្គការកំពុងប្រើក្របខ័ណ្ឌលទ្ធផលដើម្បីដឹកនាំពិតប្រាកដឬទេ ឬគ្រាន់តែផលិតវាជាកាតព្វកិច្ចរាយការណ៍។ សូមប្រើវាដើម្បីញែកថាផ្នែកណានៃយុទ្ធសាស្ត្របច្ចុប្បន្នអាស្រ័យលើទិន្នន័យលទ្ធផល ហើយផ្នែកណាថេរមិនថាទិន្នន័យបង្ហាញអ្វីក៏ដោយ។ សូមពិភាក្សាដោយស្មោះត្រង់ថា តើមានផ្នែកណាមួយនៃកម្មវិធីបច្ចុប្បន្នដែលនៅតែមិនប្រែប្រួល សូម្បីតែភស្តុតាងងាកប្រឆាំងយ៉ាងម៉ឺងម៉ាត់ ព្រោះនោះជាផ្នែកដែលក្របខ័ណ្ឌមិនបានគ្រប់គ្រងពិតប្រាកដ។

**តើសំណុំសូចនាករប្រជាជន និងសូចនាករអនុវត្តតូចបំផុតដែលយើងអាចរាយការណ៍ ហើយនៅតែអនុញ្ញាតឱ្យអ្នកខាងក្រៅវិនិច្ឆ័យថាអង្គការនេះកំពុងបង្កើតតម្លៃសាធារណៈឬទេ គឺអ្វី?** ប្រព័ន្ធវាស់វែងមានទំនោរប្រមូលសូចនាករជាយូរៗទៅលឿនជាងការដកវាចេញ រហូតដល់ក្របខ័ណ្ឌធំពេកដែលគ្មាននរណាប្រើក្នុងការសម្រេចចិត្តបាន។ សូមប្រើសំណួរនេះដើម្បីសាកល្បងធ្វើជាលំហាត់ ក្នុងការកាត់បន្ថយសំណុំសូចនាករបច្ចុប្បន្នពាក់កណ្តាល ដោយមិនបាត់បង់សមត្ថភាពឆ្លើយសំណួរគណនេយ្យភាពប្រជាជន និងគណនេយ្យភាពនៃការអនុវត្តដែលប្រធានបទនេះពិពណ៌នា។ សូមពិភាក្សាថាលំហាត់នេះបង្ហាញការប្រឆាំងអ្វីខ្លះ ព្រោះការប្រឆាំងចំពោះការកាត់សូចនាករជាក់លាក់មួយ ច្រើនតែជាសញ្ញាបង្ហាញថាអង្គការកំពុងត្រូវទទួលខុសត្រូវអំពីអ្វីពិតប្រាកដ មិនថាវាត្រូវនឹងយុទ្ធសាស្ត្រដែលបានប្រកាសឬអត់។

## នៅក្នុងការអនុវត្ត៖ ឧទាហរណ៍តម្លៃសាធារណៈ

បណ្តាញក្នុងតំបន់នៃអង្គការសប្បុរសធម៌គាំទ្រគ្រួសារមួយ ដែលជារឿងប្រឌិត ជាគំរូនៃលំនាំមួយ មិនមែនជារបាយការណ៍អំពីអង្គការពិតប្រាកដណាមួយទេ បានចំណាយពេលមួយទសវត្សរ៍ក្នុងការដំណើរការកម្មវិធីទៅសួរសុខទុក្ខតាមផ្ទះសម្រាប់មាតាបិតាថ្មី នៅសង្កាត់ដែលមានប្រាក់ចំណូលទាប។ របាយការណ៍ប្រចាំឆ្នាំរបស់វាជូនអ្នកផ្តល់ថវិកាមានលម្អិត ហើយតាមវិធានការផលិតផល គួរឱ្យចាប់អារម្មណ៍៖ ការទៅសួរសុខទុក្ខតាមផ្ទះរាប់ពាន់លើកបានបញ្ចប់ជារៀងរាល់ឆ្នាំ អត្រាចូលរួមខ្ពស់ក្នុងសិក្ខាសាលាមាតាបិតា និងពិន្ទុពេញចិត្តខ្ពស់ពីគ្រួសារដែលបានចូលរួម។ ដោយផ្អែកលើកំណត់ត្រានោះ បណ្តាញបានរីកធំជាលំដាប់ ដោយបន្ថែមសាខាមូលដ្ឋានថ្មីប្រហែលរៀងរាល់ពីរឆ្នាំ។

អ្នកផ្តល់ថវិកាថ្នាក់ជាតិថ្មីមួយ ដែលកំពុងពិចារណាផ្តល់ជំនួយច្រើនឆ្នាំមួយធំល្មមអាចបង្កើនវិសាលភាពរបស់បណ្តាញទ្វេដង បានសួរសំណួរមួយដែលក្រុមប្រឹក្សាភិបាលរបស់បណ្តាញផ្ទាល់មិនធ្លាប់សួរដល់គណៈគ្រប់គ្រងជាផ្លូវការ គឺថាតើភស្តុតាងអ្វីបង្ហាញថា កុមារក្នុងគ្រួសារដែលបានទទួលការសួរសុខទុក្ខតាមផ្ទះ ពិតជាមានស្ថានភាពប្រសើរជាងក្រោយមក ធៀបនឹងកុមារប្រៀបធៀបបានដែលគ្រួសារមិនបានទទួល។ ថ្នាក់ដឹកនាំរបស់បណ្តាញបានដឹងខ្លួន ពេលដោះស្រាយសំណួរនេះថា របាយការណ៍របស់ខ្លួនមិនធ្លាប់ញែកកិច្ចខិតខំប្រឹងប្រែងចេញពីផលទេ។ ចំនួនការសួរសុខទុក្ខ និងពិន្ទុពេញចិត្តគឺជាភស្តុតាងពិតថាកម្មវិធីកំពុងត្រូវបានផ្តល់ជូន ហើយទទួលបានការគាំទ្រល្អ ប៉ុន្តែគ្មានមួយណាជាភស្តុតាងថាវាបានផ្លាស់ប្តូរលទ្ធផលអភិវឌ្ឍន៍របស់កុមារទេ ដែលជាការអះអាងដែលបង្កប់ក្នុងសេចក្តីថ្លែងបេសកកម្មរបស់បណ្តាញផ្ទាល់ និងដោយបង្កប់ន័យ ក្នុងពាក្យស្នើសុំជំនួយនីមួយៗដែលវាធ្លាប់សរសេរ។

បណ្តាញបានសាងសង់គំរូតក្កវិទ្យាពីសូន្យ ដោយចាប់ផ្តើមពីលទ្ធផលដែលវាចង់បានពិតប្រាកដ គឺសូចនាករអភិវឌ្ឍន៍កុមារតូចដែលប្រសើរឡើង ករណីការធ្វេសប្រហែសកុមារដែលបានបញ្ជាក់មានចំនួនថយចុះ និងទំនុកចិត្តមាតាបិតាកាន់តែរឹងមាំ រួចធ្វើការថយក្រោយទៅរកសកម្មភាព និងផលិតផលដែលនីមួយៗត្រូវពឹងផ្អែក ដោយកំណត់ឈ្មោះសម្មតិកម្មនៅចន្លោះយ៉ាងច្បាស់លាស់។ វាបានអនុម័តរចនាសម្ព័ន្ធគណនេយ្យភាពផ្អែកលើលទ្ធផលដែលសម្រួលមួយ គឺសូចនាករប្រជាជនដែលតាមដានកម្រិតសង្កាត់ និងចែករំលែកជាមួយសាលារៀន និងសេវាសុខភាពក្នុងតំបន់ ព្រមទាំងសូចនាករអនុវត្តដែលក្រុមរបស់វាផ្ទាល់អាចធ្វើសកម្មភាពបានដោយផ្ទាល់ គឺភាពញឹកញាប់នៃការសួរសុខទុក្ខ ការបញ្ចប់សិក្ខាសាលា និងទំនុកចិត្តមាតាបិតាដែលបុគ្គលិកវាយតម្លៃនៅបីខែ និងដប់ពីរខែ។ សូចនាករលទ្ធផលពីរក្នុងចំណោមប្រាំមិនបានផ្លាស់ប្តូរដូចសង្ឃឹមនៅសង្កាត់មួយក្នុងវដ្តរាយការណ៍ដំបូង។ ជំនួសឱ្យការបិទបាំងលទ្ធផល បណ្តាញបានប្រើដំណើរការ "បត់ខ្សែកោង" របស់ខ្លួនដើម្បីស៊ើបអង្កេត ហើយបានរកឃើញថាការផ្លាស់ប្តូរបុគ្គលិកអ្នកសួរសុខទុក្ខផ្នែកសុខភាពក្នុងតំបន់ បានរំខានផ្លូវបញ្ជូនបន្ត ហើយបានកែសម្រួលការរៀបចំភាពជាដៃគូ ជាជាងកែគំរូស្នូលរបស់ខ្លួន។ អ្នកផ្តល់ថវិកាបានបន្តអាយុជំនួយ និងពង្រីកវា ដោយលើកឡើងថាភាពស្មោះត្រង់នៃការរាយការណ៍ មិនមែនការគ្មានកង្វះខាតនោះទេ ជាហេតុផលនៃទំនុកចិត្ត។

## ទស្សនៈវិស័យទាំងបួន

### រដ្ឋាភិបាលមូលដ្ឋាន

ក្របខ័ណ្ឌលទ្ធផលរបស់អាជ្ញាធរមូលដ្ឋាន ជាទូទៅស្ថិតក្នុងសំណុំសូចនាករដែលកំណត់ដោយជាតិទូលំទូលាយជាង ដែលខ្លួនមិនបានជ្រើសរើសប៉ុន្តែត្រូវរាយការណ៍ធៀប រួមជាមួយវិធានការដែលជ្រើសរើសក្នុងតំបន់ ដែលឆ្លុះបញ្ចាំងអាទិភាពរបស់ក្រុមប្រឹក្សាខ្លួន។ រចនាសម្ព័ន្ធពីរនេះមានន័យថា កាតពិន្ទុមូលដ្ឋានច្រើនតែត្រូវផ្សះផ្សាអ្នកទស្សនាពីរក្រុមដែលមានសំណួរខុសគ្នា គឺអ្នកត្រួតពិនិត្យថ្នាក់ជាតិដែលសួរអំពីការអនុលោមតាមសំណុំសូចនាករស្តង់ដារ និងអ្នកស្រុកក្នុងតំបន់ដែលសួរអំពីលក្ខខណ្ឌមូលដ្ឋានជាក់លាក់ដែលមើលឃើញ ដូចជាផ្លូវ ឬសេវាណាមួយ។ គំរូតក្កវិទ្យាក្នុងមាត្រដ្ឋានរដ្ឋាភិបាលមូលដ្ឋាន ជាធម្មតាងាយសាងសង់បំផុតសម្រាប់សេវាតែមួយ (ការផ្លាស់ប្តូរការប្រមូលសម្រាមកែច្នៃឡើងវិញ ការរចនាបណ្ណាល័យឡើងវិញ) ហើយពិបាកសាងសង់ឱ្យគួរឱ្យជឿជាក់បំផុតសម្រាប់លទ្ធផលឆ្លងវិស័យ ដូចជាសាមគ្គីភាពសហគមន៍ ដែលអ្នកដើរតួជាច្រើននៅក្រៅការគ្រប់គ្រងរបស់ក្រុមប្រឹក្សាក៏ចូលរួមចំណែកដែរ។

### រដ្ឋាភិបាលជាតិ

ក្របខ័ណ្ឌលទ្ធផលថ្នាក់ជាតិដំណើរការក្នុងមាត្រដ្ឋានដែលភាពខុសគ្នារវាងគណនេយ្យភាពប្រជាជន និងគណនេយ្យភាពនៃការអនុវត្តពី RBA ក្លាយជាចាំបាច់ ជាជាងជាជម្រើស៖ សូចនាករសុខភាព ឬអប់រំថ្នាក់ជាតិទទួលឥទ្ធិពលពីកម្មវិធីរាប់រយ ដែលមានតែមួយចំនួនតិចតួចប៉ុណ្ណោះដែលក្រសួងតែមួយផ្តល់ថវិកាដោយផ្ទាល់ ដូច្នេះក្រសួងមួយអះអាងថាទទួលកិត្តិយសតែម្នាក់ឯង ឬទទួលការស្តីបន្ទោសតែម្នាក់ឯងចំពោះនិន្នាការថ្នាក់ជាតិ គឺកម្រអាចការពារបាន។ រដ្ឋាភិបាលជាតិក៏ជាប្រភពទូទៅបំផុតនៃក្របខ័ណ្ឌលទ្ធផលធំៗដែលកំណត់ដោយមជ្ឈការ ដែលស្ថាប័នផ្តល់សេវាកម្រិតទាបជាងក្នុងខ្សែសង្វាក់ត្រូវតែអនុម័ត ដែលបង្កើតហានិភ័យកើតឡើងម្តងហើយម្តងទៀត ថាក្របខ័ណ្ឌត្រូវបានរចនាសម្រាប់ការប្រមូលផ្តុំ និងការប្រៀបធៀបឆ្លងក្រសួង ជាជាងសម្រាប់ដឹកនាំការផ្តល់សេវារបស់កម្មវិធីណាមួយពិតប្រាកដ។

### វិស័យសង្គម និងវិស័យមិនស្វែងរកប្រាក់ចំណេញ

អង្គការមិនស្វែងរកប្រាក់ចំណេញ និងអង្គការក្រៅរដ្ឋាភិបាល (NGO) ច្រើនតែសាងសង់គំរូតក្កវិទ្យា និងទ្រឹស្តីនៃការផ្លាស់ប្តូរផ្ទាល់ខ្លួនពីគោលការណ៍ដំបូង ជាជាងអនុម័តក្របខ័ណ្ឌដែលអ្នកផ្តល់សិទ្ធិប្រគល់ឱ្យ ព្រោះអ្នកផ្តល់ថវិការបស់ពួកគេជារឿយៗជាអ្នកស្នើសុំករណីមូលហេតុ ជាជាងផ្គត់ផ្គង់គំរូស្តង់ដារ។ នេះផ្តល់សេរីភាពកាន់តែច្រើនដល់អង្គការវិស័យសង្គម ក្នុងការរចនាក្របខ័ណ្ឌលទ្ធផលដែលស័ក្តិសមនឹងតក្កវិទ្យាកម្មវិធីពិតរបស់ខ្លួន ប៉ុន្តែក៏ប្រឈមនឹងភាពអស់កម្លាំងនៃការវាស់វែងដែលជំរុញដោយអ្នកផ្តល់ថវិកាកាន់តែខ្លាំងដែរ ដែលជំនួយនីមួយៗមានតម្រូវការរាយការណ៍ផ្ទាល់ខ្លួន ហើយអង្គការបញ្ចប់ដោយត្រូវថែរក្សាក្របខ័ណ្ឌលទ្ធផលស្របគ្នាជាច្រើនសម្រាប់ការងារមូលដ្ឋានតែមួយ។ ការបែងចែកប្រជាជនធៀបនឹងការអនុវត្តរបស់គណនេយ្យភាពផ្អែកលើលទ្ធផល មានប្រយោជន៍ជាពិសេសនៅទីនេះ ព្រោះវាផ្តល់ភាសាដល់អង្គការមិនស្វែងរកប្រាក់ចំណេញ ដើម្បីប្រាប់អ្នកផ្តល់ថវិកាឱ្យច្បាស់ថា វាអាចត្រូវទទួលខុសត្រូវចំពោះវិធានការអនុវត្តផ្ទាល់ខ្លួន ប៉ុន្តែមិនអាចទទួលខុសត្រូវតែម្នាក់ឯងចំពោះសូចនាករប្រជាជនដែលអ្នកដើរតួជាច្រើនទៀតក៏មានឥទ្ធិពលដែរ។

### កម្រិតពហុភាគី និងអន្តរជាតិ

ស្ថាប័នពហុភាគីអនុវត្តការគ្រប់គ្រងផ្អែកលើលទ្ធផលក្នុងមាត្រដ្ឋានដែលសម្មតិកម្មរបស់គំរូតក្កវិទ្យាធ្វើដំណើរឆ្លងកាត់បរិបទជាតិរាប់សិបក្នុងពេលតែមួយ ដូច្នេះតក្កវិទ្យាអន្តរាគមន៍ដែលពិតនៅប្រទេសមួយ អាចបរាជ័យដោយស្ងៀមស្ងាត់នៅប្រទេសមួយទៀត ដោយសូចនាករសកលចម្បងមិនបង្ហាញថាបរិបទណាជាបញ្ហា។ ក្របខ័ណ្ឌលទ្ធផលក្នុងបរិបទនេះច្រើនតែត្រូវបានសាងសង់ដើម្បីបំពេញចិត្តរដ្ឋាភិបាលម្ចាស់ជំនួយច្រើនព្រមគ្នា ដែលម្នាក់ៗមានចំណូលចិត្តរាយការណ៍ផ្ទាល់ខ្លួន ដែលរុញទៅរកសំណុំសូចនាករស្តង់ដារ (ដូចប្រភេទដែលលើកស្ទួយតាមរយៈការណែនាំអំពីការគ្រប់គ្រងផ្អែកលើលទ្ធផលអន្តរជាតិ) ដោយទទួលរងការបាត់បង់ភាពជាក់លាក់មូលដ្ឋាន។ គម្លាតរវាងផលិតផលដែលដៃគូអនុវត្តអាចរាយការណ៍បានលឿន ដូចជាការបណ្តុះបណ្តាលដែលបានផ្តល់ គ្លីនិកដែលបានផ្គត់ផ្គង់ និងលទ្ធផលដែលបង្ហាញតែក្នុងការស្ទង់មតិថ្នាក់ជាតិច្រើនឆ្នាំក្រោយ គឺធំបំផុតនៅទីនេះ ដែលធ្វើឱ្យវិន័យនៃការបញ្ជាក់ខ្សែសង្វាក់មូលហេតុដែលសន្មតយ៉ាងច្បាស់លាស់ ជាជាងសន្មតដោយមិនបញ្ជាក់ មានសារៈសំខាន់ជាពិសេស។

## ទម្រង់នៃការបរាជ័យដែលជួបញឹកញាប់

**ផលិតផលក្លែងជាលទ្ធផល។** អង្គការរាយការណ៍ចំនួនសកម្មភាពដោយប្រើភាសានៃលទ្ធផល ដូចជា "សុវត្ថិភាពសហគមន៍ប្រសើរឡើង" ខណៈវិធានការពិតគឺការល្បាតដែលបានធ្វើ ទោះដោយសារការច្រលំពិតប្រាកដ ឬដោយសារទិន្នន័យលទ្ធផលពិបាក និងយឺតក្នុងការប្រមូលជាង។ ថ្លៃដើមលេចឡើងនៅពេលអ្នកពិនិត្យខាងក្រៅសួរភស្តុតាងលទ្ធផលដោយផ្ទាល់ ហើយរកឃើញតែផលិតផលនៅពីក្រោយការអះអាង ដែលបំផ្លាញភាពជឿជាក់របស់អង្គការច្រើនជាងលទ្ធផលសាមញ្ញដែលរាយការណ៍ដោយស្មោះត្រង់។

**ការកើនឡើងនៃសូចនាករ។** កាតពិន្ទុ ឬគំរូតក្កវិទ្យាប្រមូលវិធានការលឿនជាងការដកវាចេញ ដោយសារសំណើរបស់អ្នកពាក់ព័ន្ធថ្មីនីមួយៗបន្ថែមបន្ទាត់មួយទៀត ជាជាងជំនួសបន្ទាត់ដែលមានស្រាប់។ លទ្ធផលចុងក្រោយគឺក្របខ័ណ្ឌធំពេកដែលគ្មានអ្នកគ្រប់គ្រងណាអាចប្រើក្នុងការសម្រេចចិត្តពិតប្រាកដ ដែលធ្វើឱ្យអង្គការត្រឡប់ទៅរកការវិនិច្ឆ័យក្រៅផ្លូវការដោយស្ងៀមស្ងាត់ ខណៈក្របខ័ណ្ឌផ្លូវការក្លាយជាឯកសាររាយការណ៍ដែលគ្មាននរណាពិគ្រោះ។

**ការចាប់យកគោលដៅ។** វិធានការអនុវត្តមួយ នៅពេលភ្ជាប់នឹងថវិកា ឬរង្វាន់ ក្លាយជាអ្វីដែលបុគ្គលិកធ្វើឱ្យល្អបំផុតដោយផ្ទាល់ ជាជាងលទ្ធផលដែលវាមានគោលបំណងតំណាង ដែលជាលំនាំដែល[ច្បាប់ Goodhart](https://en.wikipedia.org/wiki/Goodhart%27s_law)ពិពណ៌នា ដែលវិធានការដែលប្រើជាគោលដៅឈប់ជាវិធានការដែលអាចជឿជាក់បាន។ កន្លែងសម្រាប់ដោះស្រាយថាមវន្តនេះក្នុងវិស័យសាធារណៈជាពិសេស រួមទាំងរបៀបដែលរបបគុណភាពព្យាយាមរកឃើញ និងកម្រិតវា គឺនៅកន្លែងផ្សេង (សូមមើល ប្រធានបទ 3.11 — ការគ្រប់គ្រងគុណភាព សុវត្ថិភាព និងសមិទ្ធកម្ម) ឯនៅទីនេះ វាសំខាន់ជាហេតុផលដើម្បីរក្សាវិធានការកិច្ចខិតខំប្រឹងប្រែង គុណភាព និងផលឱ្យដាច់ពីគ្នា និងពិនិត្យគោលដៅឡើងវិញមុនពេលវាបង្ខូចអាកប្បកិរិយា។

**ការអះអាងលើសពីការបញ្ជាក់មូលហេតុ។** របាយការណ៍របស់កម្មវិធីបង្កប់ន័យថាទទួលខុសត្រូវមូលហេតុតែម្នាក់ឯងចំពោះលទ្ធផលកម្រិតប្រជាជន ដែលអ្នកដើរតួជាច្រើនទៀតក៏មានឥទ្ធិពលដែរ ទោះដោយសារភាសាមិនប្រុងប្រយ័ត្ន ឬជំនឿពិតប្រាកដ។ នៅពេលលទ្ធផលមិនផ្លាស់ប្តូរ ដែលច្រើនតែដោយសារហេតុផលក្រៅការគ្រប់គ្រងរបស់កម្មវិធី អង្គការទទួលការស្តីបន្ទោសមិនសមាមាត្រនឹងការចូលរួមចំណែកពិតរបស់ខ្លួន ដែលបន្ទាប់មកបន្ថយការរាយការណ៍ដោយស្មោះត្រង់អំពីកង្វះខាតក្នុងវដ្តនាពេលអនាគត។

**ក្របខ័ណ្ឌគ្មានមតិត្រឡប់។** អង្គការសាងសង់គំរូតក្កវិទ្យា និងក្របខ័ណ្ឌលទ្ធផលដែលរចនាបានល្អពិតប្រាកដ រួចមិនដែលពិនិត្យវាឡើងវិញធៀបនឹងលទ្ធផលពិត ដូច្នេះវារឹងទៅជាឯកសារថេរ ជាជាងឧបករណ៍គ្រប់គ្រងដែលមានជីវិត។ យុទ្ធសាស្ត្រនៅពីក្រោមនៅតែដំណើរការលើសម្មតិកម្មដើមជាច្រើនឆ្នាំក្រោយភស្តុតាងនឹងបញ្ជាក់ថាគួរកែប្រែ ដោយចំណាយលើការសម្របខ្លួនដែលក្របខ័ណ្ឌមានគោលបំណងផ្តល់ឱ្យ។

## បញ្ជីត្រួតពិនិត្យ

- [ ] សូមគូរគំរូតក្កវិទ្យា (ធាតុចូល សកម្មភាព ផលិតផល លទ្ធផល ផលប៉ះពាល់) សម្រាប់កម្មវិធីធំៗនីមួយៗ មុនបញ្ចប់សូចនាករ។
