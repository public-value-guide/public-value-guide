# ប្រធានបទ 2.5 — ករណីអាជីវកម្ម និងតម្លៃសមស្របនឹងថវិកា

**ករណីអាជីវកម្មសាធារណៈទទួលបានការអនុម័តលុះត្រាតែវាបង្ហាញបានប្រាំរឿងផ្សេងគ្នាក្នុងពេលតែមួយ គឺថាការចំណាយនេះជាអ្វីដែលត្រូវធ្វើ ថាវាផ្តល់តម្លៃសមស្របនឹងថវិកា ថាវាអាចទិញបាន ថាវាអាចមានលទ្ធភាពទទួលបាន និងថាវាអាចផ្តល់បាន ព្រោះផលចំណេញវិជ្ជមានលើការវិនិយោគ ដែលជាការសាកល្បងពិតតែមួយគត់របស់វិស័យឯកជន មិនដែលគ្រប់គ្រាន់ដោយខ្លួនឯង ដើម្បីបញ្ជាក់ពីការចំណាយប្រាក់ដែលជាកម្មសិទ្ធិរបស់សាធារណជននោះទេ។**

## ហេតុអ្វីបានជាបញ្ហានេះសំខាន់សម្រាប់តម្លៃសាធារណៈ

អ្នកគ្រប់គ្រងសាធារណៈគ្រប់រូបដែលធ្លាប់នាំសំណើចំណាយទៅកាន់គណៈកម្មាធិការហិរញ្ញវត្ថុ ក្រុមប្រឹក្សា ឬក្រុមប្រឹក្សាភិបាល បានជួបប្រទះពេលដែលគំនិតល្អជួបនឹងសំណួរដែលសៀវភៅបញ្ជីមិនអាចឆ្លើយបាន៖ តើនេះជាអ្វីដែលយើងគួរធ្វើឬទេ ហើយតើអ្នកអាចបង្ហាញបានមុនយើងប្តេជ្ញាឬទេ? ករណីវិនិយោគឯកជនអាចបំពេញដោយតម្លៃបច្ចុប្បន្នសុទ្ធវិជ្ជមាន និងចំណង់ទទួលហានិភ័យរបស់ក្រុមប្រឹក្សាភិបាល។ ករណីសាធារណៈមិនអាចទេ ព្រោះប្រាក់មិនមែនជារបស់អង្គការក្នុងន័យដូចគ្នា ហើយអ្នកដែលនឹងវិនិច្ឆ័យសេចក្តីសម្រេច ដូចជាអ្នកបោះឆ្នោត អ្នកប្រើសេវា សវនករ និងប្រព័ន្ធផ្សព្វផ្សាយ មិនដែលត្រូវបានសុំឱ្យទិញភាគហ៊ុនឡើយ។

គម្លាតនោះជាហេតុផលដែលសេចក្តីសម្រេចចំណាយសាធារណៈដែលមានបំណងល្អ ច្រើនតែបែកខ្ចាត់ខ្ចាយក្រោយការអនុម័ត ជាជាងមុន។ កម្មវិធីមួយឆ្លងកាត់ការវាយតម្លៃសេដ្ឋកិច្ចដោយបង្ហាញសមាមាត្រអត្ថប្រយោជន៍-ថ្លៃដើមខ្លាំង រួចជាប់គាំង ព្រោះគ្មាននរណាបានគិតថាវានឹងត្រូវទិញដោយរបៀបណា អង្គការផ្តល់សេវាមិនដែលមានបុគ្គលិកដើម្បីដំណើរការវា ឬអ្នកផ្តល់សិទ្ធិដែលមិនដែលត្រូវបានពិគ្រោះយោបល់ជំទាស់នៅពេលការចំណាយក្លាយជាសាធារណៈ។ ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព កំណត់ហេតុផលដែលភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាពត្រូវតែស្ថិតស្ថេរជាមួយគ្នា (សូមមើល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព) ហើយប្រធានបទនេះនិយាយអំពីឧបករណ៍ គឺករណីអាជីវកម្ម ដែលស្ថាប័នសាធារណៈប្រើដើម្បីសាកល្បងទាំងបីមុនប្រាក់ធ្វើចលនា និងយន្តការសវនកម្មដែលត្រួតពិនិត្យក្រោយមក ថាតើការសាកល្បងនោះស្មោះត្រង់ឬទេ។

ហានិភ័យលេចឡើងពីរដង៖ នៅពេលករណីអាជីវកម្មខ្សោយអនុញ្ញាតឱ្យសេចក្តីសម្រេចអាក្រក់ឆ្លងកាត់ និងនៅពេលករណីអាជីវកម្មធ្ងន់ និងមាននីតិវិធីខ្លាំងរហូតក្លាយជាឧបសគ្គមិនឱ្យមានអ្វីកើតឡើងសោះ។ ការធ្វើឱ្យត្រឹមត្រូវមិនមែនជាលំហាត់អនុលោមដែលភ្ជាប់ទៅនឹងការងារពិតនៃការគ្រប់គ្រងសាធារណៈទេ វាជាពេលមួយក្នុងចំណោមពេលតិចតួចដែលភាពស្របច្បាប់ តម្លៃ និងលទ្ធភាពផ្តល់ត្រូវបានបង្ខំឱ្យស្ថិតក្នុងឯកសារតែមួយ នៅចំពោះមុខអ្នកសម្រេចចិត្តតែម្នាក់ ក្នុងពេលតែមួយ។

## គំនិតស្នូល

**ករណីអាជីវកម្ម** កំណត់ហេតុផលសម្រាប់ការចំណាយដែលបានស្នើ គឺបញ្ហាដែលវាដោះស្រាយ ជម្រើសដែលបានពិចារណា ថ្លៃដើម និងអត្ថប្រយោជន៍ និងរបៀបដែលប្រាក់នឹងត្រូវចំណាយ ដើម្បីឱ្យអ្នកសម្រេចចិត្តអាចអនុម័ត បដិសេធ ឬបញ្ជូនត្រឡប់ទៅវិញ។ [ករណីអាជីវកម្ម](https://en.wikipedia.org/wiki/Business_case)របស់វិស័យឯកជន ភាគច្រើនធានាមូលធនផ្ទៃក្នុងធៀបនឹងការប្រើប្រាស់ប្រកួតប្រជែងក្នុងក្រុមហ៊ុនតែមួយ។ ករណីសាធារណៈត្រូវធ្វើច្រើនជាងនេះ៖ ទប់ទល់ការពិនិត្យពីមនុស្សដែលមិនដែលយល់ព្រមតាមរយៈការទិញផលិតផលថា ការចំណាយនោះសមនឹងតម្លៃ។

**គំរូករណីប្រាំ** គឺជាលំនាំចម្បងរបស់វិស័យសាធារណៈសម្រាប់ហេតុផលនេះ ដែលបានអភិវឌ្ឍក្នុងរដ្ឋាភិបាលកណ្តាលចក្រភពអង់គ្លេស ហើយអនុម័តក្នុងទម្រង់ជិតស្និទ្ធនៅកន្លែងផ្សេង ដូចជាក្រសួងរតនាគារនូវែលសេឡង់បានសាងសង់ការណែនាំ "Better Business Cases" លើរចនាសម្ព័ន្ធដូចគ្នាជាមួយក្រសួងរតនាគារអង់គ្លេស។ វាតម្រូវឱ្យមានករណីប្រាំផ្សេងគ្នា ជាជាងករណីមួយ គឺយុទ្ធសាស្ត្រ (តើនេះជាអ្វីដែលត្រូវធ្វើឬទេ) សេដ្ឋកិច្ច (តើវាផ្តល់តម្លៃសមស្របនឹងថវិកាក្នុងចំណោមជម្រើសឬទេ) ពាណិជ្ជកម្ម (តើវាអាចទិញបានតាមលក្ខខណ្ឌដែលទីផ្សារទទួលយកឬទេ) ហិរញ្ញវត្ថុ (តើអង្គការអាចមានលទ្ធភាពទទួលបានវាឬទេ) និងការគ្រប់គ្រង (តើវាអាចផ្តល់ គ្រប់គ្រង និងតាមដានអត្ថប្រយោជន៍របស់វាបានឬទេ)។ ការញែកទាំងប្រាំមានសារៈសំខាន់ ព្រោះសំណើអាចបរាជ័យលើករណីណាមួយ ទោះជោគជ័យលើករណីផ្សេងទៀត៖ ការប្រើប្រាស់ប្រាក់ដ៏ល្អឥតខ្ចោះអាចនៅតែមិនអាចទិញបាន គ្មានលទ្ធភាពទទួលបានក្នុងឆ្នាំនេះ ឬហួសសមត្ថភាពគ្រប់គ្រងរបស់អង្គការ។

**ករណីយុទ្ធសាស្ត្រ** ឆ្លើយសំណួរតូចជាងអ្វីដែលវាស្តាប់ទៅ៖ មិនមែន "តើនេះមានតម្លៃទេ" ប៉ុន្តែ "តើនេះជាបញ្ហាត្រឹមត្រូវ ស្របតាមគោលបំណងរបស់យើង ជាមួយករណីពិតសម្រាប់ការផ្លាស់ប្តូរ ជាជាងចំណូលចិត្តដែលបានសម្រេចរួច"។ នៅពេលសរសេរក្រោយពេលដំណោះស្រាយដែលចូលចិត្តត្រូវបានជ្រើសរើស វាមានទំនោរអានថយក្រោយ ដោយរាយបញ្ជីគោលបំណងដែលចៃដន្យត្រូវនឹងដំណោះស្រាយ ជំនួសឱ្យការសាកល្បងថាតើដំណោះស្រាយកើតចេញពីពួកវាឬទេ។

**ករណីសេដ្ឋកិច្ច** គឺជាកន្លែងដែលការវាយតម្លៃជាផ្លូវការធ្វើការ៖ បញ្ជីវែងពិតនៃជម្រើសត្រូវបានបង្រួមទៅជាបញ្ជីខ្លី ដែលនីមួយៗត្រូវបានសាកល្បងធៀបនឹង "ករណីយោង" ឬ "ធ្វើតិចបំផុត" ជាស្ថានភាពប្រៀបធៀបពិតប្រាកដ ដោយប្រើការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម ឬប្រសិទ្ធភាពថ្លៃដើមលើមូលដ្ឋានរួម (សូមមើល ប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម)។ ការវាយតម្លៃសាធារណៈជាធម្មតាប្រើ **អត្រាបញ្ចុះតម្លៃសង្គម** ដែលឆ្លុះបញ្ចាំងពីចំណូលចិត្តពេលវេលារបស់សង្គម និងថ្លៃដើមឱកាសនៃមូលធនសាធារណៈ ជាជាងថ្លៃដើមមូលធនរបស់ក្រុមហ៊ុនឯកជន ហើយរាប់ថ្លៃដើម និងអត្ថប្រយោជន៍ដែលធ្លាក់លើមនុស្សនៅក្រៅប្រតិបត្តិការ។ [តម្លៃបច្ចុប្បន្នសុទ្ធ](https://en.wikipedia.org/wiki/Net_present_value)និងសមាមាត្រអត្ថប្រយោជន៍-ថ្លៃដើមនៅតែលេចឡើង ប៉ុន្តែជាធាតុចូលមួយក្នុងចំណោមច្រើន មិនមែនជាការសាកល្បងទាំងមូលទេ។

**ភាពលម្អៀងសុទិដ្ឋិនិយម** គឺជាទំនោរជាប្រព័ន្ធដែលមានភស្តុតាងល្អក្នុងការប៉ាន់ស្មានថ្លៃដើម កាលវិភាគ និងហានិភ័យទាបពេក និងអត្ថប្រយោជន៍ខ្ពស់ពេក។ ករណីសេដ្ឋកិច្ចម៉ឺងម៉ាត់កែតម្រូវវាយ៉ាងច្បាស់លាស់ ដោយការបន្ថែមផ្អែកលើភស្តុតាង ដែលទាញពីកំណត់ត្រាលទ្ធផលជាក់ស្តែងនៃគម្រោងមុនៗដែលប្រៀបធៀបបាន ជាជាងបន្ទាត់ទុកបម្រុងក្រៅផ្លូវការ។ [ភាពលម្អៀងសុទិដ្ឋិនិយម](https://en.wikipedia.org/wiki/Optimism_bias)ដែលមិនត្រូវបានត្រួតពិនិត្យ គឺជាមូលហេតុមួយក្នុងចំណោមមូលហេតុដែលមានឯកសារល្អបំផុតនៃ[ការចំណាយលើសថវិកា](https://en.wikipedia.org/wiki/Cost_overrun)សាធារណៈលើហេដ្ឋារចនាសម្ព័ន្ធ ហើយយុត្តាធិការជាច្រើនឥឡូវតម្រូវឱ្យមានការកែតម្រូវដែលបានបញ្ជាក់ និងកំណត់បរិមាណសម្រាប់វា។

**ករណីពាណិជ្ជកម្ម ហិរញ្ញវត្ថុ និងការគ្រប់គ្រង** គឺជាកន្លែងដែលករណីអាជីវកម្មឈប់ជាការវាយតម្លៃ ហើយចាប់ផ្តើមជាផែនការប្រតិបត្តិការ។ ករណីពាណិជ្ជកម្មសាកល្បងផ្លូវលទ្ធកម្ម និងចំណង់ទីផ្សារ ដែលផ្តល់ព័ត៌មានដោយផ្ទាល់ដល់ ប្រធានបទ 3.7 — លទ្ធកម្មសាធារណៈ និងការតែងតាំងសេវា (សូមមើល ប្រធានបទ 3.7 — លទ្ធកម្មសាធារណៈ និងការតែងតាំងសេវា)។ ករណីហិរញ្ញវត្ថុសាកល្បងលទ្ធភាពទទួលបានធៀបនឹងថវិកាពិត និងទម្រង់ថវិកា ដែលជាសំណួរខុសពីថាតើការចំណាយជាគំនិតល្អឬទេ។ ករណីគ្រប់គ្រងកំណត់អភិបាលកិច្ច ហានិភ័យនៃការផ្តល់ និងផែនការសម្រេចអត្ថប្រយោជន៍ដែលដាក់ឈ្មោះអ្នកជាម្ចាស់អត្ថប្រយោជន៍រំពឹងទុកនីមួយៗ ព្រោះអត្ថប្រយោជន៍គ្មានម្ចាស់កម្រត្រូវបានសម្រេច។

**តម្លៃសមស្របនឹងថវិកា** ដូចសវនករសាធារណៈប្រើពាក្យនេះ ខុសពីករណីសេដ្ឋកិច្ច ទោះទាក់ទងគ្នា។ ជាធម្មតាវាត្រូវបានកំណត់ជុំវិញលក្ខណៈវិនិច្ឆ័យបី គឺ "បី E"៖ **សន្សំសំចៃ (economy)** (ធាតុចូលគុណភាពត្រឹមត្រូវក្នុងថ្លៃដើមសមហេតុផលទាបបំផុត គឺចំណាយតិច) **ប្រសិទ្ធភាព (efficiency)** (ផលិតផលច្រើនបំផុតពីធាតុចូលដែលបានកំណត់ គឺចំណាយបានល្អ) និង **ប្រសិទ្ធផល (effectiveness)** (សម្រេចលទ្ធផលដែលមានបំណង គឺចំណាយដោយប្រាជ្ញា)។ ក្របខ័ណ្ឌខ្លះបន្ថែម "E" ទីបួន គឺ **សមធម៌ (equity)** ព្រោះកម្មវិធីអាចសន្សំសំចៃ មានប្រសិទ្ធភាព និងប្រសិទ្ធផលជាមធ្យម ខណៈចែកចាយថ្លៃដើម និងអត្ថប្រយោជន៍មិនយុត្តិធម៌។ [តម្លៃសមស្របនឹងថវិកា](https://en.wikipedia.org/wiki/Value_for_money)ត្រូវបានវាយតម្លៃពីរដង៖ ជាមុនក្នុងករណីសេដ្ឋកិច្ច និងក្រោយដោយស្ថាប័នសវនកម្មឯករាជ្យ ហើយភាពខុសគ្នានោះមានសារៈសំខាន់ពេញប្រធានបទនេះ។

**សវនកម្មតម្លៃសមស្របនឹងថវិកា** ដែលជួនកាលហៅថាសវនកម្មការអនុវត្ត គឺជាការពិនិត្យឯករាជ្យក្រោយព្រឹត្តិការណ៍ ថាតើការចំណាយបានសម្រេចសន្សំសំចៃ ប្រសិទ្ធភាព និងប្រសិទ្ធផលពិតប្រាកដឬទេ ដែលធ្វើដោយស្ថាប័នដាច់ដោយឡែកពីស្ថាប័នដែលបានអនុម័ត ឬផ្តល់វា។ ស្ថាប័នដូចជា [National Audit Office](https://en.wikipedia.org/wiki/National_Audit_Office_(United_Kingdom)) របស់ចក្រភពអង់គ្លេស និង [Government Accountability Office](https://en.wikipedia.org/wiki/Government_Accountability_Office) របស់សហរដ្ឋអាមេរិក រាយការណ៍ទៅសភានីតិបញ្ញត្តិ ជាជាងដេប៉ាតឺម៉ង់ចំណាយ ដែលជាអ្វីដែលផ្តល់ឯករាជ្យភាពដល់[សវនកម្មការអនុវត្ត](https://en.wikipedia.org/wiki/Performance_audit)។ ករណីអាជីវកម្មគឺជាការសាកល្បងមុនព្រឹត្តិការណ៍ ឯសវនកម្មតម្លៃសមស្របនឹងថវិកាគឺជាការត្រួតពិនិត្យក្រោយព្រឹត្តិការណ៍ ថាតើការសាកល្បងនោះស្មោះត្រង់ ហើយការចំណាយបានផ្តល់អ្វីដែលវាសន្យាឬអត់។

**របៀបដែលករណីអាជីវកម្មសាធារណៈខុសពីករណីវិនិយោគឯកជន** ដំណើរការពេញអ្វីទាំងអស់ខាងលើ។ ករណីវិនិយោគឯកជនបំពេញជើងមួយនៃត្រីកោណ (សូមមើល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព) ក្នុងទម្រង់តូច គឺសមត្ថភាព និងសំណើតម្លៃដែលបានបញ្ជាក់ដោយផលចំណេញលើមូលធន ព្រោះភាពស្របច្បាប់ក្នុងការចំណាយត្រូវបានសម្រេចរួចដោយភាពជាម្ចាស់។ ករណីអាជីវកម្មសាធារណៈត្រូវបំពេញជើងភាពស្របច្បាប់ដោយច្បាស់លាស់៖ អាចការពារបានចំពោះបរិយាកាសផ្តល់សិទ្ធិដែលមិនដែលទិញភាគហ៊ុន និងទទួលខុសត្រូវចំពោះតម្លៃដែលធ្លាក់លើមនុស្សដែលមិនមែនជាអ្នកទិញផ្ទាល់ ដោយប្រើប្រាស់ប្រាក់ដែលត្រូវបានប្រមូល មិនមែនរកបានក្នុងប្រតិបត្តិការទីផ្សារ។ តម្លៃបច្ចុប្បន្នសុទ្ធវិជ្ជមានគឺចាំបាច់ ប៉ុន្តែមិនដែលគ្រប់គ្រាន់ទេ។ ករណីអាជីវកម្មដែលមិនអាចឆ្លើយ "តើអ្នកផ្តល់សិទ្ធិរបស់យើង និងមនុស្សដែលរងផលប៉ះពាល់នឹងយល់ព្រមនឹងរឿងនេះឬអត់" មិនទាន់បានបញ្ចប់ការងាររបស់ខ្លួនទេ មិនថានព្វន្ធរបស់វារឹងមាំប៉ុណ្ណា។

## ការអនុវត្តល្អបំផុត

1. **សូមចាប់ផ្តើមជាមួយករណីយុទ្ធសាស្ត្រ មិនមែនសៀវភៅបញ្ជីទេ។** សូមបង្កើតបញ្ហា និងករណីសម្រាប់ការផ្លាស់ប្តូរជាភាសាធម្មតា មុនកំណត់បរិមាណអ្វីក៏ដោយ ដើម្បីឱ្យការវាយតម្លៃសាកល្បងសំណួរពិត ជាជាងបញ្ជាក់ដំណោះស្រាយដែលចូលចិត្តរួចហើយ។
2. **សូមសាងសង់បញ្ជីវែងនៃជម្រើសពិត ហើយបង្រួមវាដោយតម្លាភាព។** បញ្ជីខ្លីនៃជម្រើសដែលចូលចិត្តតែមួយ ដែលតុបតែងជាជម្រើសជំនួសជាច្រើន បរាជ័យចំពោះគោលបំណងមូលដ្ឋានរបស់ករណីសេដ្ឋកិច្ច ដែលជាការប្រៀបធៀប មិនមែនការបញ្ជាក់ទេ។
3. **សូមប្រើករណីយោងគួរឱ្យជឿជាក់ មិនមែនតួក្លែងក្លាយទេ។** សូមកំណត់ស្ថានភាពប្រៀបធៀប "ធ្វើតិចបំផុត" នៅអ្វីដែលនឹងកើតឡើងពិតប្រាកដដោយគ្មានការចំណាយ ព្រោះមូលដ្ឋានងងឹតដោយសិប្បនិម្មិតធ្វើឱ្យគ្រប់ជម្រើសមើលទៅល្អជាងវាពិតប្រាកដ។
4. **សូមអនុវត្តអត្រាបញ្ចុះតម្លៃសង្គម និងការពិចារណាអំពីការចែកចាយជាប់លាប់ ដោយទាញពីវិធីសាស្ត្រក្នុង ប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម។** សូមកុំឱ្យការវាយតម្លៃសាធារណៈនាំចូលអត្រាបញ្ចុះតម្លៃឯកជនដោយស្ងៀមស្ងាត់ ឬមិនអើពើថាអ្នកណាទទួលរងថ្លៃដើម និងអ្នកណាទទួលបានអត្ថប្រយោជន៍ (សូមមើល ប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម)។
5. **សូមកែតម្រូវយ៉ាងច្បាស់លាស់សម្រាប់ភាពលម្អៀងសុទិដ្ឋិនិយម ដោយប្រើភស្តុតាងពីគម្រោងមុនៗដែលប្រៀបធៀបបាន។** ការបន្ថែមដែលបានបញ្ជាក់ និងគណនា អាចការពារបានក្រោមការពិនិត្យ។ បន្ទាត់ទុកបម្រុងដែលមិនបានបញ្ជាក់ ទាញយកការរកឃើញថាថ្លៃដើមត្រូវបានដឹងថាប៉ាន់ស្មានទាបពេក។
6. **សូមអភិវឌ្ឍករណីពាណិជ្ជកម្ម ហិរញ្ញវត្ថុ និងការគ្រប់គ្រងរួមជាមួយករណីសេដ្ឋកិច្ច មិនមែនក្រោយវាទេ។** កម្មវិធីដែលរឹងមាំតែលើក្រដាសក្នុងករណីសេដ្ឋកិច្ច មានទំនោរបរាជ័យនៅពេលផ្តល់ មិនមែននៅពេលអនុម័តទេ។
7. **សូមដាក់ឈ្មោះម្ចាស់សម្រាប់រាល់អត្ថប្រយោជន៍រំពឹងទុកមុនអនុម័ត មិនមែនក្រោយទេ។** ផែនការសម្រេចអត្ថប្រយោជន៍ដែលគ្មានម្ចាស់ដែលបានដាក់ឈ្មោះ និងទទួលខុសត្រូវ កម្រនៅរស់រានមានជីវិតតាមអាទិភាពប្រកួតប្រជែងនៃវដ្តថវិកាបន្ទាប់។
8. **សូមបញ្ជូនការចំណាយធំៗតាមច្រកធានាឯករាជ្យ មុនការប្តេជ្ញា។** ក្រុមពិនិត្យដាច់ដោយឡែកពីកម្មវិធីអ្នកឧបត្ថម្ភ មានអំណាចនិយាយថាទេ ចាប់បានសុទិដ្ឋិនិយម និងការរីកវិសាលភាព ដែលខ្សែសង្វាក់ចុះហត្ថលេខាផ្ទៃក្នុងមិនទំនងជាជំទាស់។
9. **សូមរក្សាសវនកម្មតម្លៃសមស្របនឹងថវិកាឱ្យឯករាជ្យពីខ្សែសង្វាក់អនុម័ត ទាំងពេលវេលា និងខ្សែរាយការណ៍។** សវនកម្មដែលរាយការណ៍ទៅស្ថាប័នដែលបានអនុម័តការចំណាយ កំពុងដាក់ពិន្ទុកិច្ចការផ្ទះរបស់ខ្លួនឯង មិនថាវិសាលភាពផ្លូវការរបស់វាថាយ៉ាងណា។
10. **សូមរាយការណ៍សន្សំសំចៃ ប្រសិទ្ធភាព និងប្រសិទ្ធផលជាការរកឃើញដាច់ដោយឡែក មិនមែនពិន្ទុលាយតែមួយទេ។** កម្មវិធីអាចសន្សំសំចៃ ប៉ុន្តែគ្មានប្រសិទ្ធផល ឬមានប្រសិទ្ធផល ប៉ុន្តែតម្លៃសមស្របនឹងថវិកាអន់ ហើយការបញ្ចូលទាំងបីលាក់ភាពខុសគ្នាដែលអ្នកសម្រេចចិត្តត្រូវការ។
11. **សូមពិនិត្យករណីអាជីវកម្មឡើងវិញនៅរាល់ចំណុចសេចក្តីសម្រេចសំខាន់ មិនមែនតែនៅការអនុម័តដំបូងទេ។** ករណីយុទ្ធសាស្ត្រគម្រោង ករណីអាជីវកម្មគម្រោង និងករណីអាជីវកម្មពេញលេញ មានគោលបំណងជាការសាកល្បងបន្តបន្ទាប់ នៅពេលដឹងច្រើនជាងមុន មិនមែនជាដំណាក់កាលដែលគ្រាន់តែបិទត្រាអនុម័តមុនទេ។
12. **សូមផ្សព្វផ្សាយករណីអាជីវកម្មឱ្យបានគ្រប់គ្រាន់ ដើម្បីឱ្យបរិយាកាសផ្តល់សិទ្ធិអាចពិនិត្យវាពិតប្រាកដ។** ករណីអាជីវកម្មដែលមិនដែលចាកចេញពីមុខងារហិរញ្ញវត្ថុ មិនអាចធ្វើការងារភាពស្របច្បាប់ដែល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព ពិពណ៌នា មិនថានព្វន្ធរបស់វារឹងមាំប៉ុណ្ណា។

## សំណួរសម្រាប់ពិភាក្សាជាមួយក្រុម

**តើគំរូករណីអាជីវកម្មរបស់យើងបង្ខំឱ្យមានករណីយុទ្ធសាស្ត្រពិតប្រាកដ ឬអនុញ្ញាតឱ្យដំណោះស្រាយដែលបានជ្រើសរើសជាមុនដើរត្រង់ទៅរកលេខ?** អង្គការជាច្រើនមានគំរូដែលមានផ្នែកករណីយុទ្ធសាស្ត្រដែលបំពេញចុងក្រោយ បន្ទាប់ពីជម្រើសចូលចិត្តត្រូវបានព្រមព្រៀងរួច ដែលផលិតគោលបំណងសរសេរឱ្យត្រូវនឹងចម្លើយ ជាជាងសាកល្បងវា។ សូមប្រើសំណួរនេះដើម្បីទាញករណីអាជីវកម្មថ្មីៗបីមកពិនិត្យថាតើករណីយុទ្ធសាស្ត្រអាចនាំទៅរកជម្រើសចូលចិត្តខុសគ្នាបានឬទេ ឬវាតែងតែនឹងធ្លាក់លើជម្រើសតែមួយ។ សូមពិភាក្សាថាមានអ្វីនឹងផ្លាស់ប្តូរ ប្រសិនបើករណីយុទ្ធសាស្ត្រត្រូវចុះហត្ថលេខាមុនការវាយតម្លៃជម្រើសណាមួយចាប់ផ្តើម។ នេះមិនមែនជាការចោទប្រកាន់ថារាល់ករណីត្រូវបានសរសេរថយក្រោយទេ គ្រាន់តែជាការជំរុញឱ្យត្រួតពិនិត្យ។

**តើបច្ចុប្បន្នយើងកែតម្រូវភាពលម្អៀងសុទិដ្ឋិនិយមដោយរបៀបណា ហើយតើយើងធ្លាប់ត្រួតពិនិត្យការវាយតម្លៃរបស់យើងធៀបនឹងអ្វីដែលបានកើតឡើងពិតប្រាកដឬទេ?** អង្គការភាគច្រើនអនុវត្តទម្រង់ណាមួយនៃការទុកបម្រុង ប៉ុន្តែតិចជាងច្រើនតាមដានថាតើថ្លៃដើម និងអត្ថប្រយោជន៍ពិតរបស់គម្រោងមុនៗត្រូវនឹងអ្វីដែលករណីអាជីវកម្មបានព្យាករឬអត់។ ភាពលម្អៀងសុទិដ្ឋិនិយមមានឯកសារល្អ និងអាចទាយបានជាទូទៅក្នុងទិសដៅ ដែលជាអ្វីដែលធ្វើឱ្យវាអាចកែតម្រូវបាន ប្រសិនបើអង្គការស្មោះត្រង់អំពីកំណត់ត្រារបស់ខ្លួន។ សូមប្រើសំណួរនេះដើម្បីរកការចំណាយធំៗដែលបានបញ្ចប់ពីរបីមក ហើយប្រៀបធៀបតួលេខព្យាករជាមួយតួលេខលទ្ធផល ទោះដោយប្រមាណ។ សូមពិភាក្សាថាតើគម្លាត ប្រសិនបើមាន កំពុងត្រូវបានបញ្ចូលទៅក្នុងរបៀបវាយតម្លៃករណីអាជីវកម្មបន្ទាប់ ឬករណីថ្មីនីមួយៗចាប់ផ្តើមពីសម្មតិកម្មសុទិដ្ឋិនិយមដូចករណីមុន។

**តើអ្នកណាធ្វើសវនកម្មតម្លៃសមស្របនឹងថវិកាក្នុងអង្គការរបស់យើង ហើយតើពួកគេឯករាជ្យពីអ្នកដែលអនុម័តការចំណាយឬទេ?** ឯករាជ្យភាពគឺជាអ្វីដែលធ្វើឱ្យសវនកម្មតម្លៃសមស្របនឹងថវិកាមានតម្លៃ។ ការពិនិត្យដែលរាយការណ៍ទៅអ្នកដែលចុះហត្ថលេខាលើករណីអាជីវកម្មដើម គឺជាការគួរសម មិនមែនសវនកម្មទេ។ ស្ថាប័នសវនកម្មជាតិទទួលបានភាពជឿជាក់ព្រោះរាយការណ៍ទៅសភានីតិបញ្ញត្តិ ជាជាងទៅអ្នកប្រតិបត្តិដែលការចំណាយរបស់ពួកគេត្រូវបានពិនិត្យ ហើយមុខងារស្រដៀងគ្នាខាងក្នុងត្រូវការការញែកដូចគ្នាដើម្បីមានន័យ។ សូមប្រើសំណួរនេះដើម្បីគូសផែនទីថាអ្នកណាធ្វើការពិនិត្យតម្លៃសមស្របនឹងថវិកានៅទីនេះ លើអ្វី និងរាយការណ៍ទៅអ្នកណា។ សូមពិភាក្សាថាតើការរៀបចំនោះនឹងរួចផុតពីការពិពណ៌នាត្រង់ៗដល់អ្នកខាងក្រៅឬទេ ហើយអ្វីជាឯករាជ្យភាពពិតប្រាកដ។

**តើនៅត្រង់ណាក្នុងដំណើរការអនុម័តរបស់យើង ដែលការបរាជ័យភាពស្របច្បាប់នឹងលេចឡើងមុនប្រាក់ត្រូវបានប្តេជ្ញា ជាជាងក្រោយ?** លំនាំ "អាចផ្តល់បាន ប៉ុន្តែគ្មានការអនុញ្ញាត" របស់ប្រធានបទ 1.2 លេចឡើងនៅទីនេះជាសំណើដែលឆ្លងកាត់ករណីសេដ្ឋកិច្ច និងហិរញ្ញវត្ថុដោយរលូន ខណៈដែលអ្នកដែលត្រូវយល់ព្រមមិនត្រូវបានចូលរួមឱ្យបានត្រឹមត្រូវ។ សូមប្រើសំណួរនេះដើម្បីកំណត់ថាអ្នកណាក្នុងដំណើរការបច្ចុប្បន្ននឹងលើកការជំទាស់ភាពស្របច្បាប់ និងនៅដំណាក់កាលណា។ សូមពិភាក្សាថាតើអ្នកនោះឃើញករណីអាជីវកម្មលឿនគ្រប់គ្រាន់ដើម្បីមានសារៈសំខាន់ ឬតែនៅពេលការផ្លាស់ប្តូរមានតម្លៃថ្លៃខាងនយោបាយ។

**តើករណីពាណិជ្ជកម្ម ហិរញ្ញវត្ថុ និងការគ្រប់គ្រងទទួលបានភាពម៉ឺងម៉ាត់ដូចករណីសេដ្ឋកិច្ចឬទេ ឬសៀវភៅបញ្ជីគ្រប់គ្រងបន្ទប់?** ជំនាញវាយតម្លៃច្រើនតែប្រមូលផ្តុំស្ទើរទាំងស្រុងលើករណីសេដ្ឋកិច្ច ទុកឱ្យបីផ្សេងទៀតស្តើង ឬបំពេញជាទម្រង់។ នេះអាចយល់បាន ព្រោះករណីសេដ្ឋកិច្ចមានវិន័យវិជ្ជាជីវៈច្បាស់បំផុត និងលេខជាក់ស្តែងបំផុតដើម្បីការពារ ប៉ុន្តែវាក៏ជារបៀបដែលកម្មវិធីដែលមានតម្លៃបច្ចុប្បន្នសុទ្ធខ្លាំងបរាជ័យពេលផ្តល់ ដោយសារហេតុផលដែលករណីអាជីវកម្មមិនដែលសាកល្បងយ៉ាងម៉ឺងម៉ាត់។ សូមប្រើសំណួរនេះដើម្បីជ្រើសរើសករណីអាជីវកម្មដែលបានអនុម័តថ្មីៗមួយ ហើយវាយតម្លៃជម្រៅនៃករណីប្រាំនីមួយៗដោយស្មោះត្រង់។ សូមពិភាក្សាថាករណីខ្សោយណាបានបង្កបញ្ហាច្រើនបំផុតពេលផ្តល់ ហើយថាតើលំនាំនោះកើតឡើងម្តងទៀតនៅកន្លែងផ្សេងឬទេ។

**តើយើងលុបចោល ឬផ្លាស់ប្តូរកម្មវិធីយ៉ាងសំខាន់ដោយសារការធ្វើបច្ចុប្បន្នភាពករណីអាជីវកម្មដំណាក់កាលក្រោយបង្ហាញថាលេខបានផ្លាស់ប្តូរ ចុងក្រោយនៅពេលណា?** ករណីអាជីវកម្មដែលប្រើជាការសាកល្បងផ្ទាល់ មិនមែនច្រកម្តងគត់ ជួនកាលគួរផលិត "ទេ" ឬ "មិនមែនបែបនេះ" នៅដំណាក់កាលគម្រោង ឬករណីពេញលេញ។ ប្រសិនបើគ្មានឧទាហរណ៍មកក្នុងគំនិតភ្លាមៗ នោះផ្ទាល់ជាព័ត៌មានអំពីរបៀបដែលដំណើរការដំណើរការពិតប្រាកដនៅទីនេះ។ សូមប្រើសំណួរនេះដើម្បីរកឧទាហរណ៍ ឬកត់សម្គាល់ថាតើគ្មានទេ។ សូមពិភាក្សាថាការអវត្តមាននោះបង្ហាញអ្វីអំពីថាតើករណីដំណាក់កាលក្រោយសាកល្បងសេចក្តីសម្រេចឡើងវិញពិតប្រាកដ ឬគ្រាន់តែកត់ត្រាការប្តេជ្ញាដែលលែងត្រូវបានចោទសួរ។

## នៅក្នុងការអនុវត្ត៖ ឧទាហរណ៍តម្លៃសាធារណៈ

អាជ្ញាធរសុខភាពសាធារណៈក្នុងតំបន់មួយ ដែលជារឿងប្រឌិត ជាគំរូនៃលំនាំកើតឡើងម្តងហើយម្តងទៀត មិនមែនជារបាយការណ៍អំពីស្ថាប័នពិតប្រាកដណាមួយទេ ត្រូវសម្រេចរបៀបពង្រីកសមត្ថភាពរូបភាពវិនិច្ឆ័យទូទាំងតំបន់ដែលពេលរង់ចាំការស្កេនបានកើនឡើងអស់បីឆ្នាំ។ ជម្រើសពីរបានទៅដល់បញ្ជីខ្លី៖ មជ្ឈមណ្ឌលវិនិច្ឆ័យធំតែមួយភ្ជាប់នឹងមន្ទីរពេទ្យចម្បង ឬអង្គភាពផ្កាយរណបតូចបីដាក់ក្នុងទីក្រុងដែលខ្វះសេវា។

ករណីយុទ្ធសាស្ត្ររឹងមាំសម្រាប់ជម្រើសទាំងពីរ៖ ការកាត់បន្ថយពេលរង់ចាំ និងកែលម្អការចូលដំណើរការគឺជាគោលបំណងដែលគ្មានការជំទាស់ ដែលគាំទ្រដោយទិន្នន័យស្ទង់មតិអ្នកជំងឺ និងស្ថិតិបញ្ជូន ដែលបង្ហាញតម្រូវការមិនទាន់បានបំពេញពិតប្រាកដ។ ករណីសេដ្ឋកិច្ចបានគាំទ្រមជ្ឈមណ្ឌលតែមួយយ៉ាងច្បាស់លាស់៖ សេដ្ឋកិច្ចមាត្រដ្ឋានក្នុងឧបករណ៍ និងបុគ្គលិកបានផ្តល់សមាមាត្រអត្ថប្រយោជន៍-ថ្លៃដើមខ្ពស់ជាងអង្គភាពផ្កាយរណបបីយ៉ាងច្បាស់ សូម្បីបន្ទាប់ពីការបន្ថែមភាពលម្អៀងសុទិដ្ឋិនិយមលើថ្លៃដើមសំណង់ដែលទាញពីគម្រោងមូលធនថ្មីៗពីររបស់អាជ្ញាធរ។ តាមលេខតែម្យ៉ាង មជ្ឈមណ្ឌលគឺជាជម្រើសច្បាស់ ហើយក្រុមវាយតម្លៃបានណែនាំវា។

ករណីពាណិជ្ជកម្ម និងហិរញ្ញវត្ថុត្រូវបានដោះស្រាយដោយគ្មានការលំបាកធំ ព្រោះមជ្ឈមណ្ឌលអាចទិញតាមកិច្ចសន្យាក្របខ័ណ្ឌដែលមានស្រាប់ ហើយថ្លៃដើមមូលធនសមស្របនឹងក្របខ័ណ្ឌថវិការបស់អាជ្ញាធរ។ គឺករណីគ្រប់គ្រងដែលបង្ហាញអ្វីដែលករណីសេដ្ឋកិច្ចមិនបានចាប់យក៖ ការផ្តល់មជ្ឈមណ្ឌលពឹងផ្អែកលើការជ្រើសរើសបុគ្គលិករូបភាពវិទ្យាឯកទេស ដែលផែនការកម្លាំងពលកម្មបង្ហាញថាមិនមានក្នុងចំនួនដែលត្រូវការ ហើយនឹងត្រូវដកចេញពីទីក្រុងដែលជម្រើសផ្កាយរណបនឹងបម្រើ ដែលសមាជិកក្រុមប្រឹក្សារបស់ទីក្រុងទាំងនោះកំពុងសួរជាសាធារណៈរួចហើយថា ហេតុអ្វីអ្នកជំងឺនៅទីនោះនៅតែធ្វើដំណើរឆ្ងាយជាងមុន។

ជំនួសឱ្យការបន្តតាមនព្វន្ធនៃករណីសេដ្ឋកិច្ចតែម្នាក់ឯង ក្រុមប្រឹក្សាភិបាលបានបញ្ជូនករណីអាជីវកម្មត្រឡប់ទៅវិញសម្រាប់ជម្រើសកូនកាត់៖ មជ្ឈមណ្ឌលកណ្តាលតូចជាងជាមួយអង្គភាពផ្កាយរណបពីរ ធ្វើជាដំណាក់កាលដើម្បីឱ្យការជ្រើសរើសអាចទាន់តាមការបើកនីមួយៗ។ ករណីសេដ្ឋកិច្ចដែលកែសម្រួលបានបង្ហាញសមាមាត្រអត្ថប្រយោជន៍-ថ្លៃដើមទាបជាង ប៉ុន្តែករណីគ្រប់គ្រងឥឡូវគួរឱ្យជឿជាក់ ហើយសមាជិកក្រុមប្រឹក្សាត្រូវបានចូលរួមមុនការអនុម័តជាផ្លូវការ មិនមែនក្រោយទេ។ បីឆ្នាំក្រោយមក សវនកម្មតម្លៃសមស្របនឹងថវិកាឯករាជ្យបានរកឃើញថា ជម្រើសជាដំណាក់កាលបានផ្តល់ការកែលម្អពេលរង់ចាំដំបូងយឺតជាងអ្វីដែលករណីមជ្ឈមណ្ឌលតែមួយបានសន្យា ប៉ុន្តែបានជៀសវាងកង្វះកម្លាំងពលកម្មដែលបើមិនដូច្នេះនឹងពន្យារការបើកច្រើនជាងមួយឆ្នាំ។ សេចក្តីសន្និដ្ឋានរបស់សវនកម្មមិនមែនថាករណីសេដ្ឋកិច្ចដើមខុសតាមលក្ខខណ្ឌរបស់ខ្លួនទេ ប៉ុន្តែថាករណីអាជីវកម្មដែលសាកល្បងលើករណីសេដ្ឋកិច្ចតែម្យ៉ាង នឹងអនុម័តផែនការដែលអាជ្ញាធរមិនអាចផ្តល់ពិតប្រាកដតាមកាលវិភាគ។

## ទស្សនៈវិស័យទាំងបួន

### រដ្ឋាភិបាលមូលដ្ឋាន

អាជ្ញាធរមូលដ្ឋានជាធម្មតាដំណើរការទម្រង់ស្រាលជាងនៃគំរូករណីប្រាំ ក្រោមកម្រិតមូលធនធំៗ ដូច្នេះចំណែកធំនៃការចំណាយមូលដ្ឋានមិនដែលទទួលការព្យាបាលពេញលេញពីយុទ្ធសាស្ត្រទៅការគ្រប់គ្រងដែលកម្មវិធីហេដ្ឋារចនាសម្ព័ន្ធថ្នាក់ជាតិនឹងទទួលបាន។ នេះច្រើនតែសមាមាត្រនឹងចំនួនទឹកប្រាក់តូចជាងពិតប្រាកដ ប៉ុន្តែក៏មានន័យថាករណីពាណិជ្ជកម្ម និងការគ្រប់គ្រងដោយឡែកទំនងត្រូវបានរំលងយ៉ាងជាក់លាក់នៅកន្លែងដែលសមត្ថភាពផ្តល់ក្នុងតំបន់ស្តើងបំផុត។ ការពិនិត្យតម្លៃសមស្របនឹងថវិកាក្នុងតំបន់ជាធម្មតាមកពីសវនកម្មផ្ទៃក្នុង សវនករខាងក្រៅក្រោមរបបសវនកម្មមូលដ្ឋានថ្នាក់ជាតិ និងក្នុងប្រទេសជាច្រើន ការជំទាស់ផ្ទាល់នៅកិច្ចប្រជុំក្រុមប្រឹក្សាបើកចំហ ដែលជាសម្ពាធបរិយាកាសផ្តល់សិទ្ធិតាមពេលវេលាជាក់ស្តែង ដែលកម្មវិធីថ្នាក់ជាតិកម្រប្រឈមតាមរបៀបដូចគ្នា។

### រដ្ឋាភិបាលជាតិ

រដ្ឋាភិបាលជាតិជាធម្មតាជាកន្លែងដែលគំរូករណីប្រាំត្រូវបានអភិវឌ្ឍបំផុត ជាមួយមុខងារវាយតម្លៃ ពាណិជ្ជកម្ម និងធានាឯកទេស និងការពិនិត្យច្រកជាផ្លូវការសម្រាប់កម្មវិធីធំៗ។ ហានិភ័យនៅទីនេះផ្ទុយនឹងរដ្ឋាភិបាលមូលដ្ឋាន៖ ករណីសេដ្ឋកិច្ចដែលមានធនធានច្រើនអាចរីកចម្រើនស្មុគស្មាញរហូតគ្រប់គ្រងការសន្ទនា ខណៈដែលករណីយុទ្ធសាស្ត្រត្រូវបានសន្មតដោយស្ងៀមស្ងាត់ ជាជាងសាកល្បង ព្រោះក្រសួងដែលមានថវិកាល្អអាចធ្វើឱ្យស្ទើរតែគ្រប់សំណើមើលទៅដូចបានវាយតម្លៃ សូម្បីតែនៅពេលករណីសម្រាប់ការផ្លាស់ប្តូរមូលដ្ឋានស្តើង។ សវនកម្មតម្លៃសមស្របនឹងថវិកាថ្នាក់ជាតិ ដែលធ្វើដោយស្ថាប័នសវនកម្មកំពូលរាយការណ៍ទៅសភានីតិបញ្ញត្តិ ជាធម្មតាឯករាជ្យបំផុតក្នុងចំណោមទស្សនៈវិស័យទាំងបួននៅទីនេះ ព្រោះវាស្ថិតនៅក្រៅអ្នកប្រតិបត្តិដែលបានអនុម័តការចំណាយ។

### វិស័យសង្គម និងវិស័យមិនស្វែងរកប្រាក់ចំណេញ

អង្គការមិនស្វែងរកប្រាក់ចំណេញកម្រប្រឈមនឹងគំរូករណីអាជីវកម្មតម្រូវ ប៉ុន្តែអ្នកផ្តល់ថវិកាកាន់តែតម្រូវអ្វីមួយដែលស្រដៀងគ្នាតាមមុខងារ មុនបញ្ចេញប្រាក់ ហើយរចនាសម្ព័ន្ធករណីប្រាំបកប្រែបានសមហេតុផលសូម្បីគ្មានឈ្មោះ។ ករណីហិរញ្ញវត្ថុច្រើនតែពិបាកបំផុតសម្រាប់អង្គការមិនស្វែងរកប្រាក់ចំណេញ ព្រោះថវិកាជាទូទៅមានកម្រិតគម្រោង មានកំណត់ពេល និងទាញពីម្ចាស់ជំនួយច្រើនដែលមានច្បាប់រាយការណ៍ខុសគ្នា ដែលធ្វើឱ្យ "តើយើងមានលទ្ធភាពទទួលបានវាតាមពេលវេលាឬទេ" ពិបាកឆ្លើយជាងស្ថាប័នដែលមានមូលដ្ឋានពន្ធស្ថិរភាព។ តម្លៃសមស្របនឹងថវិកាក៏មានការជំទាស់តាមរបៀបជាក់លាក់នៅទីនេះ៖ លក្ខណៈវិនិច្ឆ័យសន្សំសំចៃរបស់អ្នកផ្តល់ថវិកា (ចំណាយតិចជាងក្នុងមួយឯកតាសកម្មភាព) អាចស្ថិតក្នុងភាពតានតឹងពិតប្រាកដជាមួយលក្ខណៈវិនិច្ឆ័យប្រសិទ្ធផលជំរុញដោយបេសកកម្ម (ទៅដល់អ្នកដែលពិបាកបម្រើបំផុត ដែលចំណាយច្រើនជាងដើម្បីទៅដល់) ដែលជាការដោះដូរដែលប្រធានបទ 3.8 — ការគ្រប់គ្រងវិស័យសង្គម និងអង្គការមិនស្វែងរកប្រាក់ចំណេញ ស្វែងយល់ពីផ្នែកគំរូថវិកា។

### កម្រិតពហុភាគី និងអន្តរជាតិ

ស្ថាប័នពហុភាគីប្រើឧបករណ៍វាយតម្លៃគម្រោងផ្ទាល់ខ្លួន ដែលស្រដៀងជាទូទៅនឹងករណីអាជីវកម្ម ដែលត្រូវបំពេញចិត្តក្រុមប្រឹក្សាអភិបាលដែលតំណាងរដ្ឋអធិបតេយ្យសមាជិកជាច្រើន ជាជាងសភានីតិបញ្ញត្តិមួយ ដែលធ្វើឱ្យជើងភាពស្របច្បាប់មានលក្ខណៈផ្លូវការជាង (ការអនុម័តរបស់ក្រុមប្រឹក្សាដែលមានឯកសារ) និងមិនសូវសម្រេច (ការអនុម័តរបស់ក្រុមប្រឹក្សាមិនផ្តល់ភាពស្របច្បាប់ដោយស្វ័យប្រវត្តិក្នុងចំណោមប្រជាជននៅកន្លែងដែលគម្រោងត្រូវបានផ្តល់ទេ)។ សវនកម្មតម្លៃសមស្របនឹងថវិកានៅទីនេះច្រើនតែធ្វើដោយមុខងារវាយតម្លៃឯករាជ្យដែលរាយការណ៍ទៅក្រុមប្រឹក្សាអភិបាល ជាជាងគណៈគ្រប់គ្រង ដែលឆ្លុះបញ្ចាំងការញែកដែលស្ថាប័នសវនកម្មកំពូលជាតិផ្តល់ក្នុងស្រុក។ ករណីសេដ្ឋកិច្ចមានភាពស្មុគស្មាញដោយផលប្រយោជន៍ខុសគ្នាពិតប្រាកដក្នុងចំណោមរដ្ឋម្ចាស់ជំនួយ រដ្ឋទទួល និងសហគមន៍រងផលប៉ះពាល់ ដែលជាបញ្ហាពហុនិយមតម្លៃដែលសៀវភៅនេះវិលត្រឡប់មកវិញជាច្រើនដង (សូមមើល ប្រធានបទ 1.4 — ទំនិញសាធារណៈ ទំនិញគួរតម្លៃ និងពហុនិយមនៃតម្លៃ)។

## ទម្រង់នៃការបរាជ័យដែលជួបញឹកញាប់

**ករណីអាជីវកម្មដឹកនាំដោយដំណោះស្រាយ។** ជម្រើសចូលចិត្តត្រូវបានជ្រើសរើសមុន ដោយក្រៅផ្លូវការ ហើយករណីយុទ្ធសាស្ត្រ និងសេដ្ឋកិច្ចត្រូវបានសរសេរក្រោយដើម្បីបញ្ជាក់វា ផលិតឯកសារដែលអានដូចម៉ឺងម៉ាត់ ខណៈមិនដែលសាកល្បងជម្រើសជំនួសពិតប្រាកដ។ ផលវិបាកជាធម្មតាមើលមិនឃើញរហូតដល់សវនកម្មតម្លៃសមស្របនឹងថវិកានៅពេលក្រោយ ឬការផ្លាស់ប្តូរថ្នាក់ដឹកនាំសួរថាហេតុអ្វីជម្រើសក្នុងបញ្ជីខ្លីមិនដែលជាអ្នកប្រកួតប្រជែងពិត។

**ភាពលម្អៀងសុទិដ្ឋិនិយមដែលមិនបានកែតម្រូវ។** ថ្លៃដើម កាលវិភាគ និងអត្ថប្រយោជន៍ត្រូវបានប៉ាន់ស្មានដោយគ្មានការកែតម្រូវផ្អែកលើភស្តុតាងសម្រាប់ទំនោរសុទិដ្ឋិនិយមហួសហេតុដែលមានឯកសារល្អ ទុកបន្ទាត់ទុកបម្រុងជាការការពារតែមួយគត់ប្រឆាំងនឹងការពិត។ ផលវិបាកធ្លាក់ជាការចំណាយលើស ឬកាលវិភាគយឺតដែលអាចរំពឹងទុកបាន ព្រោះលំនាំអាចទាយបាន សូម្បីតែនៅពេលការលើសរបស់គម្រោងនីមួយៗមានអារម្មណ៍ដូចសំណាងអាក្រក់។

**ករណីសេដ្ឋកិច្ចរឹងមាំដែលទ្រទ្រង់ដោយករណីខ្សោយបួន។** សៀវភៅបញ្ជីបង្ហាញតម្លៃបច្ចុប្បន្នសុទ្ធវិជ្ជមានច្បាស់លាស់ ហើយករណីពាណិជ្ជកម្ម ហិរញ្ញវត្ថុ និងការគ្រប់គ្រងត្រូវបានបំពេញជាទម្រង់ ជាជាងដោះស្រាយពិតប្រាកដ។ កម្មវិធីត្រូវបានអនុម័តដោយផ្អែកលើករណីមួយក្នុងប្រាំ ហើយបរាជ័យពេលផ្តល់លើវិមាត្រដែលការវាយតម្លៃមិនដែលសាកល្បងយ៉ាងម៉ឺងម៉ាត់។

**សវនកម្មតម្លៃសមស្របនឹងថវិកាដោយគ្មានឯករាជ្យភាព។** ស្ថាប័នដែលពិនិត្យថាតើប្រាក់ត្រូវបានចំណាយបានល្អឬទេ រាយការណ៍ ដោយផ្ទាល់ ឬដោយប្រយោល ទៅស្ថាប័នដែលបានចំណាយវា ដូច្នេះការរកឃើញដែលនឹងធ្វើឱ្យអ្នកឧបត្ថម្ភអាម៉ាស់ត្រូវបានបន្ទន់ ឬមិនដែលលេចឡើង។ នេះធ្វើឱ្យអង្គការបាត់បង់សញ្ញាមានប្រយោជន៍បំផុតតែមួយដែលវាមានសម្រាប់កែតម្រូវករណីអាជីវកម្មនាពេលអនាគត ព្រោះមុខងារសវនកម្មដែលត្រូវបានចាប់យកប្រាប់ថ្នាក់ដឹកនាំនូវអ្វីដែលវាជឿរួចហើយ។

- New Zealand Treasury, *Better Business Cases* guidance — ក្របខ័ណ្ឌគំរូករណីប្រាំដែលអភិវឌ្ឍរួមគ្នាជាមួយ UK Treasury និងសម្របសម្រាប់វិស័យសាធារណៈរបស់នូវែលសេឡង់។

- US Government Accountability Office, performance audit work — ស្ថាប័នស្រដៀងគ្នារបស់សហរដ្ឋអាមេរិកចំពោះសវនកម្មតម្លៃសមស្របនឹងថវិកា ដែលរាយការណ៍ទៅ Congress។
- OECD, Recommendation on Effective Public Investment Across Levels of Government និងឧបករណ៍ដែលភ្ជាប់ — គោលការណ៍អន្តរជាតិសម្រាប់អភិបាលកិច្ចវិនិយោគសាធារណៈ។
