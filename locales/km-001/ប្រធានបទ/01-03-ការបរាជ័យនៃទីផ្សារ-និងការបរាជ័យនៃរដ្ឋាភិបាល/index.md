# ប្រធានបទ 1.3 — ការបរាជ័យនៃទីផ្សារ និងការបរាជ័យនៃរដ្ឋាភិបាល

**ទ្រឹស្តីតម្លៃសាធារណៈមានឡើង ព្រោះទាំង «ទីផ្សារនឹងដោះស្រាយវា» និង «រដ្ឋាភិបាលនឹងដោះស្រាយវា» មិនមែនជាលំនាំដើមដែលមានសុវត្ថិភាពទេ៖ របៀបនៃការផ្គត់ផ្គង់នីមួយៗបរាជ័យតាមរបៀបប្លែកខ្លួន និងអាចទាយទុកបាន ហើយការវិនិច្ឆ័យថាតើអ្នកអាចរស់នៅជាមួយការបរាជ័យមួយណា — ឬរបៀបលាយបញ្ចូលទាំងពីរ — គឺខ្លួនវាជាការសម្រេចចិត្តយុទ្ធសាស្ត្រស្នូលដែលអ្នកគ្រប់គ្រងសាធារណៈធ្វើ។**

## ហេតុអ្វីបានជាបញ្ហានេះសំខាន់សម្រាប់តម្លៃសាធារណៈ

អ្នកដឹកនាំវិស័យសាធារណៈ និងវិស័យសង្គមគ្រប់រូប នៅទីបំផុតត្រូវប្រឈមនឹងសំណួរដដែលក្នុងទម្រង់មួយ៖ តើសេវាកម្មនេះគួរត្រូវបានផ្តល់ដោយទីផ្សារ ដោយរដ្ឋាភិបាល ដោយអង្គការមិនស្វែងរកប្រាក់ចំណេញ ឬដោយការលាយបញ្ចូលនៃទាំងបី? សំណួរនេះមកដល់ក្នុងសម្លៀកបំពាក់នៃចំណុចថវិកា ការសម្រេចចិត្តលទ្ធកម្ម សំណើឯកជនភាវូបនីយកម្ម ឬការសន្យាឃោសនាបោះឆ្នោតក្នុងការ «ដណ្តើមត្រួតត្រាឡើងវិញ» លើសេវាសាធារណូបភោគ ប៉ុន្តែនៅខាងក្រោមវាជានិច្ចជាជម្រើសដដែលអំពីអ្នកណាគួរផ្តល់អ្វី និងហេតុអ្វី។

ការវិភាគប្រកួតប្រជែងគ្នាពីរកំពុងប្រជែងរកចម្លើយ ហើយទាំងពីរត្រូវពាក់កណ្តាល។ ការវិភាគការបរាជ័យនៃទីផ្សារនិយាយថា រដ្ឋាភិបាលគួរចូលជួយនៅគ្រប់កន្លែងដែលទីផ្សារផលិតតិចជាងអ្វីដែលពលរដ្ឋឱ្យតម្លៃ — ខ្យល់ស្អាត ការចាក់វ៉ាក់សាំង ការការពារទឹកជំនន់ — ព្រោះគ្មានអ្នកទិញឯកជនអាចចាប់យកអត្ថប្រយោជន៍ពេញលេញបានទេ។ ការវិភាគការបរាជ័យនៃរដ្ឋាភិបាលនិយាយថា ការផ្គត់ផ្គង់សមូហភាពមិនមែនជាថ្នាំព្យាបាលស្វ័យប្រវត្តិទេ ព្រោះមន្ត្រី និងអាជ្ញាធរបទប្បញ្ញត្តិដែលត្រូវបានស្នើឱ្យជួសជុលទីផ្សារ មានបញ្ហាការជំរុញដែលពិតដូចទីផ្សារដែរ។ អ្នកដឹកនាំដែលអានតែពាក់កណ្តាលទីមួយ ក្លាយជាអ្នកពង្រីករដ្ឋដោយមិនពិនិត្យ អ្នកដែលអានតែពាក់កណ្តាលទីពីរ ក្លាយជាអ្នកធ្វើឯកជនភាវូបនីយកម្មដោយមិនពិនិត្យ ដោយសន្មតថារាល់ការបរាជ័យសាធារណៈបង្ហាញថាទីផ្សារប្រសើរជាង សូម្បីតែនៅកន្លែងដែលទំនិញនោះជាអ្វីមួយដែលទីផ្សារមិនអាចកំណត់តម្លៃបានតាមរចនាសម្ព័ន្ធ។

ប្រធានបទនេះផ្តល់ការវិភាគទាំងពីរទន្ទឹមគ្នា ព្រោះការកាន់តែមួយគឺជាបំណុលវិជ្ជាជីវៈ។ វាក៏រៀបចំជាមូលដ្ឋានសម្រាប់ផ្នែកដែលនៅសល់នៃផ្នែកទី 1 ផងដែរ៖ ត្រីកោណយុទ្ធសាស្ត្រ (សូមមើល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព) គឺជាឧបករណ៍សម្រាប់ថ្លឹងថ្លែងជម្រើសផ្គត់ផ្គង់ នៅពេលអ្នកដឹងថាការបរាជ័យណាខ្លះកំពុងមានតួនាទី ហើយការចាត់ថ្នាក់ទំនិញ (សូមមើល ប្រធានបទ 1.4 — ទំនិញសាធារណៈ ទំនិញគួរតម្លៃ និងពហុនិយមនៃតម្លៃ) ដាក់ឈ្មោះច្បាស់លាស់ថាតើអ្នកកំពុងដោះស្រាយជាមួយទំនិញប្រភេទអ្វី មុនពេលអ្នកជ្រើសរើសថាអ្នកណាគួរផ្គត់ផ្គង់។

## គំនិតស្នូល

**ការបរាជ័យនៃទីផ្សារ** គឺជាពាក្យទូទៅសម្រាប់ការប្តូរប្រាស់ឯកជនដែលគ្មានបទប្បញ្ញត្តិ ផលិតលទ្ធផលដែលមិនមានប្រសិទ្ធភាពតាមស្តង់ដាររបស់ទីផ្សារផ្ទាល់ — តម្លៃដែលមិនត្រូវបានបង្កើត ឬធនធានដែលត្រូវបានបែងចែកមិនត្រឹមត្រូវធៀបនឹងទីផ្សារប្រកួតប្រជែងដ៏ល្អឥតខ្ចោះ។ វាមិនមែនជាការអះអាងថាទីផ្សារអាក្រក់ជាទូទៅទេ គ្រាន់តែថាលក្ខខណ្ឌជាក់លាក់ធ្វើឱ្យយន្តការដែលធម្មតាធ្វើឱ្យវាដំណើរការខូច។

**ទំនិញសាធារណៈ** គឺមិនអាចបំបែកបាន (អ្នកមិនអាចបញ្ឈប់អ្នកមិនបង់ប្រាក់មិនឱ្យប្រើប្រាស់បានទេ) និងមិនប្រកួតប្រជែង (ការប្រើប្រាស់របស់មនុស្សម្នាក់មិនកាត់បន្ថយអ្វីដែលនៅសល់សម្រាប់អ្នកផ្សេងឡើយ) — ភ្លើងបំភ្លឺផ្លូវ និងការពារជាតិគឺជាឧទាហរណ៍ក្នុងសៀវភៅសិក្សា។ ដោយសារគ្មានក្រុមហ៊ុនឯកជនអាចគិតថ្លៃអ្នកមិនបង់ប្រាក់បាន ទីផ្សារឯកជនផ្គត់ផ្គង់ទំនិញទាំងនេះតិចជាងតម្រូវការជាប្រព័ន្ធ ដែលជាករណីបុរាណសម្រាប់ការផ្គត់ផ្គង់ដោយមូលនិធិពន្ធ។

**ផលប៉ះពាល់ខាងក្រៅ** គឺជាថ្លៃដើម ឬអត្ថប្រយោជន៍នៃសកម្មភាពមួយ ដែលធ្លាក់លើអ្នកមិនមែនជាភាគីនៃប្រតិបត្តិការដែលបង្កឱ្យមានវា៖ រោងចក្រដែលបំពុលទន្លេដាក់ថ្លៃដើមលើអ្នកប្រើនៅខាងក្រោមដែលមិនបានបង់អ្វីសោះ ខណៈដែលបុគ្គលដែលបានចាក់វ៉ាក់សាំងផ្តល់អត្ថប្រយោជន៍ដល់អ្នកគ្រប់គ្នាដែលហានិភ័យឆ្លងរបស់ពួកគេធ្លាក់ចុះ។ ប្រសិនបើទុកតាមធម្មជាតិ ទីផ្សារផលិតសកម្មភាពដែលមានផលប៉ះពាល់ខាងក្រៅអវិជ្ជមានលើសតម្រូវការ និងផលិតសកម្មភាពដែលមានផលប៉ះពាល់ខាងក្រៅវិជ្ជមានតិចជាងតម្រូវការ។

**អសមមាត្រនៃព័ត៌មាន** កើតឡើងនៅពេលភាគីមួយនៃប្រតិបត្តិការដឹងច្រើនជាងភាគីម្ខាងទៀតយ៉ាងសំខាន់ — អ្នកលក់ឡានមួយទឹកដឹងអំពីប្រវត្តិរថយន្តច្រើនជាងអ្នកទិញ អ្នកជំងឺដឹងតិចជាងអំពីរោគវិនិច្ឆ័យរបស់ខ្លួនជាងគ្រូពេទ្យ។ ប្រសិនបើមិនដោះស្រាយ វាអាចបង្កឱ្យមានការជ្រើសរើសមិនល្អ (មានតែហានិភ័យអាក្រក់បំផុតប៉ុណ្ណោះដែលនៅក្នុងក្រុមធានារ៉ាប់រងនៅពេលបុព្វលាភកើនឡើង) ឬហានិភ័យសីលធម៌ (ភាគីដែលត្រូវបានការពារពីហានិភ័យទទួលហានិភ័យនោះកាន់តែច្រើន) ដែលទាំងពីរអាចធ្វើឱ្យទីផ្សារតូចទៅ ឬដួលរលំ ជាជាងគ្រាន់តែកំណត់តម្លៃខុស។

**ភាពផ្តាច់មុខតាមធម្មជាតិ** មាននៅកន្លែងដែលថ្លៃដើមថេររបស់ឧស្សាហកម្មមួយធំណាស់ធៀបនឹងថ្លៃដើមនៃការបម្រើអតិថិជនបន្ថែមមួយ ដល់ថាអ្នកផ្គត់ផ្គង់តែមួយអាចបំពេញទីផ្សារទាំងមូលបានថោកជាងអ្នកប្រកួតប្រជែង — បំពង់ទឹក និងបណ្តាញបញ្ជូនអគ្គិសនីជាឧទាហរណ៍ទូទៅ ព្រោះការធ្វើបំពង់ ឬខ្សែស្ទួនគ្នានឹងខ្ជះខ្ជាយធនធាន។ **អំណាចទីផ្សារ** គឺជាករណីទូទៅដែលភាពផ្តាច់មុខតាមធម្មជាតិជាកំណែខ្លាំងបំផុត៖ សមត្ថភាពរបស់ក្រុមហ៊ុនមួយក្នុងការបង្កើនតម្លៃ ឬកម្រិតផលិតកម្ម លើសពីកម្រិតប្រកួតប្រជែង ដោយមិនបាត់បង់អតិថិជនទាំងអស់។

**ការបរាជ័យនៃរដ្ឋាភិបាល** គឺជាគំនិតបញ្ច្រាស៖ អន្តរាគមន៍ដែលមានបំណងកែតម្រូវការបរាជ័យនៃទីផ្សារ ផ្ទុយទៅវិញទុកធនធានឱ្យបែងចែកអាក្រក់ជាងទីផ្សារមានកំហុសដែលវាជំនួស។ វាត្រូវបានសិក្សាក្រោម **ទ្រឹស្តីជម្រើសសាធារណៈ** ដែលជាសាខានៃសេដ្ឋកិច្ចវិទ្យាដែលអនុវត្តការសន្មតអំពីអ្នកដើរតួដោយហេតុផល ដែលស្វែងរកផលប្រយោជន៍ផ្ទាល់ខ្លួនចំពោះអ្នកបោះឆ្នោត អ្នកនយោបាយ និងមន្ត្រីរាជការ ដូចសេដ្ឋកិច្ចវិទ្យាបុរាណអនុវត្តចំពោះអ្នកប្រើប្រាស់ និងក្រុមហ៊ុន ដោយបោះបង់គំនិតចាស់ដែលចាត់ទុកមន្ត្រីជាអ្នកថែរក្សាមិនលំអៀងនៃផលប្រយោជន៍សាធារណៈ។

**ការស្វែងរកចំណូលពីសិទ្ធិពិសេស (Rent-seeking)** គឺជាការស្វែងរកប្រាក់ចំណូលតាមរយៈការរៀបចំច្បាប់ ឬការបែងចែក ជាជាងតាមរយៈការផលិតអ្វីដែលមានតម្លៃ — ការបញ្ចុះបញ្ចូលសម្រាប់ពន្ធការពារ អាជ្ញាបណ្ណ ឬឧបត្ថម្ភធនគឺជាការស្វែងរកចំណូលពីសិទ្ធិពិសេស នៅពេលកិច្ចខិតខំប្រឹងប្រែងចាប់យកតម្លៃដែលមានស្រាប់ ជាជាងបង្កើតតម្លៃថ្មី។ **ការចាប់យកអាជ្ញាធរបទប្បញ្ញត្តិ** គឺជាទម្រង់ជាក់លាក់ និងបំផ្លិចបំផ្លាញនៃរឿងនេះ៖ អាជ្ញាធរបទប្បញ្ញត្តិដែលបង្កើតឡើងដើម្បីត្រួតពិនិត្យឧស្សាហកម្មមួយ ផ្ទុយទៅវិញមកបម្រើផលប្រយោជន៍ឧស្សាហកម្មនោះ ជាធម្មតាតាមរយៈការចែករំលែកជំនាញ ឱកាសការងារក្រោយចាកចេញពីតំណែង និងការបញ្ចុះបញ្ចូលអសមាមាត្រ ជាជាងអំពើពុករលួយច្បាស់ៗ។

**X-inefficiency** ជាពាក្យដែលបង្កើតឡើងដោយសេដ្ឋវិទូ Harvey Leibenstein ដាក់ឈ្មោះភាពរលុងខាងក្នុង — បុគ្គលិកលើសចំនួន វិន័យថ្លៃដើមខ្សោយ កម្លាំងប្រឹងប្រែងទាប — ដែលចូលក្នុងអង្គការណាមួយដែលត្រូវបានការពារពីសម្ពាធប្រកួតប្រជែង ទោះជាក្រុមហ៊ុនផ្តាច់មុខ ឬទីភ្នាក់ងារសាធារណៈផ្តាច់មុខក៏ដោយ។ **បញ្ហាម្ចាស់ការ–ភ្នាក់ងារ** ពិពណ៌នាគម្លាតដែលបើកឡើងនៅពេលណាក៏ដោយដែលភាគីមួយ (ភ្នាក់ងារ) ធ្វើសកម្មភាពជំនួសភាគីមួយទៀត (ម្ចាស់ការ) ប៉ុន្តែមានផលប្រយោជន៍ផ្សេងគ្នា និងព័ត៌មានល្អជាង៖ ពលរដ្ឋប្រគល់ឱ្យអ្នកនយោបាយ ដែលប្រគល់ឱ្យមន្ត្រីរាជការ ខ្សែចំណងនីមួយៗបន្ថែមព័ត៌មានដែលម្ចាស់ការមិនអាចត្រួតពិនិត្យបានទាំងស្រុង។ William A. Niskanen បានធ្វើគំរូប្រធានទីភ្នាក់ងារជាអ្នកបង្កើនថវិកាអតិបរមា ដែលការជំរុញរបស់ពួកគេរុញទៅរកថវិកាធំជាងអ្វីដែលផលប្រយោជន៍សាធារណៈនឹងបង្ហាញហេតុផលបាន ព្រោះសមាជិកនីតិបញ្ញត្តិមិនងាយសង្កេតឃើញថ្លៃដើមពិតរបស់ប្រព័ន្ធរដ្ឋបាល។

## ការអនុវត្តល្អបំផុត

1. **សូមដាក់ឈ្មោះការបរាជ័យនៃទីផ្សារជាក់លាក់មុនពេលអ្នកដាក់ឈ្មោះដំណោះស្រាយ។** សូមបញ្ជាក់ថាតើអ្នកប្រឈមនឹងទំនិញសាធារណៈ ផលប៉ះពាល់ខាងក្រៅ អសមមាត្រនៃព័ត៌មាន ភាពផ្តាច់មុខតាមធម្មជាតិ ឬអំណាចទីផ្សារធម្មតា ព្រោះនីមួយៗបញ្ជាក់ការជួសជុលខុសគ្នា — ឧបត្ថម្ភធន ពន្ធ ច្បាប់បង្ហាញព័ត៌មាន បទប្បញ្ញត្តិតម្លៃ និងគោលនយោបាយប្រកួតប្រជែង មិនអាចប្តូរគ្នាបានទេ។
2. **សូមចាត់ទុកការបរាជ័យនៃរដ្ឋាភិបាលជាការវិនិច្ឆ័យ មិនមែនជាការប្រមាថទេ។** ការស្វែងរកចំណូលពីសិទ្ធិពិសេស ការចាប់យក និង X-inefficiency គឺជាលំនាំជាក់លាក់ដែលអាចសង្កេតបាន ដែលមានមូលហេតុជាក់លាក់ សូមរកឱ្យឃើញថាមួយណា (ប្រសិនបើមាន) ពិតជាមានមុនពេលសន្មតថាការកែទម្រង់មានន័យត្រឹមតែការបង្រួមអង្គការ។
3. **សូមផ្គូផ្គងឧបករណ៍ទៅនឹងការបរាជ័យ មិនមែនទៅនឹងជំហរលំនាំដើមរបស់អង្គការអ្នកទេ។** ទីភ្នាក់ងារដែលតែងតែដាក់បទប្បញ្ញត្តិ តែងតែចុះកិច្ចសន្យាខាងក្រៅ ឬតែងតែយកសេវាមកធ្វើខាងក្នុង កំពុងធ្វើឱ្យប្រសើរតាមទម្លាប់ ជាជាងតាមការបរាជ័យដែលនៅចំពោះមុខ។
4. **នៅកន្លែងដែលភាពផ្តាច់មុខតាមធម្មជាតិមិនអាចជៀសវាងបាន សូមដាក់បទប្បញ្ញត្តិ ឬគ្រប់គ្រងវា ប៉ុន្តែរក្សាទីផ្សារធាតុចូលជុំវិញវាឱ្យមានការប្រកួតប្រជែង។** អ្នកកម្រអាចបង្កើតការប្រកួតប្រជែងសម្រាប់បំពង់ទឹកផ្ទាល់ ប៉ុន្តែអ្នកអាចសម្រាប់ក្រុមហ៊ុនដែលសាងសង់ថែទាំ ឬចេញវិក្កយបត្រលើបំពង់នោះ។
5. **សូមបំបែកតួនាទីរបស់អ្នកផ្តល់មូលនិធិ អាជ្ញាធរបទប្បញ្ញត្តិ និងអ្នកផ្គត់ផ្គង់នៅគ្រប់កន្លែងដែលអាចធ្វើបាន។** ការបញ្ចូលទាំងបីក្នុងស្ថាប័នតែមួយដកការត្រួតពិនិត្យខាងក្នុងដែលនឹងលាតត្រដាងការស្វែងរកចំណូលពីសិទ្ធិពិសេស ឬការចាប់យក មុនពេលវាចាក់ឫស។
6. **សូមផ្តល់រង្វាស់តម្លៃច្បាស់លាស់ដល់លទ្ធផលមិនមែនទីផ្សារ ហើយប្រើការប្រៀបធៀបដើម្បីជំនួសសម្ពាធប្រកួតប្រជែងដែលបាត់។** ក្នុងករណីគ្មានតម្លៃទីផ្សារ ទីភ្នាក់ងារគ្មានវិន័យស្វ័យប្រវត្តិប្រឆាំងនឹង X-inefficiency ទេ ថ្លៃដើមក្នុងមួយឯកតា ឬរង្វាស់លទ្ធផលដែលបានបង្កើត ដែលប្រៀបធៀបជាមួយអ្នកផ្គត់ផ្គង់ដែលអាចប្រៀបធៀបបាន បង្កើតវិន័យនោះឡើងវិញមួយចំណែក។
7. **សូមផ្សព្វផ្សាយថាអាជ្ញាធរបទប្បញ្ញត្តិជួបអ្នកណា និងនៅពេលណា ទៀងទាត់ដូចការផ្សព្វផ្សាយសេចក្តីសម្រេចរបស់ខ្លួនដែរ។** ការចាប់យកជាទូទៅវិវឌ្ឍបន្តិចម្តងៗ និងស្របច្បាប់ តាមរយៈអត្ថប្រយោជន៍ព័ត៌មាន និងការផ្លាស់ប្តូរបុគ្គលិកតាមទ្វារបង្វិល ដូច្នេះតម្លាភាពអំពីទំនាក់ទំនងថោកជាងការស៊ើបអង្កេតអំពើពុករលួយក្រោយហេតុការណ៍។
8. **សូមបញ្ចូលចំណុចកំណត់ពេលវេលា ឬការពិនិត្យឡើងវិញក្នុងអន្តរាគមន៍ណាមួយដែលមានហេតុផលពីការបរាជ័យនៃទីផ្សារ។** លក្ខខណ្ឌដែលបង្ហាញហេតុផលវាអាចផ្លាស់ប្តូរ អំណាចដែលផ្តល់ដើម្បីជួសជុលបញ្ហាប្រចាំទសវត្សរ៍មួយ មិនគួរក្លាយជាសិទ្ធិអចិន្ត្រៃយ៍ដោយស្ងាត់ៗឡើយ។
9. **សូមបញ្ជូនសិទ្ធិពិសេសណាមួយដែលការផ្គត់ផ្គង់បង្កើតពិតប្រាកដត្រឡប់ទៅសាធារណជនវិញ ជាជាងធ្វើពុតថាវាមិនមាន។** អាជ្ញាបណ្ណ ឬសិទ្ធិផ្តាច់មុខដែលបង្កើតតម្លៃកម្រ គួរមានតម្លៃនោះត្រូវបានចាប់យកតាមរយៈការដេញថ្លៃ ឬថ្លៃសេវា មិនមែនទុកជាផលចំណេញឥតបានគិតសម្រាប់អ្នកកាន់វានោះទេ។
10. **សូមថ្លឹងថ្លែងសមត្ថភាពប្រតិបត្តិការជាមួយភាពស្របច្បាប់ និងតម្លៃ មុនពេលសម្រេចថាអ្នកណាផ្តល់ ហើយពិនិត្យព្រំដែននោះឡើងវិញតាមវដ្តកំណត់។** ដំណោះស្រាយដែលមើលទៅត្រឹមត្រូវលើបញ្ជីការបរាជ័យនៃទីផ្សារ/ការបរាជ័យនៃរដ្ឋាភិបាល ប៉ុន្តែអង្គការរបស់អ្នកមិនអាចផ្តល់ជូនបាន មិនមែនជាជម្រើសពិតប្រាកដទេ ហើយការរង់ចាំរហូតដល់រឿងអាស្រូវជំរុញការពិនិត្យឡើងវិញធានាថាវាកើតឡើងក្នុងលក្ខខណ្ឌអាក្រក់បំផុត (សូមមើល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព)។

## សំណួរសម្រាប់ពិភាក្សាជាមួយក្រុម

**តើការបរាជ័យនៃទីផ្សារជាក់លាក់ណាមួយ (ប្រសិនបើមាន) ដែលបង្ហាញហេតុផលសម្រាប់ការចូលរួមរបស់យើងក្នុងសេវាកម្មនេះ?** សូមឆ្លងកាត់ការចាត់ថ្នាក់ — ទំនិញសាធារណៈ ផលប៉ះពាល់ខាងក្រៅ អសមមាត្រនៃព័ត៌មាន ភាពផ្តាច់មុខតាមធម្មជាតិ អំណាចទីផ្សារ — ហើយសូមស្មោះត្រង់ប្រសិនបើគ្មានអ្វីមួយអនុវត្តបានច្បាស់។ ការលើកឡើងមិនច្បាស់អំពី «ការបរាជ័យនៃទីផ្សារ» ជាទូទៅមិនមែនជាការវិនិច្ឆ័យទេ វាជាវិធីដើម្បីគេចពីការវិនិច្ឆ័យ។ ប្រសិនបើអង្គការរបស់អ្នកផ្តល់អ្វីមួយដែលទីផ្សារប្រកួតប្រជែងអាចផ្គត់ផ្គង់បានយ៉ាងល្អឥតខ្ចោះ ហេតុផលត្រូវមកពីកន្លែងផ្សេង ដូចជាសមធម៌ ហើយអ្នកគួរនិយាយយ៉ាងត្រង់ ជាជាងស្វែងរកអំណះអំណាងដែលមិនត្រូវគ្នា។ សូមសរសេរការវិនិច្ឆ័យចុះមុនជុំថវិកាបន្ទាប់ ដើម្បីឱ្យក្រុមជជែកអំពីការអះអាងដូចគ្នា ជាជាងនិយាយឆ្លងគ្នា។

**តើនៅកន្លែងណាក្នុងអង្គការរបស់យើងដែលមន្ត្រីបង្កើនថវិកាអតិបរមាមានកន្លែងធ្វើការច្រើនបំផុត?** គំរូរបស់ Niskanen សន្មតថាប្រធានរដ្ឋបាលរុញរកថវិកា និងចំនួនបុគ្គលិកធំជាង ព្រោះសមាជិកនីតិបញ្ញត្តិ និងក្រុមប្រឹក្សាមិនងាយផ្ទៀងផ្ទាត់ថ្លៃដើមពិតរបស់ទីភ្នាក់ងារ។ សូមកំណត់កន្លែងដែលការត្រួតពិនិត្យស្តើងបំផុតក្នុងរចនាសម្ព័ន្ធរបស់អ្នក — មុខងារបច្ចេកទេសដែលក្រុមប្រឹក្សារបស់អ្នកមិនអាចវាយតម្លៃដោយឯករាជ្យ ឬកម្មវិធីដែលគ្មានស្តង់ដារប្រៀបធៀបពីខាងក្រៅ។ សូមសួរថាតើភស្តុតាងអ្វី (ប្រសិនបើមាន) ដែលនឹងលាតត្រដាងភាពរលុង ប្រសិនបើវាបានលូតលាស់នៅទីនោះ ជាជាងសន្មតថាបំណងល្អគ្រប់គ្រាន់ជាការការពារ។ ប្រសិនបើគ្មាននរណាអាចឆ្លើយដោយយន្តការជាក់លាក់ សូមចាត់ទុកភាពស្ងៀមស្ងាត់នោះជាការរកឃើញខ្លួនឯង។

**តើនរណាត្រួតពិនិត្យអាជ្ញាធរបទប្បញ្ញត្តិ ហើយតើយើងនឹងដឹងដោយរបៀបណាប្រសិនបើការចាប់យកកំពុងចាប់ផ្តើម?** ការចាប់យកកម្រប្រកាសខ្លួនជាអំពើពុករលួយ វាលេចឡើងជាការរសាត់យឺតៗនៃភស្តុតាងរបស់អ្នកណាត្រូវបានថ្លឹងខ្លាំងបំផុត កិច្ចប្រជុំរបស់អ្នកណាកើតឡើងញឹកញាប់បំផុត និងបុគ្គលិករបស់អ្នកណាផ្លាស់ប្តូររវាងអាជ្ញាធរបទប្បញ្ញត្តិ និងអ្នកត្រូវគ្រប់គ្រងបានស្រួលបំផុត។ គ្មានជំហានណាមួយមើលទៅមិនសមរម្យដោយឡែកទេ ដែលជាមូលហេតុពិតប្រាកដដែលលំនាំនេះគ្រោះថ្នាក់។ សូមពិភាក្សាថាតើសូចនាករព្រមានមុនអ្វីដែលអង្គការរបស់អ្នកតាមដាន (ប្រសិនបើមាន) និងអ្នកណានៅខាងក្រៅទំនាក់ទំនងដែលស្ថិតក្នុងទីតាំងសម្គាល់ការរសាត់មុនវាក្លាយជារឿងអាស្រូវ។ ប្រសិនបើចម្លើយស្មោះត្រង់គឺ «គ្មាននរណាទេ» គម្លាតនោះគឺខ្លួនវាសមនឹងជួសជុល។

**ប្រសិនបើយើងប្រគល់មុខងារនេះដល់អ្នកផ្តល់សេវាឯកជន ឬអង្គការមិនស្វែងរកប្រាក់ចំណេញនៅថ្ងៃស្អែក តើយើងកំពុងដោះដូរទម្រង់បរាជ័យមួយណាជាមួយមួយណា?** រាល់ជម្រើសផ្គត់ផ្គង់ដោះដូរទម្រង់ហានិភ័យមួយជាមួយមួយទៀត ជាជាងលុបបំបាត់ហានិភ័យទាំងស្រុង។ ការចុះកិច្ចសន្យាខាងក្រៅអាចដោះស្រាយបញ្ហា X-inefficiency ខណៈដែលណែនាំអសមមាត្រនៃព័ត៌មានថ្មី ឬដោះដូរហានិភ័យការចាប់យកជាមួយហានិភ័យតម្លៃភាពផ្តាច់មុខតាមធម្មជាតិ ប្រសិនបើមានអ្នកដេញថ្លៃគួរឱ្យជឿជាក់តែម្នាក់។ សូមប្រើសំណួរនេះដើម្បីបង្ខំឱ្យមានការប្រៀបធៀបដូចគ្នា ដោយដាក់ឈ្មោះការបរាជ័យជាក់លាក់ដែលអ្នករំពឹងថានឹងទទួលបាន និងការបរាជ័យដែលអ្នករំពឹងថានឹងបានមក ជាជាងទុកឱ្យការសន្មតមិនបានពិនិត្យថាជម្រើសជំនួសគ្មានការបរាជ័យមិនត្រូវបានជំទាស់។

**តើយើងកំពុងវាស់ថ្លៃដើម ឬកំពុងវាស់តម្លៃ?** ក្នុងករណីគ្មានតម្លៃទីផ្សារ វាងាយក្នុងការតាមដានអ្វីដែលបានចំណាយ ជាជាងអ្វីដែលបានសម្រេច ដែលជាលក្ខខណ្ឌពិតប្រាកដដែល X-inefficiency មិនត្រូវបានរកឃើញអស់រយៈពេលជាច្រើនឆ្នាំ។ គំរូចំណាយស្ថិរភាព និងមិនគួរឱ្យកត់សម្គាល់អាចមើលទៅដូចគ្នា មិនថាសេវាមូលដ្ឋានល្អឥតខ្ចោះ ឬកំពុងរលាយបន្តិចម្តងៗ។ សូមពិភាក្សាថាតើរង្វាស់បច្ចុប្បន្នរបស់អ្នកនឹងលាតត្រដាងភាពរលុងដែលលូតលាស់ពិតប្រាកដឬអត់ ហើយប្រសិនបើចម្លើយមិនច្បាស់ សូមចាត់ទុកការកសាងសមត្ថភាពនោះជាបន្ទាន់ជាងចំណុចថវិកាណាមួយ។ សូមប្រៀបធៀបចម្លើយរបស់អ្នកជាមួយរបៀបដែលដៃគូប្រកួតប្រជែងទីផ្សារពិតប្រាកដនឹងត្រូវបានវិនិច្ឆ័យ តាមតម្លៃ និងគុណភាពរួមគ្នា មិនមែនតែការចំណាយតែម្នាក់ឯងទេ។

**តើអ្វីដែលភាពស្របច្បាប់ និងការគាំទ្រមិនអាចរស់រានមានជីវិត ទោះបីជាករណីប្រសិទ្ធភាពសម្រាប់អ្នកផ្តល់សេវាផ្សេងមិនអាចជំទាស់បានក៏ដោយ?** ដំណោះស្រាយមួយចំនួនដែលមើលទៅមានប្រសិទ្ធភាពលើការវិភាគការបរាជ័យនៃទីផ្សារ/ការបរាជ័យនៃរដ្ឋាភិបាល នឹងនៅតែត្រូវបានបដិសេធដោយពលរដ្ឋ ឬអ្នកតំណាងជាប់ឆ្នោតដោយមិនគិតពីលេខ — ឯកជនភាវូបនីយកម្មសេវាមូលដ្ឋានដែលគេស្រលាញ់ជាឧទាហរណ៍ ឬការជួលអ្នកខាងក្រៅធ្វើមុខងារដែលពលរដ្ឋរំពឹងថាមន្ត្រីសាធារណៈពាក់ឯកសណ្ឋានត្រូវធ្វើ។ សូមពិភាក្សាថាតើបន្ទាត់នោះស្ថិតនៅទីណាពិតប្រាកដសម្រាប់អង្គការរបស់អ្នក ដោយផ្អែកលើភស្តុតាងនៃអ្វីដែលពលរដ្ឋបានទទួលយក ឬបដិសេធពីមុន មិនមែនការទាយរបស់ថ្នាក់ដឹកនាំទេ។ សូមសួរថាតើថ្នាក់ដឹកនាំស្មោះត្រង់អំពីឧបសគ្គនោះឬអត់ ឬកំពុងតុបតែងវាដោយស្ងាត់ៗជាអំណះអំណាងបច្ចេកទេស ដើម្បីជៀសវាងការដាក់ឈ្មោះវាថាជាអំណះអំណាងនយោបាយ (សូមមើល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព)។

រដ្ឋាភិបាលជាតិកំណត់ក្របខ័ណ្ឌ — ច្បាប់ប្រកួតប្រជែង បទប្បញ្ញត្តិវិស័យ ការរចនាពន្ធ និងឧបត្ថម្ភធន — ដែលមានបំណងកែតម្រូវការបរាជ័យនៃទីផ្សារទូទាំងសេដ្ឋកិច្ចទាំងមូល ដែលជាមូលហេតុដែលទីភ្នាក់ងារជាតិធ្វើជាម្ចាស់ផ្ទះនៃរដ្ឋបាលធំបំផុត និងចលនាថវិកាតាមបែប Niskanen មុតស្រួចបំផុត។ អាជ្ញាធរបទប្បញ្ញត្តិវិស័យសម្រាប់ថាមពល ទូរគមនាគមន៍ ឬហិរញ្ញវត្ថុ គឺជាបរិបទបុរាណនៃការចាប់យកអាជ្ញាធរបទប្បញ្ញត្តិ ព្រោះជំនាញដែលត្រូវការដើម្បីគ្រប់គ្រងឧស្សាហកម្មស្មុគស្មាញកម្ររកឃើញ ហើយប្រមូលផ្តុំក្នុងមនុស្សដូចគ្នាដែលផ្លាស់ប្តូររវាងអាជ្ញាធរបទប្បញ្ញត្តិ និងអ្នកត្រូវគ្រប់គ្រង។ ស្ថាប័នសវនកម្មជាតិមានភាគច្រើនដើម្បីផ្តល់ឱ្យនីតិបញ្ញត្តិ និងពលរដ្ឋនៅពីក្រោយវា នូវការត្រួតពិនិត្យឯករាជ្យលើភ្នាក់ងារដែលខ្លួនបានប្រគល់ឆន្ទានុសិទ្ធិ។

### វិស័យសង្គម និងវិស័យមិនស្វែងរកប្រាក់ចំណេញ

អង្គការមិនស្វែងរកប្រាក់ចំណេញ និងអង្គការក្រៅរដ្ឋាភិបាលច្រើនតែមានឡើងដើម្បីបំពេញគម្លាតទំនិញដែលទាំងទីផ្សារ និងរដ្ឋមិនផ្តល់មូលនិធិពេញលេញ — អំណះអំណាងការបរាជ័យនៃទីផ្សារដែលពង្រីកមួយជំហានទៀត ទៅដល់ទំនិញដែលត្រូវបានឱ្យតម្លៃជាសមូហភាព ប៉ុន្តែសូម្បីតែការជំរុញបោះឆ្នោតក៏មិនផ្តល់មូលនិធិដោយជឿជាក់។ ប៉ុន្តែវិស័យនេះមានបញ្ហាម្ចាស់ការ–ភ្នាក់ងារកំណែផ្ទាល់ខ្លួន៖ អ្នកផ្តល់មូលនិធិស្ថិតក្នុងទីតាំងស្រដៀងគ្នានឹងម្ចាស់ការជាពលរដ្ឋ ប៉ុន្តែជារឿយៗនៅឆ្ងាយពីការផ្តល់ជូនជាងអ្នកបោះឆ្នោតនៅឆ្ងាយពីរដ្ឋាភិបាល ហើយក្រុមប្រឹក្សាអាចប្រឈមនឹងអសមមាត្រនៃព័ត៌មានដូចគ្នាធៀបនឹងបុគ្គលិកជួរមុខ ដូចដែលសមាជិកនីតិបញ្ញត្តិប្រឈមធៀបនឹងមន្ត្រីរាជការ។ ការរសាត់បេសកកម្ម — អង្គការបម្រើអាទិភាពអ្នកផ្តល់មូលនិធិជាជាងតម្រូវការអ្នកទទួលផលបន្តិចម្តងៗ — គឺជាគូប្រៀបធៀបជិតបំផុតរបស់វិស័យនេះចំពោះការចាប់យក។

### កម្រិតពហុភាគី និងអន្តរជាតិ

ផលប៉ះពាល់ខាងក្រៅឆ្លងព្រំដែន — ការប្រែប្រួលអាកាសធាតុ ជំងឺរាតត្បាត ការឆ្លងរាលដាលហិរញ្ញវត្ថុ — គឺជាការបរាជ័យនៃទីផ្សារដែលគ្មានអធិបតេយ្យតែមួយអាចធ្វើឱ្យក្លាយជាផ្ទៃក្នុងតែម្នាក់ឯងបាន ដែលជាហេតុផលស្តង់ដារសម្រាប់ស្ថាប័នពហុភាគី។ ប៉ុន្តែគូប្រៀបធៀបការបរាជ័យនៃរដ្ឋាភិបាលពង្រីកទំហំដោយអាក្រក់៖ គ្មានអ្នកបោះឆ្នោតតែមួយដើរតួជាម្ចាស់ការចុងក្រោយលើលេខាធិការដ្ឋាន រដ្ឋសមាជិកអាចស្វែងរកចំណូលពីសិទ្ធិពិសេសរបស់ខ្លួនក្នុងច្បាប់របស់ស្ថាប័ន ហើយអសមមាត្រនៃព័ត៌មានដូចគ្នាដែលអនុញ្ញាតឱ្យរដ្ឋបាលជាតិយកឈ្នះនីតិបញ្ញត្តិរបស់ខ្លួន អាចអនុញ្ញាតឱ្យទីភ្នាក់ងារអន្តរជាតិយកឈ្នះក្រុមប្រឹក្សាតំណាងដែលវិលជុំ។ ដោយសារការចាកចេញ និងសំឡេងខ្សោយជាងនៅទំហំនេះ តម្លាភាព និងការវាយតម្លៃឯករាជ្យកាន់បន្ទុកគណនេយ្យភាពច្រើនជាងកម្រិតជាតិ។

## ទម្រង់នៃការបរាជ័យដែលជួបញឹកញាប់

**ការវិនិច្ឆ័យតាមមនោគមវិជ្ជា មិនមែនតាមការចាត់ថ្នាក់។** អ្នកដឹកនាំដែលបានសម្រេចជាមុនថាទីផ្សារតែងតែដំណើរការ ឬរដ្ឋាភិបាលតែងតែដំណើរការ ស្វែងរករឿងរ៉ាវការបរាជ័យណាដែលសមនឹងសេចក្តីសន្និដ្ឋានដែលកាន់រួច ជាជាងពិនិត្យថាតើការបរាជ័យណាមានពិតប្រាកដ។ ថ្លៃដើមគឺដំណោះស្រាយដែលមិនសំដៅលើអ្វីពិតប្រាកដ។

**ការចាប់យកដែលក្លែងជាជំនាញ។** ដោយសារការគ្រប់គ្រងឧស្សាហកម្មស្មុគស្មាញតម្រូវឱ្យមានមនុស្សដែលយល់ពីវា ហើយមនុស្សទាំងនោះជាធម្មតាមកពី ឬទៅធ្វើការក្នុងវា ការចាប់យកកម្រមើលទៅដូចអំពើពុករលួយ វាមើលទៅដូចការគោរពបច្ចេកទេសដែលស្តាប់ទៅសមហេតុផល។ ថ្លៃដើមកើនឡើងយឺតៗ នៅពេលដែលច្បាប់មកដល់ចំណុចជាប់លាប់ពេញចិត្តអ្នកកាន់តំណែងជាជាងអ្នកចូលថ្មី ឬអ្នកប្រើប្រាស់។

**គម្រោងសាកល្បងអចិន្ត្រៃយ៍។** កម្មវិធីដែលចាប់ផ្តើមដើម្បីកែតម្រូវការបរាជ័យនៃទីផ្សារ មិនដែលត្រូវបានពិនិត្យជាផ្លូវការធៀបនឹងលក្ខខណ្ឌកំណត់ពេលវេលាដែលបង្ហាញហេតុផលវាទេ ដូច្នេះវាបន្តដោយនិច្ចភាពយូរក្រោយពេលលក្ខខណ្ឌនោះបានផ្លាស់ប្តូរ។ ថ្លៃដើមគឺធនធានត្រូវបានចាក់សោក្នុងដំណោះស្រាយដែលបានឈប់ត្រូវនឹងបញ្ហាដោយស្ងាត់ៗ។

**ការទាញយកសិទ្ធិពិសេសក្លែងជាការផ្តល់សេវា។** អ្នកម៉ៅការ ឬអ្នកផ្គត់ផ្គង់ដែលត្រូវបានការពារពង្រីកវិសាលភាពនៃអ្វីដែលវាដាក់វិក្កយបត្របន្តិចម្តងៗ ដោយប្រើថ្លៃដើមប្តូរ និងអត្ថប្រយោជន៍ព័ត៌មាននៃទំនាក់ទំនងរយៈពេលវែង ជាជាងទង្វើដ៏គួរឱ្យភ្ញាក់ផ្អើលតែមួយ។ ថ្លៃដើមធ្លាក់លើអ្នកណាដែលបង់ចុងក្រោយ ជាធម្មតាដោយមើលមិនឃើញ តាមរយៈថ្លៃសេវាដែលកើនឡើងបន្តិចម្តងៗ ជាជាងការកើនឡើងតម្លៃជាចំណងជើងព័ត៌មាន។

**ស្ថាប័នមិនទទួលខុសត្រូវតាមចម្ងាយ។** ទីភ្នាក់ងារដែលបង្កើតឡើងដើម្បី «ឯករាជ្យពីនយោបាយប្រចាំថ្ងៃ» អាចរសាត់ទៅជាឯករាជ្យពីគណនេយ្យភាពទាំងស្រុង ដោយគ្មានម្ចាស់ការណាស្ថិតក្នុងទីតាំងសង្កេតថាតើអាទិភាពរបស់វានៅតែឆ្លុះបញ្ចាំងពីផលប្រយោជន៍សាធារណៈដែលវាត្រូវបានបង្កើតដើម្បីបម្រើឬអត់។ ថ្លៃដើមគឺគម្លាតម្ចាស់ការ–ភ្នាក់ងារដែលគ្មាននរណាទទួលខុសត្រូវក្នុងការបិទ។

ការចាត់ថ្នាក់ការបរាជ័យនៃទីផ្សារ ផ្អែកលើប្រពៃណីសេដ្ឋកិច្ចវិទ្យាសុខុមាលភាពនីអូក្លាស៊ិក ដោយផ្ទាល់បំផុតគឺស្នាដៃរបស់ Ronald Coase អំពីផលប៉ះពាល់ខាងក្រៅ និងថ្លៃដើមប្រតិបត្តិការ។ ប្រពៃណីការបរាជ័យនៃរដ្ឋាភិបាល និងជម្រើសសាធារណៈ ផ្អែកលើស្នាដៃរបស់ James M. Buchanan និង Gordon Tullock អំពីជម្រើសរដ្ឋធម្មនុញ្ញ និងសមូហភាព និងលើគំរូរបស់ William A. Niskanen អំពីរដ្ឋបាលបង្កើនថវិកាអតិបរមា។ ការសំយោគដែលចាត់ទុកទម្រង់បរាជ័យទាំងពីរជាធាតុចូលនៃការវិនិច្ឆ័យយុទ្ធសាស្ត្រ មិនមែនជាមធ្យោបាយសម្រេចបន្ទាប់ពីស្មើគ្នាផ្នែកបច្ចេកទេសទេ តាមស្នាដៃមូលដ្ឋានអំពីតម្លៃសាធារណៈរបស់ Mark H. Moore បន្ថែមដោយភស្តុតាងអភិបាលកិច្ចប្រៀបធៀបរបស់ OECD និងធនាគារពិភពលោក។
