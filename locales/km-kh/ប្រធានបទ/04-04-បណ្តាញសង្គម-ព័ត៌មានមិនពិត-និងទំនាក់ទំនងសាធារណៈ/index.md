# ប្រធានបទ 4.4 — បណ្តាញសង្គម ព័ត៌មានមិនពិត និងទំនាក់ទំនងសាធារណៈ

**នៅក្នុងបរិស្ថានប្រព័ន្ធផ្សព្វផ្សាយដែលបែកបាក់ និងមានវេទិកាជាអ្នកកណ្តាល ទំនាក់ទំនងរបស់វិស័យសាធារណៈ និងវិស័យសង្គម លែងជាមុខងារគាំទ្រដែលពន្យល់ការសម្រេចចិត្តក្រោយពេលធ្វើរួចទៀតហើយ — វាផ្ទាល់ជាទីតាំងនៃការបង្កើត និងការបាត់បង់ភាពស្របច្បាប់ ដែលត្រូវបានប្រកួតប្រជែងតាមពេលវេលាជាក់ស្តែងដោយតួអង្គដែលស្ថាប័នមិនអាចគ្រប់គ្រងបាន។**

## ហេតុអ្វីបានជាបញ្ហានេះសំខាន់សម្រាប់តម្លៃសាធារណៈ

អស់រយៈពេលភាគច្រើននៃសតវត្សទីម្ភៃ ទីភ្នាក់ងាររដ្ឋាភិបាល អ្នកផ្សព្វផ្សាយជាតិ ឬកាសែតដែលបានបង្កើតយូរមកហើយមួយចំនួនតូច បានបង្កើតជាបណ្តាញមានសិទ្ធិអំណាចរួមមួយ៖ កំណែតែមួយនៃព្រឹត្តិការណ៍បានទៅដល់ពលរដ្ឋភាគច្រើនតាមរយៈអ្នកយាមទ្វារចំនួនតិចតួច (សូមមើលអត្ថបទ Wikipedia អំពីការយាមទ្វារក្នុងទំនាក់ទំនង ដែលបានយោងខាងក្រោម)។ ការរៀបចំនោះបានបាត់ទៅហើយ។ ឥឡូវពលរដ្ឋជួបព័ត៌មានសាធារណៈរួមជាមួយពាក្យចចាមអារាម ការចំអក ការផ្សព្វផ្សាយពាណិជ្ជកម្ម និងការកុហកដោយចេតនា ក្នុងបញ្ជីព័ត៌មានបណ្តាញសង្គមតែមួយ ជារឿយៗមកពីប្រភពដែលគ្មានគណនេយ្យភាពផ្នែកវិចារណកថា និងគ្មានកាតព្វកិច្ចកែកំហុស។ អង្គភាពសាធារណៈដែលធ្លាប់ត្រូវការតែចេញសេចក្តីប្រកាសព័ត៌មាន ឥឡូវប្រកួតប្រជែងដើម្បីចំណាប់អារម្មណ៍ក្នុងបរិស្ថានដែលខ្លួនមិនមែនជាម្ចាស់ ហើយមិនអាចសម្របសម្រួលបាន។

រឿងនេះសំខាន់សម្រាប់តម្លៃសាធារណៈ ពីព្រោះត្រីកោណយុទ្ធសាស្ត្ររបស់ Mark Moore ចាត់ទុកភាពស្របច្បាប់ និងការគាំទ្រជាជើងមួយក្នុងចំណោមបីដែលអង្គការត្រូវការ រួមជាមួយសំណើតម្លៃ និងសមត្ថភាពប្រតិបត្តិការ (សូមមើល ប្រធានបទ 1.2 — ត្រីកោណយុទ្ធសាស្ត្រ៖ ភាពស្របច្បាប់ តម្លៃ និងសមត្ថភាព)។ ទំនាក់ទំនងគឺជារបៀបដែលបរិយាកាសផ្តល់សិទ្ធិដឹងអំពីអ្វីដែលស្ថាប័នកំពុងធ្វើ និងហេតុអ្វី ហើយវាក៏ជារបៀបដែលបរិយាកាសនោះរៀនសង្ស័យវាដែរ។ នៅពេលការអះអាងមិនពិតអំពីសុវត្ថិភាពវ៉ាក់សាំង សុចរិតភាពនៃការបោះឆ្នោត ឬភាពត្រឹមត្រូវនៃការព្រមានទឹកជំនន់ ផ្សព្វផ្សាយលឿនជាងការកែតម្រូវជាផ្លូវការ ការខូចខាតប្រតិបត្តិការអាចវាស់បាន៖ ការគ្របដណ្តប់វ៉ាក់សាំងទាបជាង ការគោរពតាមបទបញ្ជាជម្លៀសទាបជាង និងថ្លៃដើមខ្ពស់ជាងដើម្បីបង្កើតអង្គហេតុឡើងវិញ ដែលពីមុនត្រូវបានទទួលយកជាធម្មតា។ ការបរាជ័យនៃទំនាក់ទំនងមិនមែនជាភាពរអាក់រអួលផ្នែកកេរ្តិ៍ឈ្មោះដែលស្ថិតនៅក្បែរការងារពិតនៃការបង្កើតតម្លៃសាធារណៈទេ។ ក្នុងបរិស្ថានប្រព័ន្ធផ្សព្វផ្សាយដែលបែកបាក់ វាបានក្លាយជាផ្នែកមួយនៃការងារពិត។

ផលប៉ះពាល់មានលក្ខណៈមិនស្មើគ្នា។ ការអះអាងមិនពិតអាចត្រូវបានសរសេរ និងបោះពុម្ពផ្សាយក្នុងរយៈពេលប៉ុន្មានវិនាទី។ ការកែតម្រូវរបស់ស្ថាប័នជាធម្មតាត្រូវការការអនុញ្ញាត ការត្រួតពិនិត្យផ្លូវច្បាប់ និងការយល់ព្រមផ្នែកទំនាក់ទំនង មុនដល់ទស្សនិកជនដូចគ្នា — ហើយនៅពេលនោះ ការអះអាងមិនពិតច្រើនតែបានក្លាយជាកំណែព្រឹត្តិការណ៍ដែលស្គាល់ច្រើនជាង។ អ្នកដឹកនាំវិស័យសាធារណៈ និងវិស័យសង្គមដែលចាត់ទុកទំនាក់ទំនងជាការគិតក្រោយនៃការរចនាគោលនយោបាយ ក្នុងការអនុវត្តជាក់ស្តែងកំពុងប្រគល់ពាក្យរៀបរាប់ដំបូង និងជារឿយៗស្ថិតស្ថេរបំផុតអំពីការសម្រេចចិត្តរបស់ខ្លួន ទៅឱ្យអ្នកណាដែលធ្វើចលនាលឿនបំផុត។

## គំនិតស្នូល

**ការបែកបាក់ប្រព័ន្ធផ្សព្វផ្សាយ** ពិពណ៌នាអំពីការផ្លាស់ប្តូរពីបណ្តាញរួម និងមានទំនុកចិត្តខ្ពស់ចំនួនតិចតួច ទៅបណ្តាញប្រកួតប្រជែងដែលត្រូវបានជ្រើសរើសដោយក្បួនដោះស្រាយជាច្រើន ដែលនីមួយៗបង្ហាញពលរដ្ឋម្នាក់ៗនូវកំណែផ្សេងគ្នានៃជីវិតសាធារណៈ។ ការបែកបាក់ធ្វើឱ្យខ្សោយមុខងារយាមទ្វារដែលធ្លាប់ច្រោះការអះអាងសម្រាប់ភាពត្រឹមត្រូវមុនបោះពុម្ពផ្សាយ ហើយជំនួសវាដោយប្រព័ន្ធចំណាត់ថ្នាក់របស់វេទិកាដែលបង្កើនប្រសិទ្ធភាពសម្រាប់ការចូលរួម មិនមែនសច្ចភាពទេ។

**ព័ត៌មានមិនពិត** (misinformation) គឺជាព័ត៌មានមិនពិត ឬបំភាន់ដែលចែករំលែកដោយគ្មានចេតនាបោកប្រាស់ — ពលរដ្ឋដែលចែករំលែកស្ថិតិហួសសម័យដោយស្មោះត្រង់ កំពុងផ្សព្វផ្សាយព័ត៌មានមិនពិត។ **ព័ត៌មានមិនពិតដោយចេតនា** (disinformation) គឺជាព័ត៌មានមិនពិត ឬបំភាន់ដែលត្រូវបានបង្កើត ឬផ្សព្វផ្សាយដោយចេតនាដើម្បីបោកប្រាស់ សម្រាប់ផលនយោបាយ ហិរញ្ញវត្ថុ ឬយុទ្ធសាស្ត្រ។ ភាពខុសគ្នាស្ថិតទាំងស្រុងនៅចេតនា មិនមែននៅខ្លឹមសារខ្លួនឯងទេ ដែលជាហេតុផលដែលការអះអាងមិនពិតដូចគ្នាអាចចាប់ផ្តើមជាព័ត៌មានមិនពិតដោយចេតនាពីប្រភពមួយ ហើយរីករាលដាលបន្តជាព័ត៌មានមិនពិតពីប្រភពជាច្រើនផ្សេងទៀត។ ពាក្យដែលជិតស្និទ្ធ គឺ malinformation ពិពណ៌នាអំពីព័ត៌មានដែលត្រឹមត្រូវតាមអង្គហេតុ ប៉ុន្តែត្រូវបានប្រើក្រៅបរិបទដើម្បីបង្កគ្រោះថ្នាក់ (ឧទាហរណ៍ ឯកសារពិតដែលត្រូវបានលេចធ្លាយដើម្បីបំភាន់ ជាជាងផ្តល់ព័ត៌មាន)។ វាគួរដាក់ឈ្មោះ ពីព្រោះការត្រួតពិនិត្យអង្គហេតុជាផ្លូវការដែលបង្កើតតែដើម្បីចាប់ការកុហក នឹងខកខានវា។

**ជំងឺរាតត្បាតព័ត៌មាន** (infodemic) — ពាក្យដែលអង្គការសុខភាពពិភពលោក (WHO) ដែលជាទីភ្នាក់ងាររបស់អង្គការសហប្រជាជាតិសម្រាប់សុខភាពសាធារណៈអន្តរជាតិ បានអនុម័ត និងធ្វើជាផ្លូវការក្នុងអំឡុងជំងឺរាតត្បាត COVID-19 — គឺជាបរិមាណព័ត៌មានហួសប្រមាណ ទាំងត្រឹមត្រូវ និងមិនត្រឹមត្រូវ ដែលធ្វើឱ្យមនុស្សពិបាករកការណែនាំដែលគួរឱ្យទុកចិត្តនៅពេលពួកគេត្រូវការ។ ការគ្រប់គ្រងជំងឺរាតត្បាតព័ត៌មានចាត់ទុកបរិមាណ និងល្បឿននៃព័ត៌មានខ្លួនឯង មិនមែនតែតម្លៃសច្ចភាពរបស់វាប៉ុណ្ណោះទេ ជាអថេរសុខភាពសាធារណៈ និងសុវត្ថិភាពសាធារណៈដែលស្ថាប័នអាចតាមដាន និងឆ្លើយតប។

**បន្ទប់ឆ្លុះបញ្ចាំងសំឡេង** (echo chamber) គឺជាបរិស្ថានដែលមនុស្សម្នាក់ជួបព័ត៌មានដែលបញ្ជាក់ទស្សនៈដែលខ្លួនកាន់រួចហើយជាចម្បង ជាធម្មតាតាមរយៈមនុស្សដែលខ្លួនជ្រើសរើសតាមដាន។ **ពពុះតម្រង** (filter bubble) គឺជាឥទ្ធិពលពាក់ព័ន្ធ ប៉ុន្តែខុសគ្នា នៃក្បួនដោះស្រាយផ្តល់អនុសាសន៍របស់វេទិកាខ្លួនឯង ដែលកំណត់អ្វីដែលមនុស្សម្នាក់ឃើញ ដោយមិនមែនជាជម្រើសរបស់ពួកគេ។ ទាំងពីរកាត់បន្ថយការប៉ះពាល់របស់ពលរដ្ឋនឹងការកែតម្រូវ ប៉ុន្តែដោយហេតុផលខុសគ្នា — មួយជាសង្គម មួយជាបច្ចេកទេស — ហើយដូច្នេះត្រូវការការឆ្លើយតបខុសគ្នា។

**ការបង្ការព័ត៌មានមិនពិតជាមុន** (prebunking) ដែលផ្អែកលើ **ទ្រឹស្តីការចាក់ថ្នាំបង្ការ** (inoculation theory) គឺជាការអនុវត្តនៃការបង្ហាញមនុស្សនូវកំណែខ្សោយ និងជាមុននៃអំណះអំណាង ឬបច្ចេកទេសបំភាន់ មុនពេលពួកគេជួបរបស់ពិត ដើម្បីឱ្យពួកគេកសាងភាពស៊ាំនឹងវាជាមុន — ស្មើនឹងវ៉ាក់សាំងក្នុងទំនាក់ទំនង ផ្ទុយនឹង **ការកែតម្រូវព័ត៌មានមិនពិតក្រោយ** (debunking) ដែលកែការកុហកក្រោយពេលវារីករាលដាលរួច ហើយបានបង្កការខូចខាតខ្លះរួចហើយ។ **អក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយ** គឺជាសមត្ថភាពទូលំទូលាយជាង និងមានចក្ខុវិស័យរយៈពេលវែងជាងនៃការអាចចូលប្រើ វាយតម្លៃ និងវាយតម្លៃដោយរិះគន់នូវព័ត៌មាន ទោះបីជាប្រភពណាក៏ដោយ ដែលលំហាត់ការបង្ការជាមុន និងការកែតម្រូវជាក់លាក់តាមវេទិកាទាំងពីរពឹងលើពលរដ្ឋដែលមានវារួចហើយក្នុងកម្រិតខ្លះ។

**ទំនាក់ទំនងក្នុងវិបត្តិ** គឺជាការអនុវត្តនៃការទំនាក់ទំនងប្រកបដោយល្បឿន ភាពស្មោះត្រង់ និងភាពជាប់លាប់ ខណៈអាសន្នកំពុងលេចចេញ ហើយអង្គហេតុនៅមិនទាន់ពេញលេញ។ វាជាខ្សែស្រឡាយមួយនៃមុខវិជ្ជាទូលំទូលាយជាងនៃការគ្រប់គ្រងអាសន្ន និងភាពធន់ (សូមមើល ប្រធានបទ 3.12 — ការគ្រប់គ្រងវិបត្តិ អាសន្ន និងភាពធន់) ប៉ុន្តែត្រូវបានពិភាក្សានៅទីនេះជាពិសេសជាបញ្ហាទំនាក់ទំនង៖ អ្វីដែលត្រូវនិយាយ តាមបណ្តាញណា និងលឿនប៉ុនណា នៅពេលដែលស្ថាប័នមិនទាន់មានចម្លើយទាំងអស់។

**ភាពស្របច្បាប់** ក្នុងន័យវិទ្យាសាស្ត្រនយោបាយ គឺជាការទទួលយករបស់សាធារណជនចំពោះសិទ្ធិរបស់អាជ្ញាធរក្នុងការគ្រប់គ្រង ឬត្រូវបានជឿ ដោយឯករាជ្យពីថាតើគ្រប់សកម្មភាពរបស់វាមានប្រជាប្រិយភាពឬអត់។ ព័ត៌មានមិនពិត និងព័ត៌មានមិនពិតដោយចេតនាធ្វើឱ្យខូចភាពស្របច្បាប់ មិនមែនជាចម្បងដោយឈ្នះអំណះអំណាងតែមួយណាទេ ប៉ុន្តែដោយបង្រៀនពលរដ្ឋជាបន្តបន្ទាប់ថា ប្រភពផ្លូវការមិនអាចពឹងពាក់ថាមកមុន លឿន ឬពេញលេញបានទេ — ខ្វះខាតទំនុកចិត្តដូចគ្នាដែលបានពិនិត្យយ៉ាងវែងនៅកន្លែងផ្សេងក្នុងផ្នែកនេះ (សូមមើល ប្រធានបទ 4.2 — ទំនុកចិត្តលើរដ្ឋាភិបាល និងការចូលរួមរបស់ពលរដ្ឋ)។

## ការអនុវត្តល្អបំផុត

1. **សូមបោះពុម្ពផ្សាយតាមបណ្តាញផ្ទាល់ខ្លួនមុន ហើយចាត់ទុកវេទិកាជាការចែកចាយ មិនមែនជាកំណត់ត្រាទេ។** សូមរក្សាគេហទំព័រ បញ្ជីអ៊ីមែល ឬបណ្តាញផ្សេងទៀតដែលស្ថាប័នគ្រប់គ្រងទាំងស្រុង ជាកំណែមានសិទ្ធិអំណាចនៃសេចក្តីថ្លែងការណ៍ណាមួយ ហើយតភ្ជាប់គ្រប់ការផ្សព្វផ្សាយលើវេទិកាត្រឡប់ទៅវា ដើម្បីឱ្យពលរដ្ឋដែលចង់ផ្ទៀងផ្ទាត់ការអះអាងមានកន្លែងស្ថិរភាពទៅ។
2. **សូមបង្ការការកុហកដែលលោកអ្នកអាចទាយទុកជាមុនបាន។** មុនការបោះឆ្នោត យុទ្ធនាការចាក់វ៉ាក់សាំង ឬរដូវអាកាសធាតុធ្ងន់ធ្ងរ សូមកំណត់ការរៀបរាប់មិនពិតពីរឬបីដែលទំនងកើតឡើងម្តងទៀតបំផុត ហើយបោះពុម្ពផ្សាយការពន្យល់ខ្លីជាភាសាសាមញ្ញអំពីបច្ចេកទេសបំភាន់ជាមុន ជាជាងរង់ចាំប្រតិកម្មចំពោះការអះអាងនីមួយៗនៅពេលវាលេចឡើង។
3. **សូមរៀបចំបុគ្គលិកមុខងារស្តាប់តាមពេលវេលាជាក់ស្តែង ក្នុងអំឡុងព្រឹត្តិការណ៍ផ្ទាល់។** សូមចាត់តាំងអ្នកដែលមានឈ្មោះ ឬក្រុមតូចម្នាក់ដើម្បីតាមដានអ្វីដែលកំពុងរីករាលដាលពិតប្រាកដក្នុងអំឡុងវិបត្តិ ឬសេចក្តីប្រកាសសំខាន់ ដើម្បីឱ្យថ្នាក់ដឹកនាំទំនាក់ទំនងឆ្លើយតបចំពោះពាក្យចចាមអារាមដែលពលរដ្ឋកំពុងឃើញពិតប្រាកដ ជាជាងអ្វីដែលស្ថាប័នសន្មត់ថាកំពុងរីករាលដាល។
4. **សូមកំណត់ពិធីសារកែតម្រូវអចិន្ត្រៃយ៍ជាសាធារណៈ ហើយប្រើវាឱ្យជាប់លាប់។** សូមសម្រេចជាមុនថាអ្នកណាអាចអនុម័តការកែតម្រូវ និងលឿនប៉ុនណា បោះពុម្ពផ្សាយការកែតម្រូវក្នុងទម្រង់ និងលើបណ្តាញដូចគ្នានឹងការអះអាងដើម ហើយជៀសវាងការពង្រីកការកុហកគែមដោយកែវាយ៉ាងលេចធ្លោលើសលប់ ដែលមនុស្សច្រើនឃើញការកែតម្រូវជាងអ្នកដែលធ្លាប់ឃើញការអះអាងដើម។
5. **សូមកសាងរូបភាពប្រតិបត្តិការរួមតែមួយឆ្លងគ្រប់ទីភ្នាក់ងារដែលពាក់ព័ន្ធក្នុងព្រឹត្តិការណ៍មួយ។** នៅពេលអង្គភាពសាធារណៈច្រើនកំពុងទំនាក់ទំនងអំពីវិបត្តិតែមួយ សូមឯកភាពលើសំណុំអង្គហេតុតែមួយ និងរចនាសម្ព័ន្ធអ្នកនាំពាក្យតែមួយ មុនសេចក្តីថ្លែងការណ៍ដំបូងត្រូវបានចេញ ដើម្បីកុំឱ្យពលរដ្ឋទទួលព័ត៌មានផ្ទុយគ្នាពីទីភ្នាក់ងារដែលតាមពិតស្ថិតក្នុងក្រុមតែមួយ (សូមមើល ប្រធានបទ 3.12 — ការគ្រប់គ្រងវិបត្តិ អាសន្ន និងភាពធន់)។
6. **សូមវិនិយោគលើអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយជាហេដ្ឋារចនាសម្ព័ន្ធ មិនមែនជាយុទ្ធនាការទេ។** សូមផ្តល់មូលនិធិដល់ការងារអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយ — នៅសាលារៀន បណ្ណាល័យ និងអង្គការសហគមន៍ — ក្នុងរយៈពេលច្រើនឆ្នាំ ពីព្រោះសមត្ថភាពក្នុងការវាយតម្លៃប្រភពគឺជាសមត្ថភាពពលរដ្ឋស្ថិតស្ថេរដែលយុទ្ធនាការតាមរដូវតែមួយមិនអាចដំឡើងបាន។
7. **សូមជួបពលរដ្ឋលើវេទិកាដែលពួកគេប្រើពិតប្រាកដ មិនមែនវេទិកាដែលស្ថាប័នចូលចិត្តទេ។** សូមតាមដានកន្លែងដែលទស្សនិកជនដែលស្ថាប័នត្រូវការទៅដល់ជាងគេ — តាមអាយុ ភាសា ឬតំបន់ — ចំណាយពេលពិតប្រាកដ ហើយផ្តល់ធនធានទំនាក់ទំនងនៅទីនោះ ជាជាងលំនាំដើមទៅបណ្តាញណាដែលងាយបំផុតសម្រាប់ក្រុមទំនាក់ទំនងដំណើរការ។
8. **សូមញែកការកុហកចេញពីការកែតម្រូវឱ្យច្បាស់ល្មម ដើម្បីឱ្យការកែតម្រូវជាអ្វីដែលមនុស្សចងចាំ។** សូមបញ្ជាក់អង្គហេតុត្រឹមត្រូវមុនគេ និងលេចធ្លោបំផុត ដាក់ឈ្មោះការកុហកត្រឹមមួយភ្លែតយ៉ាងច្បាស់ថាមិនពិត ហើយបញ្ជាក់អង្គហេតុត្រឹមត្រូវម្តងទៀតនៅចុង ដើម្បីកុំឱ្យអ្នកអានដែលអានរហ័សត្រូវបានទុកជាមួយទេវកថាជាផ្នែកដែលចងចាំបំផុតនៃសារ។
9. **សូមធ្វើការតាមរយៈអ្នកនាំសារដែលមានទំនុកចិត្តក្នុងមូលដ្ឋាន និងសហគមន៍ មិនមែនតែអ្នកនាំពាក្យផ្លូវការប៉ុណ្ណោះទេ។** មេដឹកនាំសហគមន៍ វេជ្ជបណ្ឌិតមូលដ្ឋាន មេដឹកនាំសាសនា និងអង្គការសហគមន៍ ច្រើនតែមានទំនុកចិត្តប្រចាំថ្ងៃជាមួយទស្សនិកជនជាក់លាក់ច្រើនជាងទីភ្នាក់ងារជាតិ ហើយសារដែលបញ្ជូនរួមគ្នាតាមរយៈពួកគេជាធម្មតាធ្វើដំណើរបានឆ្ងាយជាង និងត្រូវបានជឿងាយជាង។
10. **សូមវាស់ការផ្លាស់ប្តូរជំនឿ និងអាកប្បកិរិយា មិនមែនតែវិសាលភាពប៉ុណ្ណោះទេ។** សូមតាមដានថាតើយុទ្ធនាការកែតម្រូវបានផ្លាស់ប្តូរអ្វីដែលទស្សនិកជនគំរូជឿ ឬរបៀបដែលពួកគេធ្វើសកម្មភាពពិតប្រាកដឬទេ — ការទទួលយកវ៉ាក់សាំង ការគោរពតាមការជម្លៀស ភាពត្រឹមត្រូវនៃការចុះឈ្មោះបោះឆ្នោត — ជាជាងតែចំនួនមនុស្សដែលបានឃើញ ឬចែករំលែកវា ពីព្រោះវិសាលភាពខ្ពស់ដោយគ្មានការផ្លាស់ប្តូរជំនឿគឺជាថវិកាដែលខ្ជះខ្ជាយ។
11. **សូមបោះពុម្ពផ្សាយទិន្នន័យមូលដ្ឋាន និងវិធីសាស្ត្រនៅពីក្រោយការអះអាងដែលមានជម្លោះ មិនមែនតែសេចក្តីសន្និដ្ឋានប៉ុណ្ណោះទេ។** នៅពេលស្ថិតិមួយមានជម្លោះ ការចេញផ្សាយទិន្នន័យ និងវិធីសាស្ត្រនៅពីក្រោយវា ក្នុងទម្រង់ដែលអ្នកសារព័ត៌មាន ឬពលរដ្ឋដែលចេះបច្ចេកទេសអាចពិនិត្យបាន ធ្វើការងារកេរ្តិ៍ឈ្មោះរយៈពេលវែងច្រើនជាងការនិយាយសេចក្តីសន្និដ្ឋានម្តងទៀតឱ្យខ្លាំងជាង។
12. **សូមហាត់សមផែនការទំនាក់ទំនងវិបត្តិជាមួយមនុស្សដែលនឹងប្រើវាពិតប្រាកដ មុនវិបត្តិពិតមកដល់។** ផែនការទំនាក់ទំនងដែលសាកល្បងតែលើក្រដាសតែងតែដួលរលំក្រោមសម្ពាធពេលវេលានៃព្រឹត្តិការណ៍ផ្ទាល់។ ផែនការដែលបានហាត់សមជាមួយការក្លែងធ្វើជាក់ស្តែង រួមទាំងការបញ្ចូលព័ត៌មានមិនពិត លាតត្រដាងគម្លាតនៅពេលដែលវានៅមានថ្លៃថោកក្នុងការជួសជុល។

## សំណួរសម្រាប់ពិភាក្សាជាមួយក្រុម

**តើយើងមានបណ្តាញដែលពលរដ្ឋនឹងទុកចិត្តមុនការអះអាងមិនពិតដែលរីករាលដាលលឿនឬទេ ហើយតើយើងដឹងទេថាពលរដ្ឋរបស់យើងប៉ុន្មាននាក់ប្រើវាពិតប្រាកដ?** អង្គភាពសាធារណៈជាច្រើនសន្មត់ថាគេហទំព័រ ឬគណនីបណ្តាញសង្គមផ្លូវការរបស់ខ្លួនគឺជាប្រភពលំនាំដើមដែលពលរដ្ឋនឹងពិនិត្យ ប៉ុន្តែទិន្នន័យប្រើប្រាស់ច្រើនតែបង្ហាញវិសាលភាពតូចជាងដែលបានសន្មត់ ប្រមូលផ្តុំក្នុងទស្សនិកជនដែលបានចូលរួមរួចហើយ។ ប្រសិនបើការអះអាងមិនពិតអំពីការសម្រេចចិត្តរបស់អង្គការលោកអ្នករីករាលដាលនៅថ្ងៃស្អែក សូមពិភាក្សាដោយស្មោះត្រង់ថាមនុស្សរងផលប៉ះពាល់ប៉ុន្មាននាក់នឹងឃើញការកែតម្រូវរបស់លោកអ្នកក្នុងម៉ោងដំបូង តាមបណ្តាញណា និងជាភាសាអ្វី។ នៅកន្លែងដែលចម្លើយគឺ "មិនច្រើនទេ" ក្រុមគួរចាត់ទុកការកសាងបណ្តាញជាហេដ្ឋារចនាសម្ព័ន្ធមិនទាន់បញ្ចប់ ជាជាងសន្មត់ថាវាមានរួចហើយ។ សំណួរនេះត្រូវឆ្លើយបានល្អបំផុតដោយវិភាគជាក់ស្តែង មិនមែនការសន្មត់ទេ។

**តើកម្មវិធីបង្ការជាមុនរបស់យើងសម្រាប់ព័ត៌មានមិនពិតដែលយើងអាចទាយទុកជាមុនគឺអ្វី ហើយអ្នកណាជាម្ចាស់?** ស្ថាប័នភាគច្រើនអាចដាក់ឈ្មោះការរៀបរាប់មិនពិតពីរឬបីដែលទំនងកើតឡើងម្តងទៀតបំផុតជុំវិញការបោះឆ្នោត យុទ្ធនាការសុខភាព ឬការសម្រេចចិត្តហេដ្ឋារចនាសម្ព័ន្ធបន្ទាប់ ប៉ុន្តែតិចណាស់ដែលមានកិច្ចខិតខំបង្ការជាមុនអចិន្ត្រៃយ៍ និងមានធនធាន ជាជាងយុទ្ធនាការតែមួយលើក។ សូមពិភាក្សាថាការកុហកដែលអាចទាយទុកបានអ្វីខ្លះនឹងមកក្នុង 12 ខែខាងមុខ ហើយថាតើមាននរណាមានទំនួលខុសត្រូវ ថវិកា និងសិទ្ធិអំណាចយល់ព្រមច្បាស់លាស់ដើម្បីបង្ការវាឥឡូវ មុនវារីករាលដាលឬអត់។ សូមពិចារណាផងដែរថាតើអង្គការរបស់លោកអ្នកមានជំនាញផ្ទៃក្នុងដើម្បីរចនាសារបង្ការជាមុនបានល្អឬអត់ ពីព្រោះសារដែលរចនាមិនល្អអាចផ្ទុយ ដោយនិយាយការកុហកម្តងទៀតច្រើនជាងការកែតម្រូវ។ ប្រសិនបើថ្ងៃនេះគ្មាននរណាជាម្ចាស់ សូមសម្រេចក្នុងបន្ទប់ថាអ្នកណានឹងជាម្ចាស់។

**តើយើងអាចចេញការកែតម្រូវលឿនប៉ុនណា ហើយអ្នកណាមានសិទ្ធិអំណាចអនុម័តវាក្រៅម៉ោងការងារ?** ពិធីសារកែតម្រូវជាច្រើនមានលើក្រដាស ប៉ុន្តែដួលរលំនៅពេលដំបូងដែលការអះអាងមិនពិតរីករាលដាលនៅចុងសប្តាហ៍ ឬពេលយប់ ពីព្រោះការអនុម័តស្ថិតនៅនឹងអ្នកដែលមិនអាចទាក់ទងបាន។ សូមឆ្លងកាត់ជាក្រុមអំពីអ្វីដែលកើតឡើងយ៉ាងជាក់លាក់ ចាប់ពីពេលដែលបុគ្គលិកម្នាក់ឃើញការកុហកដែលរីករាលដាលដំបូង រហូតដល់ពេលការកែតម្រូវត្រូវបានបោះពុម្ពផ្សាយ ដោយដាក់ឈ្មោះគ្រប់ជំហានអនុម័ត និងមនុស្សគ្រប់រូបដែលពាក់ព័ន្ធ។ សូមកំណត់ជំហានយឺតបំផុតតែមួយ ហើយពិភាក្សាថាតើវាអាចត្រូវបានផ្ទេរអំណាចជាមុនសម្រាប់ករណីប្រញាប់ប្រញាល់ពេលវេលាឬអត់។ ពិធីសារដែលគ្មាននរណាបានសាកល្បងក្រោមសម្ពាធពេលវេលា មិនទាន់ជាសមត្ថភាពពិតប្រាកដនៅឡើយទេ។

**តើយើងទៅដល់ពលរដ្ឋដែលមិនតាមដានយើងលើវេទិកាណាមួយ ហើយមិនអានព័ត៌មានមូលដ្ឋានដោយរបៀបណា?** ស្ថាប័នគ្រប់រូបមានទស្សនិកជនដែលវាពិបាកទៅដល់ដោយផ្ទាល់ជារចនាសម្ព័ន្ធ — ជារឿយៗជាមនុស្សដែលមានពេលតិចបំផុត ទំនុកចិត្តលើស្ថាប័នតិចបំផុត ឬភាសាមាតុភាសាខុសគ្នា — ហើយព័ត៌មានមិនពិតទៅដល់ពួកគេងាយដូចទៅដល់អ្នកផ្សេងដែរ។ សូមពិភាក្សាថាទស្សនិកជននេះសម្រាប់អង្គការរបស់លោកអ្នកជាអ្នកណាយ៉ាងជាក់លាក់ ហើយបណ្តាញមិនមែនឌីជីថល ឬសហគមន៍ជាអ្នកកណ្តាលអ្វី (អង្គការសហគមន៍ ឱសថស្ថានមូលដ្ឋាន មេដឹកនាំសាសនា កន្លែងធ្វើការ) អាចទៅដល់ពួកគេនៅកន្លែងដែលបណ្តាញផ្ទាល់ខ្លួនរបស់លោកអ្នកមិនអាច។ សូមពិចារណាថាតើថវិកាបច្ចុប្បន្នឆ្លុះបញ្ចាំងពីការលំបាកក្នុងការទៅដល់ទស្សនិកជននេះ ឬគ្រាន់តែភាពងាយស្រួលក្នុងការទៅដល់ទស្សនិកជនដែលលោកអ្នកមានរួចហើយ។

**តើយើងពិតជាដឹងអ្វីអំពីមូលហេតុដែលពលរដ្ឋមិនទុកចិត្តយើង ហើយតើយុទ្ធសាស្ត្រទំនាក់ទំនងរបស់យើងដោះស្រាយមូលហេតុនោះ ឬគ្រាន់តែរោគសញ្ញារបស់វា?** ការខ្វះខាតទំនុកចិត្តមានមូលហេតុជាក់លាក់ដែលអាចតាមដានបាន — ការសន្យាដែលបានបំពានពីមុន ការឆ្លើយតបវិបត្តិយឺត គោលនយោបាយដែលបរាជ័យយ៉ាងច្បាស់ចំពោះសហគមន៍ — ហើយទំនាក់ទំនងដែលបានកសាងដោយមិនយល់មូលហេតុច្រើនតែដោះស្រាយបញ្ហាខុស (សូមមើល ប្រធានបទ 4.2 — ទំនុកចិត្តលើរដ្ឋាភិបាល និងការចូលរួមរបស់ពលរដ្ឋ)។ សូមពិភាក្សាថាតើលោកអ្នកមានភស្តុតាងអ្វីអំពីប្រភពជាក់លាក់នៃការមិនទុកចិត្តក្នុងតំបន់សេវារបស់លោកអ្នក ផ្ទុយនឹងការសន្មត់ទូទៅថា "មនុស្សមិនទុកចិត្តរដ្ឋាភិបាល"។ សូមពិចារណាថាតើផែនការទំនាក់ទំនងរបស់លោកអ្នកត្រូវបានរចនាដើម្បីកសាងទំនុកចិត្តឡើងវិញក្នុងផ្នែកជាក់លាក់ដែលវាបានបាត់ ឬជាលំហាត់សារទូទៅដែលអនុវត្តស្មើគ្នាគ្រប់ទីកន្លែង។ ទំនាក់ទំនងមិនអាចជួសជុលបញ្ហាទំនុកចិត្តដែលវាមិនបានធ្វើរោគវិនិច្ឆ័យត្រឹមត្រូវបានទេ។

**តើយើងដឹងដោយរបៀបណាថាតើការវិនិយោគអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយរបស់យើងកំពុងដំណើរការពិតប្រាកដ?** មូលនិធិអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយងាយប្រកាស ប៉ុន្តែពិបាកវាយតម្លៃ ហើយកម្មវិធីជាច្រើនដំណើរការអស់ច្រើនឆ្នាំដោយគ្មានភស្តុតាងថាតើសមត្ថភាពរបស់អ្នកចូលរួមក្នុងការវាយតម្លៃព័ត៌មានបានផ្លាស់ប្តូរពិតប្រាកដឬអត់។ សូមពិភាក្សាថាតើវិធីសាស្ត្រវាយតម្លៃអ្វី ប្រសិនបើមាន មានសម្រាប់ការងារអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយរបស់លោកអ្នកបច្ចុប្បន្ន ហើយថាតើវាវាស់ការផ្លាស់ប្តូរជំនឿ និងអាកប្បកិរិយា ឬតែវត្តមាន និងការពេញចិត្ត។ សូមពិចារណាថាតើកម្មវិធីគួរត្រូវបានរចនាឡើងវិញជុំវិញការអះអាងដែលអាចសាកល្បងបាន ដូចជាភាពត្រឹមត្រូវប្រសើរឡើងក្នុងការកំណត់ខ្លឹមសារបំភាន់ ជាជាងជុំវិញបំណងប្រាថ្នាទូទៅ។ កម្មវិធីអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយដែលមិនបានវាយតម្លៃគឺជាជំនឿ មិនទាន់ជាអង្គហេតុទេ។

## នៅក្នុងការអនុវត្ត៖ ឧទាហរណ៍តម្លៃសាធារណៈ

នាយកដ្ឋានសុខភាពសាធារណៈនៃរដ្ឋាភិបាលស្រុកឆ្នេរទំហំមធ្យមមួយ បានទទួលលទ្ធផលមន្ទីរពិសោធន៍នៅរសៀលថ្ងៃសុក្រយឺត ដែលបង្ហាញពីការកខ្វក់បាក់តេរីកើនឡើងនៅផ្នែកមួយនៃប្រព័ន្ធផ្គត់ផ្គង់ទឹកក្រុង ដែលបម្រើអ្នករស់នៅប្រហែលសែសិបពាន់នាក់។ មុននាយកដ្ឋានបានបញ្ចប់ការរៀបចំសេចក្តីជូនដំណឹងដាំទឹកឱ្យពុះបង្ការ ការផ្សព្វផ្សាយរបស់អ្នករស់នៅម្នាក់ ដែលអះអាងថាទឹក "មានជាតិពុល" ហើយមន្ត្រី "បិទបាំងវា" ត្រូវបានចែករំលែករួចហើយរាប់ពាន់ដងក្នុងក្រុមបណ្តាញសង្គមតំបន់ ដោយលាយកង្វល់មូលដ្ឋានដែលត្រឹមត្រូវជាមួយការរៀបរាប់មិនត្រឹមត្រូវ និងគួរឱ្យព្រួយបារម្ភជាងអំពីមូលហេតុ។

ប្រធានទំនាក់ទំនងរបស់នាយកដ្ឋាន ដែលបានដំណើរការលំហាត់តុជាមួយសេណារីយ៉ូនេះយ៉ាងច្បាស់កាលពីដប់ប្រាំបីខែមុន បានបើកដំណើរការពិធីសារដែលបានឯកភាពជាមុន ជាជាងចាប់ផ្តើមពីបាតដៃទទេ។ អ្នកនាំពាក្យតែមួយត្រូវបានតែងតាំងក្នុងរយៈពេលសាមសិបនាទី ដោយជៀសវាងសេចក្តីថ្លែងការណ៍ផ្ទុយគ្នាដែលបានបំផ្លាញទំនុកចិត្តសាធារណៈក្នុងអំឡុងឧប្បត្តិហេតុមិនទាក់ទងកាលពីឆ្នាំមុន។ នាយកដ្ឋានបានបោះពុម្ពផ្សាយសេចក្តីជូនដំណឹងដាំទឹកឱ្យពុះក្នុងពេលតែមួយលើគេហទំព័រផ្ទាល់ខ្លួន តាមរយៈសារអត្ថបទជូនដំណឹងដល់អ្នករស់នៅដែលបានចុះឈ្មោះ និងជាការផ្សព្វផ្សាយលើបណ្តាញសង្គមជាភាសាសាមញ្ញដែលបានដាក់ឈ្មោះការអះអាងដែលរីករាលដាលដោយផ្ទាល់ កែភាពមិនត្រឹមត្រូវជាក់លាក់ (ការកខ្វក់គឺជាការខ្សោយនៃដំណើរការព្យាបាល មិនមែនដោយចេតនាទេ) និងបញ្ជាក់ច្បាស់អំពីអ្វីដែលនៅមិនទាន់ដឹង រួមទាំងរយៈពេលរំពឹងទុកនៃសេចក្តីជូនដំណឹង។

ដោយទទួលស្គាល់ថាអ្នកតាមដានរបស់ខ្លួនលើបណ្តាញសង្គមទៅដល់តែប្រភាគនៃអ្នករស់នៅដែលរងផលប៉ះពាល់ នាយកដ្ឋានក៏បានទាក់ទងអង្គការសហគមន៍បីនិងឱសថស្ថានមូលដ្ឋានពីរ ដែលវាបានកសាងទំនាក់ទំនងក្នុងអំឡុងលំហាត់បង្ការជាមុនកាលពីឆ្នាំមុន ដោយសុំឱ្យពួកគេចែករំលែកអង្គហេតុស្នូលដូចគ្នាជាពាក្យរបស់ពួកគេផ្ទាល់។ ក្នុងរយៈពេលបួនម៉ោង ទិន្នន័យស្រុកបានបង្ហាញថាអ្នករស់នៅច្រើនជាងបានឃើញសេចក្តីជូនដំណឹងដែលបានកែតាមរយៈបណ្តាញសហគមន៍ ជាងតាមរយៈគណនីផ្ទាល់ខ្លួនរបស់ស្រុក។ សេចក្តីជូនដំណឹងបន្តខ្លីមួយថ្ងៃក្រោយ នៅពេលដែលកំហុសព្យាបាលត្រូវបានជួសជុល បានប្រើបណ្តាញដូចគ្នាម្តងទៀត ហើយបានដាក់ឈ្មោះម្តងទៀតអំពីអ្វីដែលនៅជាវិធានការបង្ការ មិនមែនបានបញ្ជាក់។

ការពិនិត្យផ្ទៃក្នុងក្រោយមកបានផ្តល់ឥណទានដល់ជម្រើសជាក់លាក់បីដែលកំណត់ការខូចខាតកេរ្តិ៍ឈ្មោះ៖ ពិធីសារអ្នកនាំពាក្យតែមួយដែលបានឯកភាពជាមុន ការសម្រេចចិត្តដាក់ឈ្មោះ និងកែការអះអាងមិនពិត "ការបំពុលដោយចេតនា" ដោយផ្ទាល់ ជាជាងមិនអើពើវា និងការពឹងផ្អែកលើអ្នកនាំសារសហគមន៍ដែលនាយកដ្ឋានមិនចាំបាច់កសាងពីបាតដៃទទេក្នុងអំឡុងវិបត្តិខ្លួនឯង។ ការពិនិត្យក៏បានកត់សម្គាល់គម្លាតពិតប្រាកដមួយ៖ នាយកដ្ឋានគ្មានទំនាក់ទំនងស្មើគ្នាក្នុងភាសាដែលនិយាយច្រើនបំផុតទីពីរនៃស្រុក ហើយសេចក្តីជូនដំណឹងដែលបានកែបានទៅដល់សហគមន៍នោះយឺតជាងអ្នកផ្សេងប្រហែលមួយថ្ងៃ។ សេណារីយ៉ូនេះគឺជាឧទាហរណ៍ ហើយមិនពិពណ៌នាអំពីប្រព័ន្ធទឹក ទីភ្នាក់ងារ ឬឧប្បត្តិហេតុរបស់យុត្តាធិការពិតប្រាកដណាមួយទេ។

## ទស្សនៈវិស័យទាំងបួន

### រដ្ឋាភិបាលមូលដ្ឋាន

ទំនាក់ទំនងរបស់រដ្ឋាភិបាលមូលដ្ឋានដំណើរការនៅមាត្រដ្ឋានដែលទំនុកចិត្តមានលក្ខណៈផ្ទាល់ខ្លួនបំផុត និងផុយស្រួយបំផុត៖ មតិរបស់អ្នករស់នៅចំពោះស្ថាប័នទាំងមូលច្រើនតែត្រូវបានកំណត់ដោយអន្តរកម្មតែមួយជាមួយសេវាមូលដ្ឋានតែមួយ។ អង្គភាពមូលដ្ឋានជាធម្មតាមានក្រុមទំនាក់ទំនង និងថវិកាតូចជាងអង្គភាពជាតិឆ្ងាយ ដែលធ្វើឱ្យទំនាក់ទំនងជាមួយអង្គការសហគមន៍ ប្រព័ន្ធផ្សព្វផ្សាយមូលដ្ឋាន និងមេដឹកនាំសហគមន៍មិនផ្លូវការមានតម្លៃមិនសមាមាត្រ ដោយសារវាពង្រីកវិសាលភាពដែលស្ថាប័នមិនអាចទិញដោយផ្ទាល់។ ហានិភ័យដែលត្រូវគ្នាគឺថា ក្រុមតូចដែលគ្មានសមត្ថភាពតាមដានជាក់លាក់ច្រើនតែជាអ្នកចុងក្រោយដែលដឹងថាការអះអាងមិនពិតកំពុងរីករាលដាលក្នុងមូលដ្ឋាន ពិតជានៅកន្លែងដែលវាអាចធ្វើឱ្យខូចភ្លាមៗបំផុតដល់ការគោរពតាមសេចក្តីជូនដំណឹងសុវត្ថិភាព ឬការផ្លាស់ប្តូរសេវា។

### រដ្ឋាភិបាលជាតិ

រដ្ឋាភិបាលជាតិតាមប្រវត្តិសាស្ត្របានពឹងលើអ្នកផ្សព្វផ្សាយជាតិ និងកាសែតចំនួនតិចតួចជាបណ្តាញរួមជាក់ស្តែង ហើយការបាត់បង់ការរៀបចំនោះគឺជាការបាត់បង់ដែលមើលឃើញច្បាស់បំផុតក្នុងវិស័យណាមួយ ជាពិសេសក្នុងអំឡុងការបោះឆ្នោតដែលមានជម្លោះ និងអាសន្នសុខភាពសាធារណៈ។ អង្គភាពជាតិក៏ជាគោលដៅញឹកញាប់បំផុតនៃយុទ្ធនាការព័ត៌មានមិនពិតដោយចេតនាដែលសម្របសម្រួល និងឆ្លងព្រំដែន ដែលអង្គការសហប្រតិបត្តិការ និងអភិវឌ្ឍន៍សេដ្ឋកិច្ច (OECD) — ស្ថាប័នអន្តររដ្ឋាភិបាលដែលប្រទេសសមាជិកសាមសិបប្រាំបីសម្របសម្រួលគោលនយោបាយសេដ្ឋកិច្ច និងអភិបាលកិច្ចសាធារណៈ — បានចងក្រងឯកសារថាជាការគំរាមកំហែងកាន់តែខ្លាំងចំពោះប្រព័ន្ធអេកូឡូស៊ីព័ត៌មានប្រជាធិបតេយ្យ ដែលខុសពីព័ត៌មានមិនពិតប្រចាំថ្ងៃដែលស្ថាប័នណាក៏ត្រូវគ្រប់គ្រងដែរ។ រដ្ឋាភិបាលជាតិជាទូទៅមានធនធានដើម្បីកសាងកម្មវិធីបង្ការជាមុន និងអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយប្រកបដោយនិរន្តរភាព ប៉ុន្តែត្រូវតែការពារហានិភ័យខុសគ្នា៖ កិច្ចខិតខំកែតម្រូវជាផ្លូវការដែលមើលទៅដូចជាសារនយោបាយ បាត់បង់អព្យាក្រឹតភាពដែលធ្វើឱ្យវាគួរឱ្យទុកចិត្តតាំងពីដំបូង។

### វិស័យសង្គម និងវិស័យមិនស្វែងរកប្រាក់ចំណេញ

អង្គការមិនស្វែងរកប្រាក់ចំណេញ និងអង្គការមិនមែនរដ្ឋាភិបាល (NGO) ច្រើនតែមានកម្រិតទំនុកចិត្តសាធារណៈដែលលើសរដ្ឋាភិបាល ជាពិសេសលើបញ្ហាដូចជាសុខភាពសាធារណៈ ឬការឆ្លើយតបមនុស្សធម៌ ដែលធ្វើឱ្យពួកគេជាដៃគូទំនាក់ទំនងដ៏មានតម្លៃ ប៉ុន្តែក៏ជាគោលដៅសម្រាប់ព័ត៌មានមិនពិតដោយចេតនាដែលរចនាដើម្បីបំផ្លាញកេរ្តិ៍ឈ្មោះពួកគេជាពិសេស។ អង្គការវិស័យសង្គមជាធម្មតាទំនាក់ទំនងលើថវិកាតូចជាងទីភ្នាក់ងាររដ្ឋាភិបាលឆ្ងាយ ដែលរុញពួកគេទៅរកការពឹងផ្អែកលើវេទិកាតែមួយសម្រាប់វិសាលភាព — ភាពងាយរងគ្រោះផ្នែករចនាសម្ព័ន្ធ ប្រសិនបើវេទិកានោះផ្លាស់ប្តូរគោលនយោបាយ ឬក្បួនដោះស្រាយរបស់វា។ ដោយសារមូលនិធិវិស័យសង្គមភាគច្រើនជាក់លាក់តាមបេសកកម្ម ឬយុទ្ធនាការ អង្គការមិនស្វែងរកប្រាក់ចំណេញក៏អាចពិបាកបង្ហាញហេតុផលសម្រាប់ការវិនិយោគច្រើនឆ្នាំ ដែលមិនគួរឱ្យចាប់អារម្មណ៍ ដែលការងារអក្ខរកម្មប្រព័ន្ធផ្សព្វផ្សាយ និងការបង្ការជាមុនតម្រូវ ដោយសារអ្នកផ្តល់មូលនិធិច្រើនតែចូលចិត្តគាំទ្រកម្មវិធីដែលមើលឃើញជាងការកសាងភាពធន់ដែលមើលមិនឃើញ។

### កម្រិតពហុភាគី និងអន្តរជាតិ

អង្គការពហុភាគីប្រឈមនឹងជំងឺរាតត្បាតព័ត៌មានដែលមិនគោរពព្រំដែន ឬភាសា ដោយសម្របសម្រួលព័ត៌មានកែតម្រូវឆ្លងយុត្តាធិការរាប់សិប ដែលមានរបបច្បាប់ខុសគ្នាសម្រាប់ការនិយាយ និងកម្រិតទំនុកចិត្តស្ថាប័នខុសគ្នា។ ការងារគ្រប់គ្រងជំងឺរាតត្បាតព័ត៌មានរបស់ WHO ដែលបានអភិវឌ្ឍក្នុងអំឡុងជំងឺរាតត្បាត COVID-19 គឺជាឧទាហរណ៍ច្បាស់បំផុតនៃស្ថាប័នពហុភាគីដែលចាត់ទុកព័ត៌មានខ្លួនឯង មិនមែនតែការចម្លងជំងឺប៉ុណ្ណោះទេ ជាអថេរដែលវាត្រូវតាមដាន និងគ្រប់គ្រងយ៉ាងសកម្ម។ អង្គការអន្តរជាតិក៏ត្រូវរុករកភាពតានតឹងពិតប្រាកដដែលអង្គភាពជាតិ និងមូលដ្ឋានកម្រជួបក្នុងមាត្រដ្ឋានដូចគ្នា៖ ការចែករំលែកការស្រាវជ្រាវបង្ការជាមុន និងវិធីសាស្ត្រកែតម្រូវជាសកល ខណៈគោរពថាការសម្រេចចិត្តអំពីការនិយាយអ្វីដែលត្រូវគ្រប់គ្រងនៅតែជាបញ្ហាអធិបតេយ្យភាពជាតិយ៉ាងសមស្រប ជាជាងអាណត្តិពហុភាគី។

## ទម្រង់នៃការបរាជ័យដែលជួបញឹកញាប់

**អន្ទាក់នៃភាពស្ងៀមស្ងាត់។** ស្ថាប័នដែលរង់ចាំរហូតមានអង្គហេតុពេញលេញ និងបានផ្ទៀងផ្ទាត់មុននិយាយអ្វីមួយ ទុកឱ្យមានចន្លោះព័ត៌មានដែលពាក្យចចាមអារាម និងការអះអាងមិនពិតបំពេញមុន។ នៅពេលកំណែផ្លូវការមកដល់ វាកំពុងប្រកួតប្រជែងជាមួយការរៀបរាប់ដែលបានបង្កើតរួច ជាជាងកំណត់វា។ ថ្លៃដើមមិនមែនតែការកែតម្រូវយឺតជាងប៉ុណ្ណោះទេ ប៉ុន្តែជាការយល់ឃើញយូរអង្វែងថាស្ថាប័នយឺត ឬគេចវេស ទោះបីជាវានិយាយនៅទីបំផុតក៏ដោយ។

**ម៉ាស៊ីនរត់នៃការកែតម្រូវក្រោយ។** អង្គការដែលគ្មានសមត្ថភាពបង្ការជាមុនចំណាយកិច្ចខិតខំទំនាក់ទំនងទាំងអស់ទៅលើប្រតិកម្មចំពោះការអះអាងមិនពិតថ្មីនីមួយៗដោយឡែក មិនដែលនាំមុនគំរូ និងធ្វើឱ្យបុគ្គលិកទំនាក់ទំនងអស់កម្លាំង ខណៈមិនដែលកាត់បន្ថយបរិមាណការកុហកដែលត្រូវឆ្លើយ។ ដោយសារការកែតម្រូវការអះអាងអាចបង្កើនភាពស្គាល់ និងវិសាលភាពរបស់វាដោយអចេតនា ឥរិយាបថប្រតិកម្មសុទ្ធអាចទុកស្ថាប័នក្នុងស្ថានភាពអាក្រក់ជាងមិនធ្វើអ្វីទាល់តែសោះលើការអះអាងដែលមានផលវិបាកតិច។

**ការពឹងផ្អែកលើបណ្តាញតែមួយ។** ស្ថាប័នដែលកសាងសមត្ថភាពទំនាក់ទំនងដោយផ្ទាល់ទៅពលរដ្ឋទាំងមូលជុំវិញវេទិកាតែមួយ ប្រឈមនឹងហានិភ័យនៅពេលវេទិកានោះផ្លាស់ប្តូរក្បួនដោះស្រាយ គោលនយោបាយសម្របសម្រួល ឬលក្ខខណ្ឌចូលប្រើ ដែលស្ថាប័នគ្មានការគ្រប់គ្រងលើណាមួយឡើយ។ នេះបានបង្ខំអង្គភាពសាធារណៈម្តងហើយម្តងទៀតឱ្យកសាងទស្សនិកជនឡើងវិញពីស្ទើរតែសូន្យដោយគ្មានការព្រមាន ពិតជានៅពេលដែលពួកគេត្រូវការបណ្តាញអាចទុកចិត្តបានបំផុត។

**ការចាក់ទិន្នន័យ។** ការបោះពុម្ពផ្សាយទិន្នន័យឆៅ ឬការបដិសេធបច្ចេកទេសវែងឆ្លើយតបចំពោះការកុហកដែលរីករាលដាល បំពេញសភាវគតិរបស់ស្ថាប័នក្នុងការប្រុងប្រយ័ត្នលម្អិត ប៉ុន្តែកម្រទៅដល់ ឬបញ្ចុះបញ្ចូលទស្សនិកជនដែលបានឃើញការអះអាងដើម ពីព្រោះស្ទើរតែគ្មាននរណាអានហួសចំណងជើងទេ។ ថ្លៃដើមគឺការកែតម្រូវដែលត្រឹមត្រូវផ្នែកបច្ចេកទេស និងមើលមិនឃើញក្នុងការអនុវត្ត។

**ការកែតម្រូវការអះអាងគែមលើសលប់។** ការឆ្លើយតបខ្លាំង និងម្តងហើយម្តងទៀតចំពោះការកុហកដែលមានទស្សនិកជនតូចមួយ ដែលស្ថិតនៅគែម អាចពង្រីកវាទៅទស្សនិកជនធំជាងអ្នកដែលបានឃើញការអះអាងដើមច្រើនដង — ហានិភ័យដែលបានចងក្រងឯកសារយ៉ាងល្អក្នុងការស្រាវជ្រាវព័ត៌មានមិនពិត។ ថ្លៃដើមគឺការរីករាលដាលដែលបង្កដោយខ្លួនឯងនៃការអះអាងដែលស្ថាប័នកំពុងព្យាយាមទប់ស្កាត់ពិតប្រាកដ។

- [ ] សូមឯកភាពរចនាសម្ព័ន្ធអ្នកនាំពាក្យតែមួយរួមជាមួយគ្រប់ទីភ្នាក់ងារផ្សេងទៀតដែលទំនងពាក់ព័ន្ធក្នុងព្រឹត្តិការណ៍តែមួយ។

- [ ] សូមកំណត់មូលហេតុជាក់លាក់ និងមានភស្តុតាងនៃការមិនទុកចិត្តក្នុងមូលដ្ឋាន ជាជាងសន្មត់ (សូមមើល ប្រធានបទ 4.2 — ទំនុកចិត្តលើរដ្ឋាភិបាល និងការចូលរួមរបស់ពលរដ្ឋ)។

## ប្រភពសំខាន់ៗ

កម្មវិធីគ្រប់គ្រងជំងឺរាតត្បាតព័ត៌មានរបស់អង្គការសុខភាពពិភពលោក ដែលបានអភិវឌ្ឍក្នុងអំឡុងជំងឺរាតត្បាត COVID-19 គឺជាឧទាហរណ៍ដែលបានដោះស្រាយពេញលេញបំផុតនៃស្ថាប័នសាធារណៈដែលចាត់ទុកបរិមាណ និងល្បឿននៃព័ត៌មានជាអថេរដែលត្រូវគ្រប់គ្រងយ៉ាងសកម្ម ជាជាងផលរំលេចនៃវិបត្តិ។ ការងារសុចរិតភាពព័ត៌មានរបស់ OECD រួមទាំងការវាយតម្លៃ "Facts not Fakes" អំពីរបៀបដែលរដ្ឋាភិបាលសមាជិកកំពុងឆ្លើយតបនឹងព័ត៌មានមិនពិតដោយចេតនា បង្កើតមូលដ្ឋានសម្រាប់ការអះអាងរបស់ប្រធានបទនេះអំពីវិមាត្រស្ថាប័ន និងឆ្លងព្រំដែននៃបញ្ហា។ អក្សរសាស្ត្រសិក្សាស្តីពីទ្រឹស្តីការចាក់ថ្នាំបង្ការ និងការបង្ការជាមុន ជាពិសេសការស្រាវជ្រាវដែលទាក់ទងនឹង Jon Roozenbeek, Sander van der Linden និងអ្នកសហការរបស់ពួកគេ បង្កើតមូលដ្ឋានសម្រាប់ការអះអាងរបស់ប្រធានបទនេះអំពីអ្វីដែលត្រូវបានបង្ហាញក្នុងការស្រាវជ្រាវត្រួតពិនិត្យថាកសាងភាពស៊ាំនឹងព័ត៌មានមិនពិតមុនវារីករាលដាល ខុសពីការកែតម្រូវវាក្រោយ។
