# ប្រធានបទ 4.3 — អាកាសធាតុ និរន្តរភាព និងតម្លៃសាធារណៈបរិស្ថាន

**ការប្រែប្រួលអាកាសធាតុបង្ខំឱ្យទ្រឹស្តីតម្លៃសាធារណៈឆ្លើយសំណួរមួយដែលទីផ្សារមិនអាចកំណត់តម្លៃ ហើយអ្នកបោះឆ្នោតមិនអាចផ្តល់សិទ្ធិបានពេញលេញ៖ តើត្រូវជំពាក់ប៉ុន្មាននៅថ្ងៃនេះ ចំពោះមនុស្សដែលមិនទាន់មានជីវិត មិនអាចបោះឆ្នោត និងមិនអាចជំទាស់នឹងវិក្កយបត្រដែលត្រូវទុកឱ្យពួកគេ។**

## ហេតុអ្វីបានជាបញ្ហានេះសំខាន់សម្រាប់តម្លៃសាធារណៈ

អង្គភាពសាធារណៈគ្រប់រូបឥឡូវស្ថិតនៅក្នុងកាតព្វកិច្ចពីរត្រួតគ្នា ដែលទាញទល់គ្នាលើចំណាត់ថ្នាក់ថវិកាតែមួយ។ មួយគឺចំពោះមនុស្សដែលបានជ្រើសតាំង ផ្តល់ថវិកា ឬគ្រប់គ្រងវានៅឆ្នាំនេះ ដែលតម្រូវការបន្ទាន់របស់ពួកគេមើលឃើញ និងមានផលវិបាកខាងនយោបាយ។ មួយទៀតគឺចំពោះប្រជាជនអនាគត — ខ្លះរស់នៅរួចហើយជាកុមារ ខ្លះមិនទាន់កើត — ដែលផលប្រយោជន៍របស់ពួកគេពិតប្រាកដ ប៉ុន្តែដោយរចនាសម្ព័ន្ធមិនអាចប្រកួតប្រជែងក្នុងការពិគ្រោះយោបល់ ឬការពិនិត្យចំណាយនៅឆ្នាំនេះបាន។ ការប្រែប្រួលអាកាសធាតុគឺជាកំណែដ៏មុតស្រួចបំផុតនៃបញ្ហានេះដែលអ្នកគ្រប់គ្រងសាធារណៈនឹងជួប ពីព្រោះថ្លៃដើមនៃការអសកម្មកើនឡើងជាសមាសធាតុលើរយៈពេលដែលយូរជាងរដ្ឋបាលណាមួយ ខណៈថ្លៃដើមនៃសកម្មភាពធ្លាក់មកឥឡូវនេះ លើថវិកាដែលមានជម្លោះរួចហើយ។

ការពិនិត្យឡើងវិញដ៏មានឥទ្ធិពលឆ្នាំ 2006 របស់សេដ្ឋកិច្ចវិទូ Nicholas Stern សម្រាប់រដ្ឋាភិបាលអង់គ្លេស បានចាប់យកភាពមិនស្មើគ្នានេះយ៉ាងច្បាស់៖ ថ្លៃដើមសេដ្ឋកិច្ច និងមនុស្សនៃការប្រែប្រួលអាកាសធាតុដែលមិនត្រូវបានកាត់បន្ថយនឹងលើសថ្លៃដើមនៃការបង្ការពួកវាក្នុងទីបំផុត ប៉ុន្តែការចំណាយលើការបង្ការត្រូវតែត្រូវបានផ្តល់សិទ្ធិដោយអ្នកសម្រេចចិត្តដែលភាគច្រើនមិនរស់នៅឃើញការជៀសវាងសេណារីយ៉ូផ្ទុយដ៏អាក្រក់បំផុត (សូមមើល **[Stern Review](https://en.wikipedia.org/wiki/Stern_Review)** ដែលនឹងពិភាក្សាបន្ថែមខាងក្រោម)។ ទ្រឹស្តីតម្លៃសាធារណៈមានឡើងយ៉ាងពិតប្រាកដ ដើម្បីផ្តល់ពាក្យសម្តីដល់អ្នកសម្រេចចិត្តសម្រាប់កាតព្វកិច្ចដែលពិតប្រាកដ ប៉ុន្តែមិនត្រូវបានអនុវត្តដោយតម្លៃទីផ្សារណាមួយ ឬសន្លឹកឆ្នោតរបស់អ្នកបោះឆ្នោតបច្ចុប្បន្នណាមួយ ដែលជាហេតុផលដែលប្រធានបទនេះចាត់ទុកសំណួរអាកាសធាតុ និងបរិស្ថានជាសំណួរតម្លៃសាធារណៈស្នូល ជាជាងផ្នែកបន្ថែមឯកទេសសម្រាប់តែក្រសួងបរិស្ថានប៉ុណ្ណោះ។

ប្រធានបទនេះញែកបញ្ហាបីដាច់ដោយឡែកតែទាក់ទងគ្នា ដែលអ្នកដឹកនាំសាធារណៈច្រើនតែលាយបញ្ចូលគ្នាជាប្រចាំ។ បញ្ហាទីមួយគឺ **ការកាត់បន្ថយការប្រែប្រួលអាកាសធាតុ** — ការកាត់បន្ថយការបញ្ចេញឧស្ម័នផ្ទះកញ្ចក់ដែលបណ្តាលឱ្យឡើងកម្តៅ — ដែលដំណើរការខាងសេដ្ឋកិច្ចដូចទំនិញសាធារណៈសកលសុទ្ធសាធ៖ ការកាត់បន្ថយការបញ្ចេញឧស្ម័នរបស់ប្រទេសមួយផ្តល់អត្ថប្រយោជន៍ដល់បរិយាកាសនៃភពផែនដីទាំងមូល ទោះបីជានរណាជាអ្នករួមចំណែកក៏ដោយ ហើយគ្មានប្រទេសណាអាចត្រូវបានដកចេញពីអត្ថប្រយោជន៍ ឬត្រូវបានបង្ខំឱ្យបង់ចំណែករបស់ខ្លួនបានទេ (សូមមើល ប្រធានបទ 1.4 — ទំនិញសាធារណៈ ទំនិញគួរតម្លៃ និងពហុនិយមនៃតម្លៃ សម្រាប់ចំណាត់ថ្នាក់មូលដ្ឋាន)។ បញ្ហាទីពីរគឺ **ការសម្របខ្លួនទៅនឹងការប្រែប្រួលអាកាសធាតុ** — ការកែសម្រួលហេដ្ឋារចនាសម្ព័ន្ធ សេវា និងអាកប្បកិរិយាទៅនឹងអាកាសធាតុដែលកំពុងផ្លាស់ប្តូរ ដែលត្រូវបានចាក់សោមួយផ្នែករួចហើយ — ដែលមានលក្ខណៈមូលដ្ឋានជាងឆ្ងាយ៖ ទំនប់សមុទ្រការពារឆ្នេរជាក់លាក់មួយ ហើយថ្លៃដើម និងអត្ថប្រយោជន៍របស់វាភាគច្រើនធ្លាក់ក្នុងព្រំដែននៃយុត្តាធិការតែមួយ។ បញ្ហាទីបីគឺបញ្ហាគណនេយ្យនៅក្រោមទាំងពីរ៖ របៀបនាំយកតម្លៃនៃប្រព័ន្ធធម្មជាតិ — ព្រៃឈើដែលស្តុកកាបូន ដីសើមដែលស្រូបទឹកជំនន់ ដីដែលដាំដុះស្បៀង — មកក្នុងការសម្រេចចិត្តដែលតាមប្រវត្តិសាស្ត្របានថ្លឹងតែតម្លៃហិរញ្ញវត្ថុ ហើយថ្មីៗនេះតម្លៃសង្គម។ ការលាយបញ្ហាទាំងបីនេះចូលគ្នាគឺជាហេតុផលទូទៅបំផុតមួយដែលយុទ្ធសាស្ត្រអាកាសធាតុជាប់គាំង៖ ក្រុមប្រឹក្សាដែលផ្តល់មូលនិធិតែការសម្របខ្លួន ពីព្រោះការកាត់បន្ថយមើលទៅដូចជាការងាររបស់អ្នកដទៃ មិនមែនកំពុងធ្វើការសម្រេចចិត្តទេ គឺកំពុងគេចវេសពីវា។

## គំនិតស្នូល

**[ការកាត់បន្ថយការប្រែប្រួលអាកាសធាតុ](https://en.wikipedia.org/wiki/Climate_change_mitigation)** ដែលហៅផងដែរថាការកាត់បន្ថយកាបូន គឺជាសកម្មភាពដើម្បីកំណត់ឧស្ម័នផ្ទះកញ្ចក់ដែលចូលបរិយាកាស ទោះដោយកាត់បន្ថយការបញ្ចេញនៅប្រភព (ថាមពលកកើតឡើងវិញ ប្រសិទ្ធភាព អគ្គិសនីកម្ម) ឬដោយបង្កើនការដកចេញ (ការដាំព្រៃឡើងវិញ ការចាប់យកកាបូន)។ ដោយសារបរិយាកាសលាយឡំគ្នាទូទាំងពិភពលោក កាបូនឌីអុកស៊ីតមួយតោនដែលជៀសវាងបានក្នុងប្រទេសមួយផ្តល់អត្ថប្រយោជន៍អាកាសធាតុដូចគ្នា ទោះបីជៀសវាងនៅទីណាក៏ដោយ ដែលធ្វើឱ្យការកាត់បន្ថយជាករណីសៀវភៅសិក្សានៃ **ទំនិញសាធារណៈសកល**៖ មិនប្រកួតប្រជែង និងមិនអាចដកចេញតាមមុខងារ ដោយសារគ្មានប្រទេសណាអាចត្រូវបានបិទមិនឱ្យទទួលអាកាសធាតុស្ថិរភាពដែលខ្លួនមិនបានបង់ដើម្បីធានា។ ការផ្សំគ្នានោះបង្កើតការជំរុញអ្នកជិះសេរីដែលបានពិពណ៌នាក្នុងប្រធានបទ 1.4 — ទំនិញសាធារណៈ ទំនិញគួរតម្លៃ និងពហុនិយមនៃតម្លៃ ក្នុងមាត្រដ្ឋានភពផែនដី ហើយវាជាហេតុផលសំខាន់ដែលការកាត់បន្ថយបានបង្ហាញថាពិបាកចរចាជាសមូហភាពជាងការសម្របខ្លួនឆ្ងាយ ដែលអត្ថប្រយោជន៍របស់វាភាគច្រើនត្រូវបានចាប់យកដោយអ្នកដែលបង់។

**[ការសម្របខ្លួនទៅនឹងការប្រែប្រួលអាកាសធាតុ](https://en.wikipedia.org/wiki/Climate_change_adaptation)** គឺជាដំណើរការនៃការកែសម្រួលទៅនឹងផលប៉ះពាល់នៃការប្រែប្រួលអាកាសធាតុ ទាំងដែលកំពុងកើតឡើង និងត្រូវបានរំពឹងទុក ក្នុងប្រភេទធំៗបួន៖ ហេដ្ឋារចនាសម្ព័ន្ធ (ការការពារទឹកជំនន់ មជ្ឈមណ្ឌលត្រជាក់) ស្ថាប័ន (ក្រមអគារ ប្រព័ន្ធព្រមានជាមុន) អាកប្បកិរិយា (ការផ្លាស់ប្តូរការអនុវត្តកសិកម្ម ឬការប្រើប្រាស់ទឹក) និងផ្អែកលើធម្មជាតិ (ដីសើមដែលស្តារឡើងវិញ ម្លប់ដើមឈើទីក្រុង)។ មិនដូចការកាត់បន្ថយទេ ការសម្របខ្លួនអាចដកចេញបាន និងប្រកួតប្រជែងបានយ៉ាងសំខាន់ — ការការពារទឹកជំនន់ការពារទីក្រុងនៅពីក្រោយវា មិនមែនឆ្នេរបន្ទាប់ទេ — ដែលធ្វើឱ្យវាជាសំណួរតម្លៃសាធារណៈមូលដ្ឋាន និងជាតិជាចម្បង ទាក់ទងយ៉ាងជិតស្និទ្ធនឹង ប៉ុន្តែមិនដូចគ្នានឹងការធ្វើផែនការភាពធន់ក្នុងប្រធានបទ 3.12 — ការគ្រប់គ្រងវិបត្តិ អាសន្ន និងភាពធន់។ ទាំងពីរត្រួតគ្នា ពីព្រោះទាំងពីរទាក់ទងនឹងសមត្ថភាពរបស់អង្គការក្នុងការស្រូបយក និងងើបឡើងវិញពីការឆក់ ប៉ុន្តែការសម្របខ្លួនទាក់ទងនឹងមូលដ្ឋានអាកាសធាតុដែលផ្លាស់ប្តូរជាពិសេស ខណៈការគ្រប់គ្រងភាពធន់គ្របដណ្តប់ជួរទូលំទូលាយជាងនៃការឆក់ រួមទាំងការឆក់ដែលមិនទាក់ទងនឹងអាកាសធាតុ។

**[សមធម៌រវាងជំនាន់](https://en.wikipedia.org/wiki/Intergenerational_equity)** គឺជាគោលការណ៍នៃយុត្តិធម៌រវាងជំនាន់។ វាសំខាន់នៅទីនេះ ពីព្រោះស្ទើរតែគ្រប់ការសម្រេចចិត្តអាកាសធាតុដែលមានផលវិបាក ដោះដូរថ្លៃដើមថ្ងៃនេះសម្រាប់អត្ថប្រយោជន៍ដែលនៅឆ្ងាយច្រើនទសវត្សរ៍ ហើយរបៀបដែលការដោះដូរនោះត្រូវបានវាយតម្លៃពឹងលើ **អត្រាបញ្ចុះតម្លៃសង្គម** ដែលប្រើដើម្បីប្រៀបធៀបតម្លៃបច្ចុប្បន្ន និងអនាគត ដែលត្រូវបានអភិវឌ្ឍយ៉ាងពេញលេញក្នុងប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម។ អត្រាខ្ពស់ជាងបន្ថយទម្ងន់បច្ចុប្បន្នដែលផ្តល់ឱ្យអត្ថប្រយោជន៍ឆ្ងាយ ដែលអាចធ្វើឱ្យកម្មវិធីកាត់បន្ថយមើលទៅដូចជាតម្លៃមិនល្អ ទោះបីជាអត្ថប្រយោជន៍ចុងក្រោយរបស់វាធំធេងក៏ដោយ។ អត្រាទាបជាងផ្តល់សុខុមាលភាពរបស់ជំនាន់អនាគតនូវឋានៈជិតស្មើនឹងបច្ចុប្បន្ន ក្នុងតម្លៃនៃការធ្វើឱ្យកម្មវិធីច្រើនទៀតមើលទៅថ្លៃនៅថ្ងៃនេះ។ ការសម្រេចចិត្តរបស់ Stern Review ក្នុងការប្រើអត្រាបញ្ចុះតម្លៃទាបគួរកត់សម្គាល់សម្រាប់គោលនយោបាយអាកាសធាតុ នៅតែជាការអនុវត្តមួយដែលជាជម្លោះជាសាធារណៈបំផុតនៃការបញ្ចុះតម្លៃក្នុងសេដ្ឋកិច្ចសាធារណៈ ពិតជាដោយសារជម្រើសអត្រាគឺជាការវិនិច្ឆ័យតម្លៃអំពីយុត្តិធម៌រវាងជំនាន់ ដែលស្លៀកពាក់ជាប៉ារ៉ាម៉ែត្របច្ចេកទេស។

**[ដើមទុនធម្មជាតិ](https://en.wikipedia.org/wiki/Natural_capital)** គឺជាស្តុកធនធានធម្មជាតិរបស់ពិភពលោក — ភូគព្ភវិទ្យា ដី ខ្យល់ ទឹក និងសារពាង្គកាយមានជីវិតទាំងអស់ — ដែលត្រូវបានចាត់ទុក តាមការប្រៀបធៀបនឹងដើមទុនហិរញ្ញវត្ថុ ជាទ្រព្យសកម្មដែលផ្តល់លំហូរតម្លៃតាមពេលវេលា។ លំហូរនោះត្រូវបានបង្កើតឡើងដោយ **[សេវាប្រព័ន្ធអេកូឡូស៊ី](https://en.wikipedia.org/wiki/Ecosystem_service)**៖ អត្ថប្រយោជន៍ដែលមនុស្សទទួលបានពីប្រព័ន្ធអេកូឡូស៊ី ចាប់ពីសេវាផ្គត់ផ្គង់ដូចជាឈើ និងជលផល តាមរយៈសេវាបទប្បញ្ញត្តិដូចជាការគ្រប់គ្រងទឹកជំនន់ និងការស្តុកកាបូន ទៅសេវាវប្បធម៌ដូចជាការកម្សាន្ត។ **[គណនេយ្យដើមទុនធម្មជាតិ](https://en.wikipedia.org/wiki/Natural_capital_accounting)** ព្យាយាមវាស់ស្តុក និងលំហូរទាំងនេះក្នុងលក្ខខណ្ឌដែលស្ថិតនៅជាមួយគណនីហិរញ្ញវត្ថុ និងសង្គម ដោយប្រើក្របខ័ណ្ឌដូចជា System of Environmental-Economic Accounting របស់អង្គការសហប្រជាជាតិ និងកម្មវិធី Changing Wealth of Nations របស់ធនាគារពិភពលោក ដូច្នេះតម្លៃការពារទឹកជំនន់របស់ដីសើមអាចផ្តល់ព័ត៌មានដល់ការសម្រេចចិត្តថវិកា ដូចតម្លៃទ្រព្យសកម្មហិរញ្ញវត្ថុបានធ្វើរួចហើយ។ នេះនៅតែជាមុខវិជ្ជាដែលកំពុងលេចចេញ បើធៀបនឹងការអនុវត្តរាប់សតវត្សរ៍នៃគណនេយ្យហិរញ្ញវត្ថុ៖ មានប្រយោជន៍ពិតប្រាកដ និងកំពុងចាស់ទុំយ៉ាងឆាប់រហ័ស ប៉ុន្តែមិនទាន់ត្រូវបានដោះស្រាយពេញលេញទេ។

**[ថ្លៃដើមសង្គមនៃកាបូន](https://en.wikipedia.org/wiki/Social_cost_of_carbon)** ភ្ជាប់ដើមទុនធម្មជាតិ ការបញ្ចុះតម្លៃ និងការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍ក្នុងលេខតែមួយ៖ ការប៉ាន់ស្មាន ដែលជាធម្មតាបង្ហាញជាប្រាក់ក្នុងមួយតោន អំពីការខូចខាតសេដ្ឋកិច្ចដែលបណ្តាលដោយការបញ្ចេញកាបូនឌីអុកស៊ីតបន្ថែមមួយតោន ដែលប្រើតាមរបៀបដែលប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម ប្រើតម្លៃស្រមោលសម្រាប់ទំនិញមិនមែនទីផ្សារផ្សេងទៀត។ វាងាយរងឥទ្ធិពលយ៉ាងខ្លាំងពីអត្រាបញ្ចុះតម្លៃដែលបានជ្រើសរើស ដោយសារការខូចខាតភាគច្រើនរបស់មួយតោនធ្លាក់ក្នុងអនាគតច្រើនទសវត្សរ៍។

ស្ថាប័នបីជាយុថ្កាស្ថាបត្យកម្មសកលជុំវិញគំនិតទាំងនេះ។ **[Intergovernmental Panel on Climate Change](https://en.wikipedia.org/wiki/Intergovernmental_Panel_on_Climate_Change)** (IPCC) ដែលជាស្ថាប័នមួយរបស់អង្គការសហប្រជាជាតិ វាយតម្លៃ និងសំយោគវិទ្យាសាស្ត្រដែលបានបោះពុម្ពផ្សាយអំពីការប្រែប្រួលអាកាសធាតុសម្រាប់អ្នកធ្វើគោលនយោបាយ ដោយមិនធ្វើការស្រាវជ្រាវបឋមផ្ទាល់ខ្លួន។ **[United Nations Framework Convention on Climate Change](https://en.wikipedia.org/wiki/United_Nations_Framework_Convention_on_Climate_Change)** (UNFCCC) គឺជាដំណើរការសន្ធិសញ្ញាដែលប្រទេសនានាចរចាសកម្មភាពអាកាសធាតុរួម ហើយ **[កិច្ចព្រមព្រៀងប៉ារីស](https://en.wikipedia.org/wiki/Paris_Agreement)** ដែលអនុម័តក្រោមវាក្នុងឆ្នាំ 2015 គឺជាក្របខ័ណ្ឌច្បាប់បច្ចុប្បន្នដែលចងភាគីឱ្យរក្សាការឡើងកម្តៅឱ្យទាបជាងពីរអង្សាសេពីកម្រិតមុនឧស្សាហកម្ម ដែលត្រូវបានអនុវត្តតាមរយៈការរួមចំណែកកំណត់ជាតិដែលប្រទេសនីមួយៗកំណត់ និងពង្រឹងជាកាលកំណត់។ រួមជាមួយការកាត់បន្ថយ និងការសម្របខ្លួន អ្នកចរចាកាន់តែទទួលស្គាល់ប្រភេទទីបី គឺ **[ការបាត់បង់ និងការខូចខាត](https://en.wikipedia.org/wiki/Loss_and_damage_(climate_change))**៖ គ្រោះថ្នាក់សេសសល់ដែលកើតឡើងបន្ទាប់ពីការកាត់បន្ថយបរាជ័យក្នុងការទប់ស្កាត់ការផ្លាស់ប្តូរ ហើយការសម្របខ្លួនបរាជ័យ ឬមិនអាចស្រូបយក បង្កើតសំណួរសមធម៌ផ្ទាល់ខ្លួនអំពីអ្នកណាទទួលបន្ទុកថ្លៃដើមដែលទាំងការបង្ការ ឬការកែសម្រួលមិនបានជៀសវាង។

## ការអនុវត្តល្អបំផុត

1. **សូមញែកសំណួរកាត់បន្ថយចេញពីសំណួរសម្របខ្លួន ក្នុងគ្រប់ឯកសារយុទ្ធសាស្ត្រ។** ការកាត់បន្ថយគឺជាការរួមចំណែកដល់ទំនិញសាធារណៈសកល ដែលអត្ថប្រយោជន៍មូលដ្ឋានរបស់វាខ្ចាត់ខ្ចាយ (សូមមើល ប្រធានបទ 1.4 — ទំនិញសាធារណៈ ទំនិញគួរតម្លៃ និងពហុនិយមនៃតម្លៃ)។ ការសម្របខ្លួនការពារទ្រព្យសកម្ម ឬប្រជាជនមូលដ្ឋានជាក់លាក់។ មូលនិធិ និងអភិបាលកិច្ចខុសគ្នាយ៉ាងខ្លាំងរវាងទាំងពីរ ហើយការលាយបញ្ចូលទៅជាបន្ទាត់ "យុទ្ធសាស្ត្រអាកាសធាតុ" តែមួយ ជាធម្មតាទុកឱ្យមួយខ្វះធនធាន។
2. **សូមបញ្ជាក់អត្រាបញ្ចុះតម្លៃដែលប្រើក្នុងការវាយតម្លៃអាកាសធាតុណាមួយ និងអ្នកណាបានជ្រើសរើសវា។** សូមបោះពុម្ពផ្សាយអត្រា ប្រភពរបស់វា និងការវិភាគភាពប្រែប្រួលដែលបង្ហាញពីរបៀបដែលអនុសាសន៍ផ្លាស់ប្តូរនៅអត្រាទាបជាង និងខ្ពស់ជាង (សូមមើល ប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម)។
3. **សូមចាត់ទុកការសម្របខ្លួនជាការវិនិយោគភាពធន់ជាមួយអភិបាលកិច្ចផ្ទាល់ខ្លួន មិនមែនជាសំណុំរងនៃការឆ្លើយតបអាសន្នទេ។** ការសម្របខ្លួនចែករំលែកឧបករណ៍ជាមួយការគ្រប់គ្រងភាពធន់ក្នុងប្រធានបទ 3.12 — ការគ្រប់គ្រងវិបត្តិ អាសន្ន និងភាពធន់ ប៉ុន្តែត្រូវការកម្មវិធីអចិន្ត្រៃយ៍ច្រើនឆ្នាំ ជាជាងកម្មវិធីដែលបង្កឡើងដោយវិបត្តិ ដោយសារហានិភ័យកើនឡើងក្នុងរយៈពេលច្រើនទសវត្សរ៍។
4. **សូមចាប់ផ្តើមបញ្ជីដើមទុនធម្មជាតិសម្រាប់ទ្រព្យសកម្មដីធ្លី និងទឹកសំខាន់ៗរបស់លោកអ្នក ទោះបីមិនពេញលេញក៏ដោយ។** បញ្ជីសារពើភ័ណ្ឌដំបូងនៃព្រៃឈើ ដីសើម និងទីធ្លាបៃតងដែលអង្គការមាន ជាមួយកំណត់សម្គាល់គុណភាពអំពីសេវាប្រព័ន្ធអេកូឡូស៊ីនៃទ្រព្យសកម្មនីមួយៗ មានប្រយោជន៍ជាងការរង់ចាំការវាយតម្លៃជាប្រាក់ឥតខ្ចោះ។
5. **សូមកុំឱ្យតួលេខថ្លៃដើមសង្គមនៃកាបូនតែមួយជំនួសឱ្យករណីអាកាសធាតុទាំងមូល។** តួលេខនេះស្របច្បាប់ និងមានប្រយោជន៍ ប៉ុន្តែភាពប្រែប្រួលរបស់វាចំពោះអត្រាបញ្ចុះតម្លៃមានន័យថាការសម្រេចចិត្តដែលឆ្លងកាត់តែការធ្វើតេស្តថ្លៃដើម-អត្ថប្រយោជន៍ ដោយសារជម្រើសអត្រាអំណោយផល សមនឹងទទួលបានការត្រួតពិនិត្យបន្ថែម មិនមែនការយល់ព្រមដោយស្វ័យប្រវត្តិទេ។
6. **សូមរាយការណ៍ការប៉ះពាល់នឹងហានិភ័យអាកាសធាតុក្នុងវដ្តតែមួយជាមួយហានិភ័យហិរញ្ញវត្ថុ មិនមែនជាឧបសម្ព័ន្ធប្រចាំឆ្នាំដែលគ្មាននរណាអានទេ។** សូមផ្តល់ហានិភ័យអាកាសធាតុរូបវន្ត — ទឹកជំនន់ កំដៅ គ្រោះរាំងស្ងួត ការប៉ះពាល់នឹងព្យុះ — នូវចង្វាក់ពិនិត្យអចិន្ត្រៃយ៍ដូចគ្នានឹងហានិភ័យហិរញ្ញវត្ថុ និងអ៊ីនធឺណិត ដោយទៅដល់ស្ថាប័នអភិបាលកិច្ចដូចគ្នាដោយភាពម៉ឺងម៉ាត់ដូចគ្នា។
7. **សូមដាក់ឈ្មោះការដោះដូររវាងជំនាន់ឱ្យច្បាស់លាស់ក្នុងករណីអាជីវកម្មដែលមានចក្ខុវិស័យពេលវេលាវែង។** នៅកន្លែងដែលការសម្រេចចិត្តដាក់ថ្លៃដើមលើជំនាន់បច្ចុប្បន្នដើម្បីអត្ថប្រយោជន៍ដល់ជំនាន់អនាគត ឬផ្ទុយមកវិញ សូមបញ្ជាក់ការដោះដូរនោះជាភាសាសាមញ្ញ ជាជាងទុកឱ្យវាលាក់នៅក្នុងអត្រាបញ្ចុះតម្លៃដែលមិនត្រូវបានសួរចម្លើយ។
8. **សូមកសាងផែនការសម្របខ្លួនជុំវិញប្រជាជនដែលងាយរងគ្រោះបំផុតមុន មិនមែនទ្រព្យសកម្មដែលមានតម្លៃបំផុតមុនទេ។** ផែនការដែលថ្លឹងតែតម្លៃទ្រព្យសកម្មការពារតំបន់ដែលមានជាងមុនតំបន់ក្រីក្រ ពីព្រោះតម្លៃទ្រព្យសកម្មនៅទីនោះខ្ពស់ជាង។ សូមថ្លឹងភាពងាយរងគ្រោះ និងការប៉ះពាល់ឱ្យច្បាស់លាស់ នៅពេលរៀបលំដាប់វិនិយោគ។
9. **សូមសម្របសម្រួលការប្តេជ្ញាកាត់បន្ថយបញ្ឈរ ពីការសន្យាមូលដ្ឋានទៅការរួមចំណែកកំណត់ជាតិ។** គោលដៅសូន្យសុទ្ធមូលដ្ឋានគួរឱ្យជឿជាក់លុះត្រាតែវាស្របនឹងការប្តេជ្ញាកិច្ចព្រមព្រៀងប៉ារីសរបស់រដ្ឋាភិបាលជាតិ។ សូមបោះពុម្ពផ្សាយរបៀបដែលគោលដៅមូលដ្ឋានបូកសរុបឡើង ជាជាងកំណត់វាដាច់ដោយឡែក។
10. **សូមញែកការបាត់បង់ និងការខូចខាតដែលជៀសវាងបាន ចេញពីជោគជ័យនៃការសម្របខ្លួន ក្នុងការរាយការណ៍របស់លោកអ្នកផ្ទាល់។** មូលនិធិសម្រាប់ការបាត់បង់ និងការខូចខាតទូទាត់សំណងគ្រោះថ្នាក់ដែលបានកើតឡើងទោះបីមានការកាត់បន្ថយ និងការសម្របខ្លួន។ ការលាយវាជាមួយបន្ទាត់ថវិកាសម្របខ្លួនធ្វើឱ្យប៉ាន់ស្មានទាបអំពីបរិមាណការការពារដែលបរាជ័យពិតប្រាកដ។
11. **សូមសាកល្បងគណនេយ្យដើមទុនធម្មជាតិលើថ្នាក់ទ្រព្យសកម្មតែមួយ មុនដាក់ឱ្យប្រើទូទាំងអង្គការ។** សូមជ្រើសរើសករណីមានព្រំដែនមួយ — អាងទន្លេតែមួយ ឬបណ្តាញទីធ្លាបៃតង — ដើម្បីរៀនដែនកំណត់ជាក់ស្តែងនៃវិធីសាស្ត្រ មុនប្តេជ្ញាចិត្តចំពោះតារាងតុល្យការដើមទុនធម្មជាតិពេញលេញ។
12. **សូមពិនិត្យឡើងវិញការសន្មត់អាកាសធាតុតាមវដ្តកំណត់ មិនមែនតែក្រោយវិបត្តិប៉ុណ្ណោះទេ។** គន្លងការបញ្ចេញឧស្ម័ន និងការរកឃើញនៃការវាយតម្លៃ IPCC ត្រូវបានធ្វើបច្ចុប្បន្នភាពជាកាលកំណត់។ សូមកំណត់កាលវិភាគសម្រាប់ការពិនិត្យការសន្មត់នៅពីក្រោយផែនការធំៗ ជាជាងរង់ចាំទឹកជំនន់ ឬរលកកំដៅបង្ខំសំណួរ។

## សំណួរសម្រាប់ពិភាក្សាជាមួយក្រុម

**តើអត្រាបញ្ចុះតម្លៃអ្វីជាមូលដ្ឋានរបស់ការសម្រេចចិត្តអាកាសធាតុរយៈពេលវែងរបស់យើង ហើយតើយើងអាចការពារវាចំពោះមនុស្សដែលវាធ្វើឱ្យមានគុណវិបត្តិបានឬទេ?** ស្ទើរតែគ្រប់ការសម្រេចចិត្តកាត់បន្ថយ ឬសម្របខ្លួនសំខាន់ៗ បង្កប់អត្រាបញ្ចុះតម្លៃដែលកំណត់ទម្ងន់ដែលសុខុមាលភាពរបស់ជំនាន់អនាគតទទួលបានធៀបនឹងជំនាន់បច្ចុប្បន្ន។ សូមសួរថាតើអត្រារបស់អង្គការលោកអ្នកមកពីណា — ការណែនាំខ្ចី កំណត់ដោយចេតនាសម្រាប់ការងារអាកាសធាតុ ឬមិនដែលបានសម្រេចពិតប្រាកដសោះ។ សូមពិភាក្សាថាតើមានអ្វីកើតឡើងចំពោះករណីអាជីវកម្មអាកាសធាតុចម្បងរបស់លោកអ្នកនៅអត្រាទាបជាងយ៉ាងមានន័យ ជិតនឹងចំណូលចិត្តពេលវេលាសុទ្ធជិតសូន្យដែល Stern Review បានប្រើ ឬអត្រាហេដ្ឋារចនាសម្ព័ន្ធធម្មតាខ្ពស់ជាង។ ប្រសិនបើចំណាត់ថ្នាក់ត្រឡប់ក្រោមជម្រើសជំនួសដែលអាចទទួលយកបាន នោះអស្ថិរភាពនោះគឺជាអ្វីដែលបរិយាកាសផ្តល់សិទ្ធិរបស់លោកអ្នកត្រូវឃើញ (សូមមើល ប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម)។

**តើយើងកំពុងផ្តល់មូលនិធិការកាត់បន្ថយដូចជាវាជាការងាររបស់អ្នកដទៃ ពីព្រោះអត្ថប្រយោជន៍របស់វាខ្ចាត់ខ្ចាយ និងសកលឬ?** ឋានៈនៃការកាត់បន្ថយជាទំនិញសាធារណៈសកលមានន័យថា ការរួមចំណែករបស់អង្គការតែមួយណាមួយ ផ្តល់ត្រឡប់មកវិញតែចំណែកអត្ថប្រយោជន៍តូចតាច និងខ្ចាត់ខ្ចាយ បង្កើតការល្បួងពិតប្រាកដក្នុងការវិនិយោគមិនគ្រប់គ្រាន់ ហើយទុកឱ្យអ្នកដទៃទទួលបន្ទុកថ្លៃដើម។ សូមឆ្លងកាត់ការប្តេជ្ញាចិត្តរបស់លោកអ្នក ហើយសួរដោយស្មោះត្រង់ថា តើវាត្រូវនឹងការរួមចំណែកដ៏សមរម្យរបស់លោកអ្នកចំពោះកិច្ចខិតខំសកលរួម ឬគ្រាន់តែអ្វីដែលមើលទៅអាចការពារបានធៀបនឹងដៃគូប្រៀបធៀប។ សូមពិភាក្សាថាតើកម្រិតមហិច្ឆតាដែលគួរឱ្យជឿជាក់ ជាជាងគ្រាន់តែអាចប្រៀបធៀបបាន មើលទៅដូចម្តេចសម្រាប់អង្គការទំហំរបស់លោកអ្នក (សូមមើល ប្រធានបទ 1.4 — ទំនិញសាធារណៈ ទំនិញគួរតម្លៃ និងពហុនិយមនៃតម្លៃ)។

**តើភាពងាយរងគ្រោះរបស់អ្នកណា មិនមែនទ្រព្យសកម្មរបស់អ្នកណា ដែលកំពុងជំរុញអាទិភាពសម្របខ្លួនរបស់យើង?** ផែនការសម្របខ្លួនដែលកសាងជុំវិញការការពារទ្រព្យសម្បត្តិមានតម្លៃខ្ពស់បំផុត នឹងពេញចិត្តជាប្រព័ន្ធដល់តំបន់ដែលមានជាង ពីព្រោះនោះជាកន្លែងដែលតម្លៃទ្រព្យសកម្មប្រមូលផ្តុំ ទុកឱ្យប្រជាជនក្រីក្រ និងប៉ះពាល់ខ្លាំងជាងទទួលការការពារមិនគ្រប់គ្រាន់។ សូមពិនិត្យឡើងវិញលំដាប់បច្ចុប្បន្ន ឬសេចក្តីព្រាងរបស់លោកអ្នក ហើយសួរថាតើវាផ្តល់ទម្ងន់អ្វី ប្រសិនបើមាន ដល់ភាពងាយរងគ្រោះសង្គម ជាជាងតម្លៃទ្រព្យសកម្មតែម្យ៉ាង។ សូមពិភាក្សាថាអ្នកណានឹងទទួលការការពារមុន និងក្រោយគេក្រោមផែនការបច្ចុប្បន្ន ហើយថាតើលំដាប់នោះអាចការពារបានជាសាធារណៈឬអត់។

**តើយើងដឹងទេថាដើមទុនធម្មជាតិអ្វីដែលយើងទទួលខុសត្រូវ ហើយអ្វីដែលវាកំពុងធ្វើសម្រាប់យើង?** អង្គការភាគច្រើនដែលជាម្ចាស់ ឬជះឥទ្ធិពលលើដីធ្លី ទឹក ឬហេដ្ឋារចនាសម្ព័ន្ធបៃតង មិនដែលធ្វើសារពើភ័ណ្ឌសេវាប្រព័ន្ធអេកូឡូស៊ីដែលទ្រព្យសកម្មទាំងនោះផ្តល់ ចាប់ពីការកាត់បន្ថយទឹកជំនន់ រហូតដល់ការស្តុកកាបូនទេ។ សូមពិភាក្សាថាតើបញ្ជីដើមទុនធម្មជាតិដំបូងត្រូវគ្របដណ្តប់អ្វី អ្នកណានឹងជាម្ចាស់វា និងការសម្រេចចិត្តអ្វីដែលវានឹងផ្តល់ព័ត៌មានពេលកសាងរួច — បញ្ជីដែលគ្មាននរណាពិគ្រោះមិនសមនឹងកសាងទេ។ សូមពិចារណាចាប់ផ្តើមជាមួយថ្នាក់ទ្រព្យសកម្មមានព្រំដែនមួយ ជាជាងការវាយតម្លៃទូលំទូលាយភ្លាមៗ។

**តើយើងដោះស្រាយភាពខុសគ្នារវាងការសម្របខ្លួនដែលបានជោគជ័យ និងការបាត់បង់ និងការខូចខាតដែលមានន័យថាវាមិនបានជោគជ័យដោយរបៀបណា?** ការសម្របខ្លួន និងការបាត់បង់ និងការខូចខាតអាចមើលទៅស្រដៀងគ្នាក្នុងបន្ទាត់ថវិកា ប៉ុន្តែវាមានន័យខុសគ្នាខ្លាំងអំពីរបៀបដែលអង្គការត្រៀមខ្លួនបានល្អពិតប្រាកដ។ សូមពិភាក្សាថាតើការរាយការណ៍របស់លោកអ្នកញែក "យើងបានសម្របខ្លួនដោយជោគជ័យ" ពី "គ្រោះថ្នាក់បានកើតឡើងដែលការសម្របខ្លួនរបស់យើងមិនបានទប់ស្កាត់ ហើយឥឡូវយើងកំពុងងើបឡើងវិញពីវា" ឬអត់ និងអ្វីដែលត្រូវការដើម្បីធ្វើឱ្យការញែកនោះមើលឃើញដល់ស្ថាប័នគ្រប់គ្រងរបស់លោកអ្នក ជាជាងត្រូវបានធ្វើឱ្យព្រិលចូលក្នុងចំនួនសរុបតែមួយ។

**តើយើងបានធ្វើបច្ចុប្បន្នភាពការសន្មត់អាកាសធាតុនៅពីក្រោយផែនការរយៈពេលវែងសំខាន់ៗរបស់យើងចុងក្រោយនៅពេលណា ហើយធៀបនឹងភស្តុតាងអ្វី?** គន្លងការបញ្ចេញឧស្ម័ន ការព្យាករណ៍គ្រោះថ្នាក់ និងការវាយតម្លៃវិទ្យាសាស្ត្រនៅពីក្រោយពួកវាផ្លាស់ប្តូរតាមពេលវេលា ហើយផែនការដែលកសាងលើការសន្មត់ចាស់ជាងអាចហួសសម័យយ៉ាងសំខាន់រួចហើយ។ សូមកំណត់ផែនការរយៈពេលវែងសំខាន់ៗ — ហេដ្ឋារចនាសម្ព័ន្ធ ការប្រើប្រាស់ដី ការការពារទឹកជំនន់ ភាពបន្តសេវា — ដែលពឹងលើការសន្មត់អាកាសធាតុ ហើយពិនិត្យថាតើនីមួយៗត្រូវបានសាកល្បងធៀបនឹងភស្តុតាងបច្ចុប្បន្នចុងក្រោយនៅពេលណា រួមទាំងវដ្តវាយតម្លៃ Intergovernmental Panel on Climate Change ចុងក្រោយបំផុត។ សូមឯកភាពថាអ្នកណាជាម្ចាស់វដ្តពិនិត្យនោះ ដើម្បីកុំឱ្យផែនការហួសសម័យតាមលំនាំដើម។

សូមពិចារណាតំបន់ដីសណ្តទន្លេប្រឌិតមួយ គឺ Meridian Delta Authority ដែលជាអង្គភាពសាធារណៈតំបន់ទទួលខុសត្រូវចំពោះការការពារទឹកជំនន់ និងការធ្វើផែនការប្រើប្រាស់ដីលើតំបន់កសិកម្ម និងទីក្រុងទំនាបដែលមានទន្លេបីកាត់កាត។ អស់ច្រើនឆ្នាំ ផែនការមូលធនរបស់វាបានរៀបលំដាប់គម្រោងលើសមាមាត្រអត្ថប្រយោជន៍-ថ្លៃដើមតែមួយ គណនានៅអត្រាបញ្ចុះតម្លៃហេដ្ឋារចនាសម្ព័ន្ធស្តង់ដាររបស់តំបន់ ដែលទទួលមរតកដោយមិនផ្លាស់ប្តូរពីគំរូរតនាគារជាតិអាយុមួយទសវត្សរ៍ ដែលត្រូវបានរចនាសម្រាប់ការវាយតម្លៃការសាងសង់ផ្លូវ។

ក្រោមការរៀបលំដាប់នោះ ជញ្ជាំងការពារទឹកជំនន់រឹងដែលបានស្នើជុំវិញទីក្រុងធំបំផុតរបស់តំបន់ ទទួលបានពិន្ទុខ្ពស់ជាងជាប់លាប់ បើធៀបនឹងសំណើប្រកួតប្រជែងក្នុងការស្តារដីសើម 600 ហិកតានៅផ្នែកខាងលើ ដែលនឹងបន្ថយ និងស្តុកទឹកជំនន់តាមធម្មជាតិ ខណៈស្រូបយកកាបូន និងគាំទ្រជលផល។ អត្ថប្រយោជន៍របស់ជញ្ជាំងមកដល់ស្ទើរតែភ្លាមៗក្រោយការសាងសង់ ហើយត្រូវបានវាយតម្លៃដោយប្រើវិធីសាស្ត្រវាយតម្លៃទ្រព្យសម្បត្តិ និងហេដ្ឋារចនាសម្ព័ន្ធដែលបានបង្កើតយ៉ាងរឹងមាំ។ អត្ថប្រយោជន៍របស់ដីសើមកើនឡើងក្នុងរយៈពេលមួយទសវត្សរ៍នៅពេលរុក្ខជាតិចាស់ទុំ ហើយចំណែកសំខាន់នៃវា — ការស្តុកកាបូន ជីវចម្រុះ ភាពធន់វាលទំនាបរយៈពេលវែង — មិនដែលត្រូវបានកំណត់តម្លៃទាល់តែសោះ ពីព្រោះអាជ្ញាធរគ្មានការអនុវត្តគណនេយ្យដើមទុនធម្មជាតិ ហើយគ្រាន់តែទុកអត្ថប្រយោជន៍ទាំងនោះចេញពីផលបូក ជាជាងកំណត់តម្លៃសូន្យ ដែលបានធ្វើឱ្យករណីរបស់ជម្រើសដីសើមត្រូវបានប៉ាន់ស្មានទាបដោយស្ងាត់ៗ ដោយគ្មាននរណាសម្រេចចិត្តធ្វើដូច្នោះ។

ប្រធានយុទ្ធសាស្ត្រដែលទើបតែងតាំងថ្មីម្នាក់បានសួរនូវអ្វីដែលការវាយតម្លៃស្តង់ដារមិនធ្លាប់ឆ្លើយដោយឡែក៖ តើអត្រាបញ្ចុះតម្លៃអ្វីសមស្របពិតប្រាកដសម្រាប់ការសម្រេចចិត្តដែលអត្ថប្រយោជន៍នឹងលាតសន្ធឹងយ៉ាងហោចណាស់ហាសិបឆ្នាំ ហើយតើករណីរបស់ជម្រើសដីសើមនឹងមើលទៅដូចម្តេច ប្រសិនបើសេវាប្រព័ន្ធអេកូឡូស៊ីរបស់វាត្រូវបានប៉ាន់ស្មាន ជាជាងត្រូវបានចាត់ទុកដោយស្ងាត់ៗថាមិនមានតម្លៃ។ ដោយធ្វើការជាមួយសេដ្ឋកិច្ចវិទូបរិស្ថាន ក្រុមបានបង្កើតការប៉ាន់ស្មានដើមទុនធម្មជាតិដំបូងសម្រាប់ដីសើម ដោយយកតម្លៃផ្ទេរដែលបានបោះពុម្ពផ្សាយសម្រាប់គម្រោងស្តារវាលទំនាបប្រៀបធៀប ដែលត្រូវបានដាក់ស្លាកយ៉ាងច្បាស់ថាជាការប៉ាន់ស្មានដែលមានជួរសមហេតុផលទូលំទូលាយ។ ពួកគេក៏បានដំណើរការការវាយតម្លៃឡើងវិញនៅអត្រាបញ្ចុះតម្លៃទាបជាង ដែលឆ្លុះបញ្ចាំងពីចក្ខុវិស័យហានិភ័យទឹកជំនន់ច្រើនទសវត្សរ៍របស់តំបន់ផ្ទាល់ ហើយបានរាយការណ៍ចំណាត់ថ្នាក់ទាំងពីរក្បែរគ្នា ជាជាងជ្រើសរើសមួយ ហើយលាក់មួយទៀត។

ក្រោមការសន្មត់ដើម ជញ្ជាំងរឹងនៅតែឈ្នះយ៉ាងងាយ។ ក្រោមការសន្មត់កែសម្រួល ជម្រើសទាំងពីរនៅជិតគ្នាគ្រប់គ្រាន់ ដែលជម្រើសមិនអាចត្រូវបានបង្ហាញជាសេចក្តីសម្រេចប្រសិទ្ធភាពត្រង់ៗទៀតទេ ហើយបានក្លាយជាការវិនិច្ឆ័យពិតប្រាកដសម្រាប់ក្រុមប្រឹក្សាអភិបាលរបស់អាជ្ញាធរ ដែលក៏បានថ្លឹងថា គម្រោងដីសើមកសាងភាពធន់រយៈពេលវែងដែលជញ្ជាំងតែម្យ៉ាងមិនមាន ខណៈជញ្ជាំងការពារកណ្តាលទីក្រុងបានលឿនជាង ប៉ុន្តែមិនបានធ្វើអ្វីសម្រាប់ដីកសិកម្មឆ្ងាយឡើងលើទេ។ ក្រុមប្រឹក្សាបានជ្រើសរើសកូនកាត់៖ គម្រោងការពារឯករាជ្យតូចជាងសម្រាប់កណ្តាលទីក្រុង ផ្តល់មូលនិធិលើគុណសម្បត្តិផ្ទាល់ខ្លួនរបស់វា រួមជាមួយការស្តារដីសើមជាដំណាក់កាល ដែលផ្តល់មូលនិធិមួយផ្នែកតាមរយៈជំនួយឥតសំណងស្តារប្រព័ន្ធអេកូឡូស៊ីជាតិ និងមួយផ្នែកទៀតដោយរាប់ការស្តុកកាបូនរបស់វាចូលក្នុងគោលដៅកាត់បន្ថយផ្ទាល់ខ្លួនរបស់អាជ្ញាធរ។ របាយការណ៍របស់ប្រធានយុទ្ធសាស្ត្រអំពីឧប្បត្តិហេតុនេះត្រង់ៗ៖ ការវាយតម្លៃដើមមិនមែនមិនស្មោះត្រង់ទេ ប៉ុន្តែវាបានធ្វើការវិនិច្ឆ័យតម្លៃដោយស្ងាត់ៗ — អំពីការបញ្ចុះតម្លៃ និងអំពីអ្វីដែលរាប់ជាអត្ថប្រយោជន៍ទាល់តែសោះ — ដែលគ្មាននរណាម្នាក់ក្នុងបន្ទប់បានជ្រើសរើសធ្វើពិតប្រាកដ។

## ទស្សនៈវិស័យទាំងបួន

### រដ្ឋាភិបាលមូលដ្ឋាន

អាជ្ញាធរមូលដ្ឋាន និងតំបន់ទទួលបន្ទុកជាក់ស្តែងភាគច្រើននៃការសម្របខ្លួន ពីព្រោះការការពារទឹកជំនន់ ការគ្រប់គ្រងកំដៅទីក្រុង ភាពធន់នៃការផ្គត់ផ្គង់ទឹក និងការធ្វើផែនការប្រើប្រាស់ដីមូលដ្ឋាន ជាធម្មតាជាសមត្ថកិច្ចមូលដ្ឋាន ឬតំបន់ ដែលភ្ជាប់យ៉ាងជិតស្និទ្ធនឹងការងារភាពធន់ក្នុងប្រធានបទ 3.12 — ការគ្រប់គ្រងវិបត្តិ អាសន្ន និងភាពធន់។ បញ្ហាប្រឈមពិសេសរបស់ពួកគេគឺមាត្រដ្ឋានមូលធនធៀបនឹងមាត្រដ្ឋានហានិភ័យ៖ អាជ្ញាធរមូលដ្ឋានតែមួយច្រើនតែប្រឈមនឹងហានិភ័យទឹកជំនន់ ឬកំដៅធំគ្រប់គ្រាន់ដើម្បីតម្រូវឱ្យមានការផ្តល់មូលនិធិរួមពីជាតិ ឬពហុភាគី ហើយអត្រាបញ្ចុះតម្លៃ និងវិធីសាស្ត្រវាយតម្លៃរបស់វាច្រើនតែត្រូវបានខ្ចីពីការណែនាំជាតិ ដែលមិនត្រូវបានរចនាសម្រាប់ចក្ខុវិស័យសម្របខ្លួនមូលដ្ឋានច្រើនទសវត្សរ៍។

### រដ្ឋាភិបាលជាតិ

រដ្ឋាភិបាលជាតិកំណត់ការប្តេជ្ញាកាត់បន្ថយដែលបង្កើតជាការរួមចំណែករបស់ប្រទេសមួយចំពោះកិច្ចព្រមព្រៀងប៉ារីស កាន់ការណែនាំអត្រាបញ្ចុះតម្លៃដែលមានសិទ្ធិអំណាច ដែលការវាយតម្លៃតំបន់ និងមូលដ្ឋានទទួលមរតក (សូមមើល ប្រធានបទ 2.2 — ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម និងការវិភាគប្រសិទ្ធភាពថ្លៃដើម) ហើយកាន់តែសាកល្បងគណនេយ្យដើមទុនធម្មជាតិនៅកម្រិតគណនីជាតិ តាមក្របខ័ណ្ឌដូចជា System of Environmental-Economic Accounting របស់អង្គការសហប្រជាជាតិ។ ភាពតានតឹងចំបងរបស់ពួកគេស្ថិតរវាងការប្តេជ្ញាអាកាសធាតុអន្តរជាតិដែលបានចរចាយ៉ាងលឿនក្នុងវេទិកាពហុភាគី និងសេដ្ឋកិច្ចនយោបាយក្នុងស្រុកអំពីអ្នកណាទទួលថ្លៃដើមរយៈពេលជិតនៃការបំពេញវា។

### វិស័យសង្គម និងវិស័យមិនស្វែងរកប្រាក់ចំណេញ

អង្គការមិនស្វែងរកប្រាក់ចំណេញ និងអង្គការមិនមែនរដ្ឋាភិបាល (NGO) ច្រើនតែផ្តល់ការសម្របខ្លួននៅមូលដ្ឋាន ក្នុងសហគមន៍ដែលមិនអាចផ្តល់មូលនិធិខ្លួនឯងបានតិចបំផុត — ការត្រៀមខ្លួនទឹកជំនន់សហគមន៍ ទីជម្រកធន់នឹងកំដៅ ការពង្រីកកសិកម្មឆ្លាតអាកាសធាតុ — ហើយធ្វើសកម្មភាពជាអ្នកតស៊ូមតិ ដែលជំរុញការទាមទារការបាត់បង់ និងការខូចខាត និងយុត្តិធម៌អាកាសធាតុ ក្នុងនាមប្រជាជនដែលមានសំឡេងផ្លូវការតិចតួចក្នុងការចរចាជាតិ ឬពហុភាគី។ ភាពស្របច្បាប់របស់ពួកគេក្នុងលំហនេះស្ថិតលើភាពជិតស្និទ្ធនឹងសហគមន៍ដែលរងផលប៉ះពាល់ និងភាពគួរឱ្យជឿជាក់នៃបេសកកម្ម ជាជាងសិទ្ធិអំណាចផ្លូវការ ដែលមានន័យថាការរួមចំណែកមានតម្លៃបំផុតរបស់ពួកគេច្រើនតែជាការដាក់ឈ្មោះភាពងាយរងគ្រោះ ឬភាពអយុត្តិធម៌ ដែលការវាយតម្លៃរបស់រដ្ឋាភិបាល ដែលកសាងជុំវិញសមាមាត្រថ្លៃដើម-អត្ថប្រយោជន៍បូកសរុប នឹងមិនលើកឡើងបើមិនដូច្នោះ។

### កម្រិតពហុភាគី និងអន្តរជាតិ

ការកាត់បន្ថយគឺជាករណីគំរូនៃទំនិញសាធារណៈសកលពិតប្រាកដដែលបានពិភាក្សាក្នុងប្រធានបទ 1.4 — ទំនិញសាធារណៈ ទំនិញគួរតម្លៃ និងពហុនិយមនៃតម្លៃ ដែលជាហេតុផលដែល Intergovernmental Panel on Climate Change ដំណើរការ United Nations Framework Convention on Climate Change និងកិច្ចព្រមព្រៀងប៉ារីសមាន៖ ដើម្បីសម្របសម្រួលបញ្ហាសកម្មភាពសមូហភាពដែលគ្មានរដ្ឋណាមួយអាចដោះស្រាយតែម្នាក់ឯង ហើយដែលទុកឱ្យរដ្ឋគ្រប់រូបមានការជំរុញឱ្យអ្នកដទៃធ្វើសកម្មភាពមុន។ ស្ថាប័នអភិវឌ្ឍន៍ពហុភាគី រួមទាំងធនាគារពិភពលោក ក៏ជាអ្នកជំរុញចម្បងនៃការធ្វើស្តង់ដារគណនេយ្យដើមទុនធម្មជាតិឆ្លងប្រទេសដែលមានកម្រិតចំណូល និងសមត្ថភាពទិន្នន័យខុសគ្នាខ្លាំង និងនៃការរៀបចំស្ថាបត្យកម្មហិរញ្ញវត្ថុសម្រាប់ការបាត់បង់ និងការខូចខាតដែលកំពុងលេចចេញ រវាងប្រទេសបញ្ចេញខ្ពស់ និងទាប។

## ទម្រង់នៃការបរាជ័យដែលជួបញឹកញាប់

- **អត្រាបញ្ចុះតម្លៃដែលខ្ចី។** ការវាយតម្លៃអាកាសធាតុ និងហេដ្ឋារចនាសម្ព័ន្ធជាប្រចាំទទួលមរតកអត្រាបញ្ចុះតម្លៃដែលរចនាសម្រាប់គម្រោងផ្លូវ ឬអគាររយៈពេលខ្លី ដែលប៉ាន់ស្មានតម្លៃអត្ថប្រយោជន៍រយៈពេលវែងនៃការកាត់បន្ថយ និងការសម្របខ្លួនទាបជាប្រព័ន្ធ ដោយគ្មាននរណាសម្រេចចិត្តដោយចេតនាថាជំនាន់អនាគតគួររាប់តិចជាង។
- **ការកាត់បន្ថយជាទំនួលខុសត្រូវរបស់អ្នកដទៃ។** ដោយសារអត្ថប្រយោជន៍នៃការកាត់បន្ថយខ្ចាត់ខ្ចាយ និងសកល អង្គការនីមួយៗវិនិយោគតិចជាងចំណែកដ៏សមរម្យរបស់ខ្លួន ដោយវែកញែកដោយប្រយោលថា ការរួមចំណែកតូច និងមូលដ្ឋានចំពោះបញ្ហាភពផែនដីមិនសមនឹងថ្លៃដើមមូលដ្ឋាន — ការគណនាដែលមើលទៅសមហេតុផលសម្រាប់តួអង្គតែមួយ ប៉ុន្តែបំផ្លិចបំផ្លាញជាសមូហភាព ប្រសិនបើតួអង្គគ្រប់រូបធ្វើវា។
- **ការសម្របខ្លួនដែលរៀបលំដាប់តាមតម្លៃទ្រព្យសកម្ម មិនមែនតាមភាពងាយរងគ្រោះ។** ការការពារទ្រព្យសម្បត្តិថ្លៃបំផុតមុន ជាជាងប្រជាជនដែលប៉ះពាល់ខ្លាំងបំផុតមុន បង្កើតវិសមភាពដែលមានស្រាប់ឡើងវិញដោយស្ងាត់ៗ ក្រោមទង់នៃការបែងចែកធនធានប្រកបដោយប្រសិទ្ធភាព។
- **ការកំណត់តម្លៃសូន្យស្ងាត់ៗចំពោះដើមទុនធម្មជាតិ។** ការទុកសេវាប្រព័ន្ធអេកូឡូស៊ីចេញពីការវាយតម្លៃ ពីព្រោះវាមិនទាន់ត្រូវបានបម្លែងជាប្រាក់ មិនមែនអព្យាក្រឹតទេ។ វាស្មើនឹងការកំណត់តម្លៃសូន្យដល់ពួកវា ដែលធ្វើឱ្យជម្រើសផ្អែកលើធម្មជាតិមានគុណវិបត្តិជាប្រព័ន្ធធៀបនឹងហេដ្ឋារចនាសម្ព័ន្ធធម្មតារាល់ពេល ដោយជម្រើសនោះមិនដែលត្រូវបានបញ្ជាក់ ឬការពារ។
- **ការលាយការបាត់បង់ និងការខូចខាតជាមួយការចំណាយលើការសម្របខ្លួន។** ការរាយការណ៍សំណងសម្រាប់គ្រោះថ្នាក់ដែលការសម្របខ្លួនបរាជ័យក្នុងការទប់ស្កាត់ ដូចជាភស្តុតាងថាការសម្របខ្លួនដំណើរការ បិទបាំងបរិមាណការការពារដែលបរាជ័យពិតប្រាកដ ហើយពន្យាពេលការគណនាអំពីថាតើការបង្ការបន្ថែមនឹងមានថ្លៃប៉ុនណា។
- **ផែនការជាប់គាំងដែលកសាងលើការសន្មត់ហួសសម័យ។** ផែនការហេដ្ឋារចនាសម្ព័ន្ធ និងការប្រើប្រាស់ដីច្រើនទសវត្សរ៍ ដែលកសាងលើការសន្មត់ការបញ្ចេញឧស្ម័ន ឬកម្រិតទឹកសមុទ្រដែលហួសសម័យច្រើនឆ្នាំ ក្លាយជាបំណុលដោយស្ងាត់ៗ ពីព្រោះគ្មាននរណាទទួលការងារពិនិត្យថាតើវិទ្យាសាស្ត្រអាកាសធាតុនៅពីក្រោយបានផ្លាស់ប្តូរចាប់តាំងពីផែនការត្រូវបានអនុម័តឬអត់។

## បញ្ជីត្រួតពិនិត្យ

- [ ] សូមធ្វើថវិកា គ្រប់គ្រង និងរាយការណ៍ការកាត់បន្ថយ និងការសម្របខ្លួនជាកម្មវិធីដាច់ដោយឡែក មិនមែនជាបន្ទាត់អាកាសធាតុតែមួយគ្មានភាពខុសគ្នាទេ។
- [ ] សូមបញ្ជាក់ អះអាងប្រភព និងសាកល្បងភាពប្រែប្រួលនៃអត្រាបញ្ចុះតម្លៃដែលប្រើក្នុងការវាយតម្លៃអាកាសធាតុណាមួយ ធៀបនឹងអត្រាជំនួសយ៉ាងហោចណាស់មួយ។
- [ ] សូមដាក់ឈ្មោះការដោះដូររវាងជំនាន់ក្នុងការសម្រេចចិត្តរយៈពេលវែងឱ្យច្បាស់លាស់សម្រាប់ស្ថាប័នផ្តល់សិទ្ធិ មិនមែនកប់ក្នុងឧបសម្ព័ន្ធបច្ចេកទេសទេ។
- [ ] សូមមានបញ្ជីដើមទុនធម្មជាតិ ទោះបីជាទម្រង់ដំបូងក៏ដោយ សម្រាប់ទ្រព្យសកម្មដីធ្លី និងទឹកសំខាន់ៗដែលអង្គការជាម្ចាស់ ឬមានឥទ្ធិពល។
- [ ] សូមដាក់ទង់សេវាប្រព័ន្ធអេកូឡូស៊ីដែលទុកចេញពីការវាយតម្លៃជាប្រាក់ថាត្រូវបានដកចេញ មិនមែនកំណត់តម្លៃសូន្យដោយស្ងាត់ៗទេ។
- [ ] សូមរៀបលំដាប់ការវិនិយោគសម្របខ្លួនដោយផ្តល់ទម្ងន់ច្បាស់លាស់ដល់ភាពងាយរងគ្រោះសង្គម មិនមែនតែតម្លៃទ្រព្យសកម្មប៉ុណ្ណោះទេ។
- [ ] សូមរាយការណ៍ការចំណាយលើការបាត់បង់ និងការខូចខាតដោយឡែកពី ហើយមិនរាប់ជាជោគជ័យនៃការសម្របខ្លួនទេ។
- [ ] សូមសម្របសម្រួលការប្តេជ្ញាកាត់បន្ថយមូលដ្ឋានជាមួយការរួមចំណែកកំណត់ជាតិក្រោមកិច្ចព្រមព្រៀងប៉ារីស។
- [ ] សូមពិនិត្យឡើងវិញការសន្មត់អាកាសធាតុនៅពីក្រោយផែនការរយៈពេលវែងសំខាន់ៗតាមវដ្តកំណត់ ធៀបនឹងការវាយតម្លៃវិទ្យាសាស្ត្របច្ចុប្បន្ន។
- [ ] សូមរាយការណ៍ហានិភ័យអាកាសធាតុរូបវន្តចំពោះទ្រព្យសកម្ម និងសេវាតាមបណ្តាញអភិបាលកិច្ចដូចគ្នានឹងហានិភ័យហិរញ្ញវត្ថុ និងប្រតិបត្តិការ។
